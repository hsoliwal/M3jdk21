cmake_minimum_required(VERSION 3.20)
project(m3_lz_compression LANGUAGES C)
find_package(JNI REQUIRED)
add_library(m3_lz_compression SHARED ${CMAKE_CURRENT_SOURCE_DIR}/../../../src/java.base/share/native/libm3compression/m3_lz_jni.c)
target_include_directories(m3_lz_compression PRIVATE ${JNI_INCLUDE_DIRS})
target_compile_features(m3_lz_compression PRIVATE c_std_11)
if(MSVC)
  target_compile_options(m3_lz_compression PRIVATE /W4 /WX)
else()
  target_compile_options(m3_lz_compression PRIVATE -Wall -Wextra -Wconversion -Wsign-conversion -Wshadow -Wpedantic -Werror)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(m3_lz_test ${CMAKE_CURRENT_SOURCE_DIR}/../../../test/jdk/jdk/internal/m3/compression/native/m3_lz_test.c ${CMAKE_CURRENT_SOURCE_DIR}/../../../src/java.base/share/native/libm3compression/m3_lz_jni.c)
  target_compile_definitions(m3_lz_test PRIVATE M3_LZ_STANDALONE=1)
  target_compile_features(m3_lz_test PRIVATE c_std_11)
  if(MSVC)
    target_compile_options(m3_lz_test PRIVATE /W4 /WX)
  else()
    target_compile_options(m3_lz_test PRIVATE -Wall -Wextra -Wconversion -Wsign-conversion -Wshadow -Wpedantic -Werror)
  endif()
  add_test(NAME m3_lz_test COMMAND m3_lz_test)
endif()
