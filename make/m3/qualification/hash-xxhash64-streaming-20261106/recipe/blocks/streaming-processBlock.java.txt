    //FLLDCIM: HPCPOP - XXH64_KERNEL_ATOMS v1: one scratch stripe through the owner's round atom (M3XxHash.round), no second kernel
    private void processBlock(byte[] data, int offset) {
        v1 = M3XxHash.round(v1, STRIPE.i64(data, offset));
        v2 = M3XxHash.round(v2, STRIPE.i64(data, offset + M3XxHash.LANE_2_OFFSET));
        v3 = M3XxHash.round(v3, STRIPE.i64(data, offset + M3XxHash.LANE_3_OFFSET));
        v4 = M3XxHash.round(v4, STRIPE.i64(data, offset + M3XxHash.LANE_4_OFFSET));
    }
