from pathlib import Path
import argparse
import hashlib
import json
import os
import subprocess

B = Path(__file__).resolve().parent
R = B / 'receiver/m3/fs'
parser = argparse.ArgumentParser()
parser.add_argument('--classes', type=Path, default=B / 'qualification-final/classes')
parser.add_argument('--out', type=Path, default=B / 'facts-final')
parser.add_argument('--java-home', type=Path, default=B / 'toolchain/jdk21')
args = parser.parse_args()
Q = args.classes
O = args.out
O.mkdir(exist_ok=True)
T = O / 'tests'
T.mkdir(exist_ok=True)
J = args.java_home / 'bin'
coordinates = [
    ('org/junit/jupiter/junit-jupiter-api', '5.10.2'),
    ('org/junit/jupiter/junit-jupiter-engine', '5.10.2'),
    ('org/junit/platform/junit-platform-commons', '1.10.2'),
    ('org/junit/platform/junit-platform-engine', '1.10.2'),
    ('org/junit/platform/junit-platform-launcher', '1.10.2'),
    ('org/opentest4j/opentest4j', '1.3.0'),
    ('org/apiguardian/apiguardian-api', '1.1.2'),
]
jars = [B / 'm2' / p / v / (Path(p).name + '-' + v + '.jar') for p, v in coordinates]
cp = os.pathsep.join(map(str, [Q, *jars]))
sources = [R / 'src/test/java/com/m3/fs' / name for name in ('MFileObservationSelfTest.java', 'MFileObservationTest.java')]
sources.append(B / 'authored/facade/verification/FacadeTestLauncher.java')
assert (Q / 'com/m3/fs/MFileDelta.class').is_file()
compile_command = [str(J / 'javac'), '--release', '21', '-Xlint:all', '-Werror', '-sourcepath', '', '-cp', cp, '-d', str(T), *map(str, sources)]
runtime_cp = str(T) + os.pathsep + cp
commands = [compile_command,
            [str(J / 'java'), '-cp', runtime_cp, 'FacadeTestLauncher', 'com.m3.fs.MFileObservationTest'],
            [str(J / 'java'), '-ea', '-cp', runtime_cp, 'com.m3.fs.MFileObservationSelfTest']]
receipts = []
for name, command in zip(('compile', 'junit', 'consumer'), commands):
    result = subprocess.run(command, text=True, stdout=subprocess.PIPE, stderr=subprocess.STDOUT)
    (O / (name + '.log')).write_text(result.stdout)
    receipts.append({'phase': name, 'command': command, 'exit_code': result.returncode})
    if result.returncode:
        print(result.stdout)
        raise SystemExit(result.returncode)
report = {'status': 'PASS', 'scope': 'Test-only compilation against final generated product classes; no production overlays',
          'junit_tests': 1, 'consumer_assertions': 106, 'commands': receipts,
          'sources': {str(s.relative_to(R)): hashlib.sha256(s.read_bytes()).hexdigest() for s in sources if s.is_relative_to(R)}}
assert 'MFILE_OBSERVATION_CHECKS=106' in (O / 'consumer.log').read_text()
(O / 'result.json').write_text(json.dumps(report, indent=2) + '\n')
print((O / 'junit.log').read_text())
print((O / 'consumer.log').read_text())
