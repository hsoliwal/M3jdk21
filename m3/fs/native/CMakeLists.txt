cmake_minimum_required(VERSION 3.20)
project(m3fs LANGUAGES C)
find_package(SQLite3 REQUIRED)
find_package(Threads REQUIRED)
option(M3FS_JNI "Build the M3JDK21 JNI bridge into m3fs" OFF)
option(M3FS_FUSE "Build the optional FUSE3 projection" OFF)
function(m3fs_warnings target)
  target_compile_features(${target} PRIVATE c_std_11)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -Wconversion -Wshadow -Wpedantic)
  endif()
endfunction()
add_library(m3fs SHARED src/m3fs.c src/m3index_semantic_core.c src/m3fs_semantic.c)
target_include_directories(m3fs PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(m3fs PRIVATE SQLite::SQLite3 Threads::Threads)
m3fs_warnings(m3fs)
if(APPLE)
  set_target_properties(m3fs PROPERTIES INSTALL_RPATH "@loader_path" BUILD_WITH_INSTALL_RPATH TRUE)
elseif(UNIX)
  set_target_properties(m3fs PROPERTIES INSTALL_RPATH "$ORIGIN" BUILD_WITH_INSTALL_RPATH TRUE)
endif()
if(M3FS_JNI)
  find_package(JNI REQUIRED)
  target_sources(m3fs PRIVATE src/m3fs_jni.c src/m3_fs_jni.c)
  target_include_directories(m3fs PRIVATE ${JNI_INCLUDE_DIRS})
endif()
add_executable(m3fs-tool src/m3fs_tool.c)
target_link_libraries(m3fs-tool PRIVATE m3fs)
m3fs_warnings(m3fs-tool)
if(M3FS_FUSE)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(FUSE3 REQUIRED IMPORTED_TARGET fuse3)
  add_executable(m3fs-fuse src/m3fs_fuse.c)
  target_link_libraries(m3fs-fuse PRIVATE m3fs PkgConfig::FUSE3)
  m3fs_warnings(m3fs-fuse)
endif()
include(CTest)
if(BUILD_TESTING)
  foreach(suite IN ITEMS core indexed snapshot snapshot-loss hotpath scrub semantic sparse metadata replace inode-identity checked-io)
    string(REPLACE "-" "_" source ${suite})
    set(target "m3fs-${suite}-test")
    add_executable(${target} "tests/m3fs_${source}_test.c")
    target_include_directories(${target} PRIVATE include)
    if(suite STREQUAL "snapshot-loss" OR suite STREQUAL "hotpath")
      # White-box suites include the actual engine to inspect bounded statement lifetimes.
      target_link_libraries(${target} PRIVATE SQLite::SQLite3 Threads::Threads)
    else()
      target_link_libraries(${target} PRIVATE m3fs SQLite::SQLite3 Threads::Threads)
    endif()
    m3fs_warnings(${target})
    add_test(NAME ${target} COMMAND ${CMAKE_COMMAND}
      -DTEST=$<TARGET_FILE:${target}>
      -DDB=${CMAKE_CURRENT_BINARY_DIR}/${target}.sqlite
      -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/run_indexed_test.cmake)
  endforeach()
endif()
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/native-path-$<CONFIG>.txt" CONTENT "$<TARGET_FILE:m3fs>\n")
