  @Test

    MIndexRegexProgram program = MIndexRegexProgram.compile(expression);
    assertTrue(program.hasAsciiStateMasks());
    assertTrue(program.asciiStateMaskPayloadBytes() > 0L);
    assertTrue(program.hasPrecomputedEpsilonClosures());
    assertTrue(program.epsilonClosurePayloadBytes() > 0L);

        List.of("", "ab", "cd", "ababx", "cdab", "xxabcdxzz", "ababy", "éabx");

    for (String sample : samples) {

          re2.matcher(sample).lookingAt(),
          program.lookingAt(sample),

    }
  }

  @Test
  void frozenAndMappedAsciiFactsDriveCompactDfaWithoutChangingTruth() throws Exception {
    String expression = "(?:ab|cd)+x?";
    Pattern jdk = Pattern.compile(expression);
    MIndexRegexProgram program = MIndexRegexProgram.compile(expression);
    IndexDictionary dictionary =
        IndexDictionary.of(
            1,
            "en",
            "dfa-facts-v1",
            List.of("ababx", "cd", "éabx", "zzabcdxzz"));

    for (int wordId = 0; wordId < dictionary.size(); wordId++) {
      IndexLexiconWordView view = new IndexLexiconWordView(dictionary, wordId);
      String text = dictionary.valueOfLocal(wordId);
      assertEquals(jdk.matcher(text).matches(), program.matches(view), text + " heap matches");
      assertEquals(jdk.matcher(text).find(), program.find(view), text + " heap find");
      boolean ascii = text.chars().allMatch(value -> value < 128);
      assertEquals(
          ascii,
          (view.characterFlags() & IndexCharacterFlags.ASCII) != 0,
          text + " heap ASCII fact");
    }

    Path image = temp.resolve("dfa-facts.sidx");
    IndexDictionaryImage.write(dictionary, image);
    MappedIndexDictionary mapped =
        MappedIndexDictionary.open(image, "en", "dfa-facts-v1");
    try {
      for (int wordId = 0; wordId < mapped.size(); wordId++) {
        IndexLexiconWordView view = new IndexLexiconWordView(mapped, wordId);
        String text = mapped.valueOfLocal(wordId);
        assertEquals(jdk.matcher(text).matches(), program.matches(view), text + " mapped matches");
        assertEquals(jdk.matcher(text).find(), program.find(view), text + " mapped find");
        boolean ascii = text.chars().allMatch(value -> value < 128);
        assertEquals(
            ascii,
            (view.characterFlags() & IndexCharacterFlags.ASCII) != 0,
            text + " mapped ASCII fact");
      }
    } finally {
      mapped.close();
    }
  }

  @Test
  void bitKernelPrecomputationIsSparseBoundedAndAssertionsUseDynamicClosure() {
    MIndexRegexProgram anchored = MIndexRegexProgram.compile("^ab(?:cd|ef)*x$");
    assertTrue(anchored.hasAsciiStateMasks());
    assertFalse(anchored.hasPrecomputedEpsilonClosures());
    assertFalse(anchored.hasAsciiDfa());
    assertTrue(anchored.matches("abcdefcdx"));
    assertFalse(anchored.matches("zabcdefcdx"));

    MIndexRegexProgram longLiteral =
        MIndexRegexProgram.tryCompile(
            "a{7000}", MIndexRegexProgram.Semantics.JDK, 8_000);
    assertNotNull(longLiteral);
    assertTrue(longLiteral.hasAsciiStateMasks());
    assertTrue(longLiteral.hasPrecomputedEpsilonClosures());
    assertTrue(longLiteral.epsilonClosurePayloadBytes() < 65_536L);
    assertFalse(longLiteral.hasAsciiDfa());
    assertTrue(longLiteral.matches("a".repeat(7000)));
    assertFalse(longLiteral.matches("a".repeat(6999)));

    String manyOptionalAtoms = "a?".repeat(5000);
    assertNull(
        MIndexRegexProgram.tryCompile(
            manyOptionalAtoms, MIndexRegexProgram.Semantics.JDK, 12_000));
    // Lowering uses three states per optional atom, one concat jump and one terminal match.
    MIndexRegexProgram budgetFallback =
        MIndexRegexProgram.tryCompile(
            manyOptionalAtoms, MIndexRegexProgram.Semantics.JDK, 15_002);
    assertNotNull(budgetFallback);
    assertEquals(15_002, budgetFallback.stateCount());
    assertTrue(budgetFallback.hasAsciiStateMasks());
    assertFalse(budgetFallback.hasPrecomputedEpsilonClosures());
    assertTrue(budgetFallback.matches(""));
    assertTrue(budgetFallback.matches("a".repeat(5000)));
  }

  @Test
  void acceptedSubsetIsDifferentiallyEquivalentAcrossGeneratedCorpus() {
    List<String> expressions =
        List.of(
            "a*",
            "ab+c",
            "(?:ab|ba)+",
            "[ab]{1,3}",
            "[^x]{0,2}a",
            "\\d+",
            "\\w{1,3}",
            "\\s?",
            "\\A(?:ab|a)*\\z",
            "a?b?x?",
            "[a-c0-2]+");

    List<String> corpus = new ArrayList<>();
    generateCorpus(corpus, new char[] {'a', 'b', 'x', '1', ' ', '\n'}, new StringBuilder(), 4);

    for (String expression : expressions) {
      MIndexRegexProgram program =
          MIndexRegexProgram.tryCompile(
              expression, MIndexRegexProgram.Semantics.JDK, 10_000);
      assertNotNull(program, expression);
      Pattern jdk = Pattern.compile(expression);
      for (String sample : corpus) {
        boolean expectedMatches = jdk.matcher(sample).matches();
        boolean expectedFind = jdk.matcher(sample).find();
        boolean expectedLookingAt = jdk.matcher(sample).lookingAt();
        assertEquals(
            expectedMatches,
            program.matches(sample),
            expression + " matches " + printable(sample));
        assertEquals(
            expectedFind,
            program.find(sample),
            expression + " find " + printable(sample));
        assertEquals(
            expectedLookingAt,
            program.lookingAt(sample),
            expression + " lookingAt " + printable(sample));
        if (expectedMatches || expectedFind || expectedLookingAt) {
          assertTrue(
              MIndexStringBitOps.mayContain(
                  MIndexStringBitOps.signal(sample), program.requiredPresence64()),
              expression + " presence proof " + printable(sample));
        }
      }
    }
  }

  @Test
  void astPresenceProofIsConservativeAcrossRegexAlgebra() {
    long abx = MIndexStringBitOps.signal("abx");
    long aby = MIndexStringBitOps.signal("aby");
    long z = MIndexStringBitOps.signal("z");

    MIndexRegexProgram repeatedAlternative =
        MIndexRegexProgram.compile("(?:abx|aby)+z");
    assertEquals((abx & aby) | z, repeatedAlternative.requiredPresence64());

    MIndexRegexProgram optionalAlternative =
        MIndexRegexProgram.compile("(?:abx|aby)?z");
    assertEquals(z, optionalAlternative.requiredPresence64());

    long classCommon =
        MIndexStringBitOps.signal("a") & MIndexStringBitOps.signal("b");
    MIndexRegexProgram classThenLiteral =
        MIndexRegexProgram.compile("[ab]x");
    assertEquals(
        classCommon | MIndexStringBitOps.signal("x"),
        classThenLiteral.requiredPresence64());

    MIndexRegexProgram sharedGroup =
        MIndexRegexProgram.compile("(?:prefix-a|prefix-b)-tail");
    assertTrue(
        MIndexStringBitOps.mayContain(
            MIndexStringBitOps.signal("prefix-a-tail"),
            sharedGroup.requiredPresence64()));
    assertTrue(
        MIndexStringBitOps.mayContain(
            MIndexStringBitOps.signal("prefix-b-tail"),
            sharedGroup.requiredPresence64()));
  }

    MIndexRegexProgram whitespace = MIndexRegexProgram.compile("\\s+");
    assertEquals(
        IndexCharacterFlags.HAS_ASCII_SPACE,

    MIndexRegexProgram sharedWord =
        MIndexRegexProgram.compile("(?:\\d+|[A-Z]+)");

        IndexCharacterFlags.HAS_ASCII_WORD,

    assertEquals(
        0,
        MIndexRegexProgram.compile("(?:\\d+|!+)").requiredCharacterFlags());
    assertEquals(0, MIndexRegexProgram.compile("\\d*").requiredCharacterFlags());
  }

  @Test
  void resolverFamiliesExposeConsistentPrecomputedRegexFacts() {
    IndexDictionary dictionary =
        IndexDictionary.of(5, "en", "facts-v1", List.of("letters_only", "12345"));
    int lettersLocal = dictionary.lookupLocal("letters_only");
    long expectedSignal = dictionary.bitSignalOfLocal(lettersLocal);
    int expectedFlags = dictionary.characterFlagsOfLocal(lettersLocal);

    IndexSpace space =
        IndexSpace.builder()
            .add(dictionary, IndexAffixIndex.freeze(dictionary))
            .build();
    long spaceId = space.intern(5, "letters_only");
    assertEquals(expectedSignal, space.bitSignal(spaceId));
    assertEquals(expectedFlags, space.characterFlags(spaceId));

    HierarchicalIndexResolver hierarchical =
        HierarchicalIndexResolver.builder().dictionary(dictionary).build();
    long hierarchicalId = hierarchical.intern(5, "letters_only");
    assertEquals(expectedSignal, hierarchical.bitSignal(hierarchicalId));
    assertEquals(expectedFlags, hierarchical.characterFlags(hierarchicalId));

    long literalId = hierarchical.intern(5, "literal_only");
    assertEquals(
        IndexCharacterFlags.of("literal_only"),
        hierarchical.characterFlags(literalId));
    assertEquals(
        MIndexStringBitOps.signal("literal_only"),
        hierarchical.bitSignal(literalId));

    IndexExactLexicon exact =
        IndexExactLexicon.builder("en")
            .entry(IndexLexemeKind.ACRONYM, "ABC")
            .freeze();
    IndexExactLexiconResolver exactResolver = new IndexExactLexiconResolver(5, exact);
    long exactId = exactResolver.intern(5, "ABC");
    assertEquals(MIndexStringBitOps.signal("ABC"), exactResolver.bitSignal(exactId));
    assertEquals(IndexCharacterFlags.of("ABC"), exactResolver.characterFlags(exactId));

    MIndexRegexProgram digits = MIndexRegexProgram.compile("\\d+");
    assertFalse(digits.matches(MIndexString.intern(space, 5, "letters_only")));
    assertFalse(digits.matches(MIndexString.intern(hierarchical, 5, "letters_only")));
    assertFalse(digits.matches(MIndexString.intern(exactResolver, 5, "ABC")));
  }

  @Test

    List<String> expressions =
        List.of(

            "a*?b",

            "\\Qfoo.bar[1]\\E\\d+",

        List.of(
            "",
            "ab",

            "xxfoo.bar[1]7yy",
            "ab",

    for (String expression : expressions) {
      MIndexRegexProgram program =
          MIndexRegexProgram.tryCompile(

      assertNotNull(program, expression);
      MIndexHybridRegex hybrid = MIndexHybridRegex.compile(expression);
      MIndexRegexBackend expectedBackend =
          expression.equals("\\Qunterminated.*")
              ? MIndexRegexBackend.LITERAL_INDEX
              : MIndexRegexBackend.MINDEX_AUTOMATON;
      assertEquals(expectedBackend, hybrid.backend(), expression);
      com.google.re2j.Pattern re2 = com.google.re2j.Pattern.compile(expression);
      MIndexRegexProgram jdkProgram =
          MIndexRegexProgram.tryCompile(
              expression, MIndexRegexProgram.Semantics.JDK, 10_000);
      assertNotNull(jdkProgram, expression + " JDK subset");
      Pattern jdk = Pattern.compile(expression);

      for (String sample : samples) {

            re2.matcher(sample).matches(),

            expression + " RE2 matches " + printable(sample));

            expression + " RE2 find " + printable(sample));
        assertEquals(
            re2.matcher(sample).lookingAt(),
            program.lookingAt(sample),
            expression + " RE2 lookingAt " + printable(sample));
        assertEquals(
            jdk.matcher(sample).matches(),
            jdkProgram.matches(sample),
            expression + " JDK matches " + printable(sample));
        assertEquals(
            jdk.matcher(sample).find(),
            jdkProgram.find(sample),

  @Test
  void supplementaryInputDelegatesOnlyWhenAnAtomCanConsumeSurrogateUnits() {
    List<String> transparentExpressions =
        List.of(
            "abc",
            "[a-z]+",
            "\\d+",
            "\\w+",
            "\\s+",
            "(?:abc|\\d+)+",
            "a*");
    List<String> supplementarySamples =
        List.of(
            "😀abc",
            "abc😀",
            "😀123",
            "😀 abc",
            "x😀abc123");

    ExactStringIndexResolver resolver = new ExactStringIndexResolver();
    for (String expression : transparentExpressions) {
      MIndexRegexProgram program = MIndexRegexProgram.compile(expression);
      com.google.re2j.Pattern re2 = com.google.re2j.Pattern.compile(expression);
      assertFalse(program.mayConsumeSurrogateUnit(), expression);
      for (String sample : supplementarySamples) {
        assertTrue(program.supportsInput(sample), expression + " supports String");
        MIndexString indexed = MIndexString.intern(resolver, 0, sample);
        assertTrue(program.supportsInput(indexed), expression + " supports MIndexString");
        assertEquals(
            re2.matcher(sample).matches(),
            program.matches(indexed),
            expression + " matches " + sample);
        assertEquals(
            re2.matcher(sample).find(),
            program.find(indexed),
            expression + " find " + sample);
        assertEquals(
            re2.matcher(sample).lookingAt(),
            program.lookingAt(indexed),
            expression + " lookingAt " + sample);
      }
    }

    List<String> sensitiveExpressions =
        List.of(".", "a.", "[^x]+", "\\D+", "[\\uD7FF-\\uE000]");
    for (String expression : sensitiveExpressions) {
      MIndexRegexProgram program = MIndexRegexProgram.compile(expression);
      assertTrue(program.mayConsumeSurrogateUnit(), expression);
      assertFalse(program.supportsInput("😀"), expression);
      assertThrows(IllegalArgumentException.class, () -> program.matches("😀"), expression);
    }

    MIndexHybridRegex hybrid = MIndexHybridRegex.compile(".");
    assertEquals(MIndexRegexBackend.MINDEX_AUTOMATON, hybrid.backend());
    assertTrue(hybrid.matches("😀"));
  }

  @Test
  void hybridUsesNativeBackendForNoCaptureSubsetAndRe2ForCaptures() {
    MIndexHybridRegex nativeRegex = MIndexHybridRegex.compile("cat[a-z]+\\d{2,4}");
    assertEquals(MIndexRegexBackend.MINDEX_AUTOMATON, nativeRegex.backend());

    assertTrue(captureRegex.find("xx name-123 yy"));
  }

  @Test
  void mindexStringMatchesPreservesJdkSemanticsWithoutRequiringStringInputCopy() {
    ExactStringIndexResolver resolver = new ExactStringIndexResolver();
    List<String> samples = List.of("dogma123", "foo\n", "aa", "😀");
    List<String> expressions =
        List.of("(?:dog|cat)[a-z]*\\d{2,4}", "foo$", "(a)\\1", ".");

    for (String sample : samples) {
      MIndexString indexed = MIndexString.intern(resolver, 0, sample);
      for (String expression : expressions) {
        assertEquals(
            sample.matches(expression),
            indexed.matches(expression),
            sample + " / " + expression);
      }
    }
  }
  private static void generateCorpus(
      List<String> out, char[] alphabet, StringBuilder value, int remaining) {
    out.add(value.toString());
    if (remaining == 0) return;
    for (char unit : alphabet) {
      value.append(unit);
      generateCorpus(out, alphabet, value, remaining - 1);
      value.setLength(value.length() - 1);
    }
  }

  private static String printable(String value) {
    return value.replace("\n", "\\n");
  }
