import java.util.concurrent.ProgressJob;
import java.util.concurrent.ProgressMonitor;

/* Intentionally fails: this new Callable base is not also a Runnable base. */
final class ProgressJobRunnableConflict extends ProgressJob<String> implements Runnable {
    @Override protected String run(ProgressMonitor monitor) { return "status"; }
    @Override public void run() { }
}

/* Widening the hook to public does not fix the incompatible return type. */
final class ProgressJobPublicRunnableConflict extends ProgressJob<String> implements Runnable {
    @Override public String run(ProgressMonitor monitor) { return "status"; }
    @Override public void run() { }
}
