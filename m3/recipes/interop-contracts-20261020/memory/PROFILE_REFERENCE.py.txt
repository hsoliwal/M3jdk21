def canonical(value):
    if value is None:
        return 'null'
    if type(value) is bool:
        return 'true' if value else 'false'
    if type(value) is int:
        if not -(1 << 63) <= value < (1 << 63):
            raise ValueError('integer outside signed 64-bit range')
        return str(value)
    if isinstance(value, str):
        result = []
        for char in value:
            code = ord(char)
            if 0xD800 <= code <= 0xDFFF:
                raise ValueError('unpaired surrogate')
            if code < 32:
                result.append('\\u%04x' % code)
            elif char == '"':
                result.append('\\"')
            elif char == '\\':
                result.append('\\\\')
            else:
                result.append(char)
        return '"' + ''.join(result) + '"'
    if isinstance(value, list):
        return '[' + ','.join(map(canonical, value)) + ']'
    if isinstance(value, dict):
        if any(not isinstance(k, str) or not k.isascii() for k in value):
            raise ValueError('non-ASCII schema key')
        return '{' + ','.join(canonical(k) + ':' + canonical(value[k]) for k in sorted(value)) + '}'
    raise ValueError('unsupported JSON value')

def raw(value):
    return canonical(value).encode('utf-8')

def schema_check(value, schema, path='$'):
    if isinstance(schema, dict):
        if not isinstance(value, dict) or set(value) != set(schema):
            raise ValueError('missing or unknown field at ' + path)
        for key in schema:
            schema_check(value[key], schema[key], path + '.' + key)
    elif isinstance(schema, list):
        if not isinstance(value, list):
            raise ValueError('not an array at ' + path)
        for item in value:
            schema_check(item, schema[0], path + '[]')
    else:
        allowed = schema if isinstance(schema, tuple) else (schema,)
        if type(value) not in allowed:
            raise ValueError('invalid type at ' + path)

def walk_artifacts(value):
    if isinstance(value, dict):
        if set(value) == set(ART):
            yield value
        else:
            for child in value.values():
                yield from walk_artifacts(child)
    elif isinstance(value, list):
        for child in value:
            yield from walk_artifacts(child)

def json_schema(schema):
    if isinstance(schema, dict):
        return {'type': 'object', 'additionalProperties': False, 'required': sorted(schema), 'properties': {k: json_schema(v) for k, v in sorted(schema.items())}}
    if isinstance(schema, list):
        return {'type': 'array', 'items': json_schema(schema[0])}
    mapping = {str: 'string', bool: 'boolean', type(None): 'null'}
    return {'type': [mapping[s] for s in schema]} if isinstance(schema, tuple) else {'type': mapping[schema]}
