 * @test
 * @summary Independent single-byte CharsetEncoder oracle over actual M3 tuple/range Strings
 * @build M3StringUtf16DifferentialCorpus
 * @run main/othervm --add-opens=java.base/java.lang=ALL-UNNAMED -XX:+UnlockExperimentalVMOptions -XX:+UseM3StringStorage -XX:+CompactStrings M3StringSingleByteCursorTest
 * @run main/othervm --add-opens=java.base/java.lang=ALL-UNNAMED -XX:+UnlockExperimentalVMOptions -XX:+UseM3StringStorage -XX:-CompactStrings M3StringSingleByteCursorTest
 */
import java.lang.reflect.Field;
import java.nio.ByteBuffer;
import java.nio.CharBuffer;
import java.nio.charset.CharacterCodingException;
import java.nio.charset.Charset;
import java.nio.charset.CodingErrorAction;
import java.nio.charset.StandardCharsets;
import java.util.Arrays;

public class M3StringSingleByteCursorTest {
    private static long checks;

    public static void main(String[] args) throws ReflectiveOperationException {
        Field storage = String.class.getDeclaredField("m3");
        storage.setAccessible(true);
        int index = 0;
        for (char[] value : M3StringUtf16DifferentialCorpus.values()) {
            verify(value, M3StringUtf16DifferentialCorpus.composed(value), storage, "case-" + index++);
        }
        for (char[] value : new char[][] {
                {'\u007f', '\u0080', '\u00ff', '\u0100'},
                {'A', '\ud800', '\udc00', 'B'},
                {'\ud800', '\ud800', '\udc00', '\udc00'},
                {'\udc00', '\ud800'}, {'\ud800'}, {'\udc00'}, {}}) {
            verify(value, M3StringUtf16DifferentialCorpus.composed(value), storage, "edge-" + index++);
        }
        char[] longValue = new char[4097];
        Arrays.fill(longValue, 'a');
        longValue[1023] = '\ud800';
        longValue[1024] = '\udc00';
        longValue[2048] = '\u0100';
        longValue[4096] = '\ud800';
        String seam = M3StringUtf16DifferentialCorpus.fresh(Arrays.copyOfRange(longValue, 0, 1024))
                .concat(M3StringUtf16DifferentialCorpus.fresh(Arrays.copyOfRange(longValue, 1024, longValue.length)));
        Object m3 = storage.get(seam);
        check(m3 != null, "directed seam must use actual M3 backing");
        Field owner = m3.getClass().getDeclaredField("owner");
        owner.setAccessible(true);
        check(owner.get(m3).getClass().getName().equals("java.lang.M3StringTuple"),
                "directed seam must traverse the actual tuple owner");
        verify(longValue, seam, storage, "long-surrogate-seam");
        System.out.println("M3_STRING_SINGLE_BYTE_CURSOR_PASS|cases=" + index + "|checks=" + checks);
    }

    private static void verify(char[] value, String composed, Field storage, String label)
            throws IllegalAccessException {
        String flat = M3StringUtf16DifferentialCorpus.fresh(value);
        check(value.length == 0 || storage.get(composed) != null, label + " actual M3 backing");
        compare(flat, composed, value, label);
        int[] bounds = value.length <= 8 ? allBounds(value.length)
                : new int[] {0, 1, value.length / 3, value.length / 2, value.length - 1, value.length};
        for (int start : bounds) {
            for (int end : bounds) {
                if (end < start) continue;
                compare(flat.substring(start, end), composed.substring(start, end),
                        Arrays.copyOfRange(value, start, end), label + " slice " + start + ":" + end);
            }
        }
    }

    private static int[] allBounds(int length) {
        int[] result = new int[length + 1];
        for (int i = 0; i <= length; i++) result[i] = i;
        return result;
    }

    private static void compare(String flat, String composed, char[] original, String label) {
        for (Charset charset : new Charset[] {StandardCharsets.US_ASCII, StandardCharsets.ISO_8859_1}) {
            byte[] expected = oracle(original, charset);
            check(Arrays.equals(expected, flat.getBytes(charset)), label + " flat " + charset.name());
            check(Arrays.equals(expected, composed.getBytes(charset)), label + " tuple " + charset.name());
            check(Arrays.equals(original, flat.toCharArray()), label + " flat unchanged");
            check(Arrays.equals(original, composed.toCharArray()), label + " tuple unchanged");
        }
    }

    private static byte[] oracle(char[] original, Charset charset) {
        try {
            ByteBuffer encoded = charset.newEncoder()
                    .onMalformedInput(CodingErrorAction.REPLACE)
                    .onUnmappableCharacter(CodingErrorAction.REPLACE)
                    .encode(CharBuffer.wrap(original));
            byte[] result = new byte[encoded.remaining()];
            encoded.get(result);
            return result;
        } catch (CharacterCodingException failure) {
            throw new AssertionError("replacement encoder refused " + charset.name(), failure);
        }
    }

    private static void check(boolean condition, String message) {
        checks++;
        if (!condition) throw new AssertionError(message);
    }
}
