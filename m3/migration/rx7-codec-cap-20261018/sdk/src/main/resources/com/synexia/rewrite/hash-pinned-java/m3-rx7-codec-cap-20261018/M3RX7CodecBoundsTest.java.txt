/*
 * @test
 * @summary Reject oversized RX7 byte arrays before taking a defensive snapshot
 * @modules java.base/jdk.internal.mindex
 * @run main/othervm -Xmx96m -Xint M3RX7CodecBoundsTest
 */

import java.lang.ref.Reference;
import java.nio.ByteBuffer;
import jdk.internal.mindex.M3RX7Codec;

/** A bounded-heap witness. OutOfMemoryError is never an acceptable decoder refusal. */
public final class M3RX7CodecBoundsTest {
    private static final int HEADER_BYTES = 44;
    private static final int LIMIT = 64 * 1024 * 1024;

    private M3RX7CodecBoundsTest() {}

    public static void main(String[] arguments) {
        expectNull(() -> M3RX7Codec.decodePrecomputed((byte[]) null));
        expectNull(() -> M3RX7Codec.decodePrecomputed((ByteBuffer) null));
        for (int length : new int[]{0, 7, HEADER_BYTES - 1}) {
            expectGeometry(() -> M3RX7Codec.decodePrecomputed(new byte[length]));
        }
        expectMessage("RX7 image magic",
                () -> M3RX7Codec.decodePrecomputed(new byte[HEADER_BYTES]));

        // Allocate before invoking the decoder. A missing marker means this is not the
        // intended baseline failure; the launcher also requires a clone-specific stack.
        byte[] oversized = new byte[LIMIT + 1];
        System.out.println("M3RX7CodecBoundsTest INPUT_READY bytes=" + oversized.length);
        System.out.flush();
        try {
            expectGeometry(() -> M3RX7Codec.decodePrecomputed(oversized));
            // Geometry must win even if the v7 header is otherwise malformed.
            ByteBuffer.wrap(oversized).putInt(0x534e5852).putInt(Integer.MAX_VALUE);
            expectGeometry(() -> M3RX7Codec.decodePrecomputed(oversized));
            ByteBuffer positioned = ByteBuffer.wrap(oversized);
            int position = positioned.position();
            int limit = positioned.limit();
            expectGeometry(() -> M3RX7Codec.decodePrecomputed(positioned));
            if (positioned.position() != position || positioned.limit() != limit) {
                throw new AssertionError("caller buffer geometry changed");
            }
        } finally {
            Reference.reachabilityFence(oversized);
        }
        System.out.println("M3RX7CodecBoundsTest PASS");
    }

    private static void expectGeometry(Runnable action) {
        expectMessage("RX7 v7 image byte geometry", action);
    }

    private static void expectMessage(String expected, Runnable action) {
        try {
            action.run();
        } catch (IllegalArgumentException failure) {
            if (expected.equals(failure.getMessage())) return;
            throw new AssertionError("unexpected decoder refusal", failure);
        }
        throw new AssertionError("missing decoder refusal: " + expected);
    }

    private static void expectNull(Runnable action) {
        try {
            action.run();
        } catch (NullPointerException failure) {
            if ("encoded".equals(failure.getMessage())) return;
            throw new AssertionError("unexpected null contract", failure);
        }
        throw new AssertionError("missing encoded null refusal");
    }
}
