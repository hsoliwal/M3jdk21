class FakeResponse(io.BytesIO):
    def __init__(self, payload, url, headers=None, status=200):
        super().__init__(payload)
        self.url = url
        self.headers = headers or {}
        self.status = status

    def geturl(self):
        return self.url


class FakeOpener:
    def __init__(self, response):
        self.response = response
        self.requests = []

    def open(self, request, timeout):
        self.requests.append((request.full_url, timeout))
        return self.response


class BootstrapBoundaryTest(unittest.TestCase):
    def setUp(self):
        self.assertIsNotNone(TEST_OUTPUT, "use bootstrap.py --self-test --output")
        self.temp = tempfile.TemporaryDirectory(dir=Path(TEST_OUTPUT) / "tmp")
        self.addCleanup(self.temp.cleanup)
        self.root = Path(self.temp.name)
        self.lock = bootstrap.load_lock(LOCK_PATH)

    def output(self, name="owned"):
        out = self.root / name
        out.mkdir()
        for child in ("logs", "tmp", "home", "wheelhouse"):
            (out / child).mkdir()
        return out

    def assert_refused(self, call):
        with self.assertRaises(bootstrap.Refusal):
            call()

    def wheel(self, mutations=None, extra=None, duplicate=None, record_bad=False):
        item = copy.deepcopy(self.lock["artifacts"][1])
        item["requiresDist"] = []
        item["coreMetadataSha256"] = None
        license_bytes = b"fixture-license\n"
        item["license"] = {"expression": "MIT", "sourcePath": "LICENSE",
                          "sha256": bootstrap.digest(license_bytes),
                          "bytes": len(license_bytes)}
        info = "jsonschema-4.26.0.dist-info/"
        files = {
            "jsonschema/__init__.py": b"VALUE = 1\n",
            info + "METADATA": b"Metadata-Version: 2.4\nName: jsonschema\nVersion: 4.26.0\nRequires-Python: >=3.10\n\n",
            info + "WHEEL": b"Wheel-Version: 1.0\nGenerator: fixture\nRoot-Is-Purelib: true\nTag: py3-none-any\n\n",
            info + "licenses/LICENSE": license_bytes,
        }
        files.update(mutations or {})
        if extra:
            files.update(extra)
        rows = []
        for name, payload in files.items():
            encoded = bootstrap.base64.urlsafe_b64encode(
                hashlib.sha256(payload).digest()).rstrip(b"=").decode("ascii")
            rows.append([name, "sha256=" + encoded, str(len(payload))])
        rows.append([info + "RECORD", "", ""])
        if record_bad:
            rows[0][1] = "sha256=wrong"
        record = io.StringIO()
        csv.writer(record, lineterminator="\n").writerows(rows)
        files[info + "RECORD"] = record.getvalue().encode()
        path = self.root / ("case-" + str(len(list(self.root.glob("case-*.whl")))) + ".whl")
        with zipfile.ZipFile(path, "w", compression=zipfile.ZIP_DEFLATED) as archive:
            for name, payload in files.items():
                if isinstance(name, zipfile.ZipInfo):
                    archive.writestr(name, payload)
                else:
                    archive.writestr(name, payload)
            if duplicate:
                archive.writestr(duplicate, b"duplicate")
        data = path.read_bytes()
        item["bytes"] = len(data)
        item["sha256"] = bootstrap.digest(data)
        return path, item

    def test_lock_missing_duplicate_extra_and_reordered_distributions_refused(self):
        for mode in ("missing", "duplicate", "extra", "reordered"):
            item = copy.deepcopy(self.lock)
            if mode == "missing":
                item["artifacts"].pop()
            elif mode == "duplicate":
                item["artifacts"][1] = item["artifacts"][0]
            elif mode == "extra":
                item["artifacts"].append(item["artifacts"][0])
            else:
                item["artifacts"].reverse()
            with self.subTest(mode=mode):
                self.assert_refused(lambda: bootstrap.validate_lock(item))

    def test_embedded_genuine_probe_drivers_have_valid_syntax_and_real_discriminators(self):
        for name in ("PROBE", "EMPTY_SITE_PROBE", "PROVIDER_PROBE", "UNIT_DRIVER"):
            with self.subTest(driver=name):
                ast.parse(getattr(bootstrap, name), filename=name)
        tree = ast.parse(bootstrap.PROVIDER_PROBE)
        literals = {}
        for node in tree.body:
            if isinstance(node, ast.Assign) and len(node.targets) == 1 \
                    and isinstance(node.targets[0], ast.Name) \
                    and node.targets[0].id in ("tuple_schema", "unevaluated_schema"):
                literals[node.targets[0].id] = ast.literal_eval(node.value)
        self.assertEqual([{"type": "integer"}, {"type": "string"}],
                         literals["tuple_schema"]["prefixItems"])
        self.assertIs(literals["tuple_schema"]["items"], False)
        self.assertIs(literals["unevaluated_schema"]["unevaluatedProperties"], False)
        calls = [node for node in ast.walk(tree) if isinstance(node, ast.Call)]
        self.assertTrue(any(isinstance(n.func, ast.Name) and
                            n.func.id == "HashTrieMap" for n in calls))
        self.assertTrue(any(isinstance(n.func, ast.Attribute) and
                            n.func.attr == "check_schema" for n in calls))
        self.assertIn("loaded_native", {n.id for n in ast.walk(tree)
                                       if isinstance(n, ast.Name)})

            _get_packages=lambda: {"pip": package}, version=lambda: "25.0.1")

        selector = namespace["inspect_selected_ensurepip"]
