    /** Returns the preserved mapping for one exact Synexia source identity. */
    public Optional<SourceMapping> findMapping(String sourceId, String recordId) {
        lifecycle.readLock().lock();
        try {
            return Optional.ofNullable(mappingValueWhileLocked(sourceId, recordId));
        } finally { lifecycle.readLock().unlock(); }
    }

    /** Resolves the checked immutable value; the caller owns the lifecycle read lock. */
    private SourceMapping mappingValueWhileLocked(String sourceId, String recordId) {
        ensureOpen();
        Objects.requireNonNull(sourceId);
        Objects.requireNonNull(recordId);
        SourceMapping mapping = identityMapping(sourceId, recordId);
        if (mapping == null) return null;
        imageAt(mapping.coordinate().shardId());
        return mapping;
    }
