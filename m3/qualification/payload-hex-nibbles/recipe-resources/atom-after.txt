    private static StringBuilder appendString(StringBuilder out, String value) {
        out.append('"');
        for (int at = 0; at < value.length(); at++) {
            char current = value.charAt(at);
            switch (current) {
                case '"' -> out.append("\\\"");
                case '\\' -> out.append("\\\\");
                case '\b' -> out.append("\\b");
                case '\f' -> out.append("\\f");
                case '\n' -> out.append("\\n");
                case '\r' -> out.append("\\r");
                case '\t' -> out.append("\\t");
                default -> {
                    if (current < 0x20 || current > 0x7f) {
                        // Jackson's nibble-emission pattern; retain M3's lowercase escape policy.
                        String hex = "0123456789abcdef";
                        out.append("\\u")
                                .append(hex.charAt(current >>> 12))
                                .append(hex.charAt((current >>> 8) & 0xf))
                                .append(hex.charAt((current >>> 4) & 0xf))
                                .append(hex.charAt(current & 0xf));
                    }
                    else out.append(current);
                }
            }
        }
        return out.append('"');
    }
