name: M3 Eclipse framework indexed forward receiver
on:
  pull_request:
    branches: [master]
    paths:
      - 'm3/eclipse/framework/**'
      - 'm3/collections/src/main/java/**'
      - 'm3/tooling/migration-recipes/src/main/java/com/synexia/rewrite/M3HashPinnedJavaSnapshotRecipe.java'
      - '.github/workflows/m3-eclipse-framework-forward-20261017.yml'
  workflow_dispatch:
permissions:
  contents: read
jobs:
  eclipse-framework-forward:
    runs-on: ubuntu-latest
    timeout-minutes: 35
    steps:
      - uses: actions/checkout@v4
        with:
          ref: ${{ github.event.pull_request.head.sha || github.sha }}
          sparse-checkout: |
            m3/eclipse/framework
            m3/collections
            m3/tooling/migration-recipes
      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '21'
          cache: maven
      - name: Bind five exact recipe-generated candidate postimages
        shell: bash
        run: |
          set -euxo pipefail
          mkdir -p proof
          git rev-parse HEAD > proof/head.txt
          python3 - <<'PY'
          from pathlib import Path
          import hashlib
          import subprocess
          root = Path('m3/eclipse/framework/recipe/src/main/resources/com/synexia/rewrite/hash-pinned-java/eclipse-framework-m3-index-forward')
          rows = [r.split('\t') for r in (root/'manifest.tsv').read_text().splitlines() if r and not r.startswith('#')]
          assert len(rows) == 8
          assert [r[0] for r in rows] == sorted(r[0] for r in rows)
          tracked = set(subprocess.check_output(['git', 'ls-files', 'm3/eclipse/framework/src'], text=True).splitlines())
          assert tracked == {r[0] for r in rows}, tracked ^ {r[0] for r in rows}
          for path, before, wanted, source in rows:
              assert before == 'ABSENT'
              for p in (Path(path),root/source):
                  assert p.is_file() and not p.is_symlink(), p
                  got = hashlib.sha256(p.read_bytes()).hexdigest()
                  assert got == wanted, (p,got,wanted)
          print('EXACT_EIGHT_JAVA_POSTIMAGES')
          PY
      - name: Execute canonical source-sealed OpenRewrite recipe tests
        run: |
          set -euxo pipefail
          mvn -B -ntp -f m3/eclipse/framework/recipe/pom.xml test 2>&1 | tee proof/recipe.txt
      - name: Compile existing M3 collections and install exact jar
        run: |
          set -euxo pipefail
          mkdir -p proof/classes
          find m3/collections/src/main/java -name '*.java' | sort > proof/m3-sources.txt
          javac --release 21 -Xlint:all,-module -Werror -d proof/classes @proof/m3-sources.txt 2>&1 | tee proof/m3-core-compile.txt
          jar --create --file proof/m3-collections.jar -C proof/classes .
          jar --create --file proof/m3-collections-osgi.jar --manifest m3/eclipse/framework/osgi/m3-collections.MF -C proof/classes .
          jar --describe-module --file proof/m3-collections-osgi.jar > proof/core-module-describe.txt
          mvn -B -ntp org.apache.maven.plugins:maven-install-plugin:3.1.3:install-file \
            -Dfile="$GITHUB_WORKSPACE/proof/m3-collections.jar" \
            -DgroupId=com.m3 -DartifactId=m3-collections -Dversion=0.1.0-SNAPSHOT \
            -Dpackaging=jar -DgeneratePom=true
      - name: Compile and execute real Eclipse API tests
        run: |
          set -euxo pipefail
          mvn -B -ntp -f m3/eclipse/framework/pom.xml verify 2>&1 | tee proof/eclipse-maven.txt
          python3 - <<'PY'
          from pathlib import Path
          import xml.etree.ElementTree as E
          reports=list(Path('m3/eclipse/framework/target/surefire-reports').glob('TEST-*.xml'))
          assert reports, 'no JUnit reports'
          roots=[E.parse(p).getroot() for p in reports]
          n={k:sum(int(v.attrib.get(k,0)) for v in roots) for k in ('tests','errors','failures','skipped')}
          assert n=={'tests':6,'errors':0,'failures':0,'skipped':0},n
          print('ECLIPSE_API_OSGI_SERVICE_JUNIT_TESTS',n)
          PY
          unzip -p m3/eclipse/framework/target/m3-eclipse-framework-forward-0.1.0-SNAPSHOT.jar META-INF/MANIFEST.MF > proof/eclipse-bundle-manifest.txt
          grep -q 'Bundle-Activator: com.m3.eclipse.framework.M3EclipseRuntimeActivator' proof/eclipse-bundle-manifest.txt
          unzip -p proof/m3-collections-osgi.jar META-INF/MANIFEST.MF > proof/collections-bundle-manifest.txt
          grep -q 'Export-Package: com.m3.collections' proof/collections-bundle-manifest.txt
          git diff --exit-code -- m3/eclipse/framework/src m3/collections/src/main/java
      - name: Retain exact-head evidence including failures
        uses: actions/upload-artifact@v4
        if: always()
        with:
          name: eclipse-m3-framework-${{ github.event.pull_request.head.sha || github.sha }}
          retention-days: 30
          path: |
            proof/
            m3/eclipse/framework/recipe/target/surefire-reports/
            m3/eclipse/framework/target/surefire-reports/
