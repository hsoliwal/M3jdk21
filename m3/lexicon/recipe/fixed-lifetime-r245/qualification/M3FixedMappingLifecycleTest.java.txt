  var buffer=owner.readOnlyBuffer();assertTrue(buffer.isDirect());assertTrue(buffer.isReadOnly());var sha=MessageDigest.getInstance("SHA-256");sha.update(buffer);assertEquals(hash,HexFormat.of().formatHex(sha.digest()));assertThrows(ReadOnlyBufferException.class,()->owner.readOnlyBuffer().put(0,(byte)7));
 }
 public static final class BrokenDigest extends MessageDigestSpi {
  public BrokenDigest(){}
  static void fail(){Throwable f=digestFailure;if(f instanceof Error e)throw e;throw (RuntimeException)f;}
  @Override protected void engineUpdate(byte value){fail();}
  @Override protected void engineUpdate(byte[] value,int offset,int length){fail();}

 }

  private static final long serialVersionUID=1L;

 }
 static void injectedDigest(int slot,Throwable failure)throws Exception{

 }
 @Test void missingProviderReleasesVerifiedMapping()throws Exception{
