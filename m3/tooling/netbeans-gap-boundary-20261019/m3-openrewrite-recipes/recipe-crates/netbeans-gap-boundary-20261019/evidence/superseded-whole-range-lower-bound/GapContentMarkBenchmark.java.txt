package javax.swing.text;

import java.lang.management.ManagementFactory;
import java.lang.reflect.Field;

/** Complete-call timing plus separate test-only counted mark-vector observations. */
public final class GapContentMarkBenchmark {
    private static volatile int sink;
    private GapContentMarkBenchmark() { }

    @SuppressWarnings("serial")
    private static final class Counted extends GapContent.MarkVector {
        private long reads;
        private Counted(int size) { super(size + 2); }
        public GapContent.MarkData elementAt(int index) {
            reads++;
            return super.elementAt(index);
        }
    }

    private static void counted() throws Exception {
        int[] sizes = {0, 1, 2, 64, 4096, 65536};
        Field field = GapContent.class.getDeclaredField("marks");
        field.setAccessible(true);
        for (int size : sizes) {
            GapContent content = new GapContent();
            Counted marks = new Counted(size);
            for (int i = 0; i < size; i++) marks.addElement(content.new MarkData(4));
            field.set(content, marks);
            marks.reads = 0;
            int answer = content.findMarkAdjustIndex(4);
            if (answer != 0) throw new AssertionError(answer);
            System.out.println("probes," + size + "," + marks.reads);
        }
    }

    private static long call(GapContent content, int key, int count) {
        int value = 0;
        long start = System.nanoTime();
        for (int i = 0; i < count; i++) value += content.findMarkAdjustIndex(key);
        long end = System.nanoTime();
        sink = value;
        return end - start;
    }

    public static void main(String[] arguments) throws Exception {
        if (arguments.length > 0 && arguments[0].equals("counts")) {
            counted();
            return;
        }
        com.sun.management.ThreadMXBean memory =
                (com.sun.management.ThreadMXBean) ManagementFactory.getThreadMXBean();
        if (!memory.isThreadAllocatedMemorySupported()) throw new AssertionError("allocation probe unavailable");
        memory.setThreadAllocatedMemoryEnabled(true);
        long thread = Thread.currentThread().threadId();
        for (int size : new int[] {0, 1, 64, 4096, 65536}) {
            for (String shape : new String[] {"equal", "unique", "runs64", "above"}) {
                int[] offsets = new int[size];
                for (int i = 0; i < size; i++) offsets[i] = switch (shape) {
                    case "unique" -> i;
                    case "runs64" -> i / 64;
                    default -> 4;
                };
                int key = switch (shape) {
                    case "unique" -> Math.max(1, size / 2);
                    case "runs64" -> Math.max(1, size / 128);
                    case "above" -> Integer.MAX_VALUE;
                    default -> 4;
                };
                GapContent content = GapContentLowerBoundTest.fixture(offsets);
                int iterations = shape.equals("equal") ? Math.max(600, Math.min(100000, 20000000 / (size / 2 + 1))) : 100000;
                for (int warm = 0; warm < 8; warm++) call(content, key, iterations);
                for (int round = 0; round < 5; round++) {
                    long before = memory.getThreadAllocatedBytes(thread);
                    long elapsed = call(content, key, iterations);
                    long allocated = memory.getThreadAllocatedBytes(thread) - before;
                    System.out.println("time," + size + "," + shape + "," + round + "," + iterations + "," + elapsed + "," + allocated);
                }
            }
        }
    }
}
