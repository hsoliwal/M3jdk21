"""Task-specific ordered runner; transformation remains the canonical OpenRewrite SDK."""
from pathlib import Path
import argparse
import hashlib
import json
import os
import shutil
import subprocess

parser = argparse.ArgumentParser()
parser.add_argument('phase', choices=['recipe', 'product', 'measure'])
for argument in ['work', 'jdk', 'maven', 'm2']:
    parser.add_argument('--' + argument, type=Path, required=True)
args = parser.parse_args()
crate = Path(__file__).resolve().parent.parent
work = args.work.resolve()
evidence = work / 'evidence'
evidence.mkdir(parents=True, exist_ok=True)
env = os.environ.copy()
env['JAVA_HOME'] = str(args.jdk)
env['PATH'] = str(args.jdk / 'bin') + os.pathsep + env['PATH']


def run(name, command):
    log = evidence / (name + '.log')
    receipt = evidence / (name + '.json')
    assert not log.exists() and not receipt.exists(), name
    command = list(map(str, command))
    with log.open('w') as output:
        process = subprocess.Popen(command, cwd=work, env=env, stdout=output, stderr=subprocess.STDOUT)
        _, status, usage = os.wait4(process.pid, 0)
        process.returncode = os.waitstatus_to_exitcode(status)
    record = {'command': command, 'exit': process.returncode,
              'log_sha256': hashlib.sha256(log.read_bytes()).hexdigest(),
              'user_seconds': usage.ru_utime, 'system_seconds': usage.ru_stime,
              'max_rss_kib_linux': usage.ru_maxrss}
    receipt.write_text(json.dumps(record, indent=2) + '\n')
    print(name, process.returncode, '\n'.join(log.read_text().splitlines()[-4:]), flush=True)
    if process.returncode != 0:
        raise SystemExit('HOLD ' + name)


def completed(name):
    record = json.loads((evidence / (name + '.json')).read_text())
    assert record['exit'] == 0
    assert record['log_sha256'] == hashlib.sha256((evidence / (name + '.log')).read_bytes()).hexdigest()


def lint(name, files):
    dependencies = (crate / 'proof/lint-classpath.txt').read_text().splitlines()
    classpath = os.pathsep.join(str(args.m2 / item) for item in dependencies)
    run(name, ['java', '-cp', classpath, 'com.puppycrawl.tools.checkstyle.Main',
               '-c', crate / 'proof/checkstyle.xml', *files])


pins = json.loads((crate / 'SOURCE_PINS.json').read_text())
java_owner = 'src/java.desktop/share/classes/javax/swing/text/GapContent.java'
test_owner = 'test/jdk/javax/swing/text/GapContent/GapContentLowerBoundTest.java'
management = ['--add-modules', 'java.management,jdk.management',
              '--add-reads', 'java.desktop=java.management', '--add-reads', 'java.desktop=jdk.management']
if args.phase == 'recipe':
    lintdir = work / 'lint-source'
    lintdir.mkdir()
    templates = crate / 'resources/com/synexia/rewrite/hash-pinned-java/netbeans-gap-boundary-20261019'
    for filename in ['GapContent.java', 'GapContentLowerBoundTest.java']:
        shutil.copyfile(templates / (filename + '.txt'), lintdir / filename)
    lint('01-source-lint', [*lintdir.glob('*.java'), *crate.glob('src/test/**/*.java'),
                            *crate.glob('proof/*.java')])
    maven = [args.maven, '-o', '-B', '-ntp', '-Dmaven.repo.local=' + str(args.m2), '-f', crate / 'pom.xml']
    run('02-source-compile', [*maven, 'test-compile'])
    run('03-sdk-execution', [*maven, 'surefire:test', '-Dmaterialize.dir=' + str(work / 'generated')])
elif args.phase == 'product':
    completed('03-sdk-execution')
    for pin in pins['files']:
        assert hashlib.sha256((work / 'generated' / pin['path']).read_bytes()).hexdigest() == pin['after']
    for state in ['baseline', 'candidate']:
        stage = work / state
        source = stage / 'source/javax/swing/text'
        testsource = stage / 'test-source/javax/swing/text'
        classes = stage / 'classes'
        tests = stage / 'test-classes'
        external = stage / 'external'
        donor = stage / 'donor'
        for directory in [source, testsource, classes, tests, external, donor]:
            directory.mkdir(parents=True)
        origin = crate / 'proof/inputs/GapContent.java' if state == 'baseline' else work / 'generated' / java_owner
        shutil.copyfile(origin, source / 'GapContent.java')
        shutil.copyfile(crate / 'proof/inputs/GapVector.java', source / 'GapVector.java')
        shutil.copyfile(work / 'generated' / test_owner, testsource / 'GapContentLowerBoundTest.java')
        lint(state + '-04-product-lint', [*source.glob('*.java'), *testsource.glob('*.java')])
        run(state + '-05-product-compile', ['javac', '-Xlint:all', '-Werror', '--patch-module',
                'java.desktop=' + str(stage / 'source'), '-d', classes, *source.glob('*.java')])
        run(state + '-06-test-compile', ['javac', '-Xlint:all', '-Werror', '--patch-module',
                'java.desktop=' + str(classes) + os.pathsep + str(stage / 'test-source'),
                '-d', tests, *testsource.glob('*.java')])
        run(state + '-07-donor-compile', ['javac', '-Xlint:all', '-d', donor,
                *sorted((crate / 'third_party/netbeans').glob('*.java'))])
        patch = 'java.desktop=' + str(classes) + os.pathsep + str(tests)
        run(state + '-08-external-tests-compile', ['javac', '-Xlint:all', '--patch-module', patch,
                '-cp', donor, '-d', external, crate / 'proof/NetBeansOffsetOracle.java',
                crate / 'proof/inputs/bug4496801.java'])
        for name in ['GapContent', 'GapContent$MarkData', 'GapContent$StickyPosition', 'GapContent$MarkVector',
                     'GapContent$UndoPosRef', 'GapContent$InsertUndo', 'GapContent$RemoveUndo', 'GapVector']:
            run(state + '-09-api-' + name.replace('$', '-'), ['javap', '-p', '-s', classes / 'javax/swing/text' / (name + '.class')])
        base = ['java', '-Xmx512m', '-Djava.awt.headless=true', '--patch-module', patch]
        for flags, label in [([], 'mixed'), (['-Xint'], 'interpreter'), (['-XX:TieredStopAtLevel=1'], 'c1')]:
            run(state + '-10-contract-' + label, [*base, *flags, 'javax.swing.text.GapContentLowerBoundTest'])
        run(state + '-11-netbeans-oracle', [*base, '-cp', str(external) + os.pathsep + str(donor), 'NetBeansOffsetOracle'])
        run(state + '-12-html-regression', [*base, '-cp', external, 'bug4496801'])
    compared = []
    for baseline in sorted(evidence.glob('baseline-*.log')):
        suffix = baseline.name.removeprefix('baseline-')
        if suffix.startswith(('09-api-', '10-contract-', '11-netbeans-', '12-html-')):
            assert baseline.read_bytes() == (evidence / ('candidate-' + suffix)).read_bytes(), suffix
            compared.append(suffix)
    (evidence / 'PARITY.json').write_text(json.dumps({'identical_logs': compared,
         'scope': 'single production-file pass; actual java.desktop patch-module execution',
         'full_image': 'PENDING', 'full_jtreg': 'PENDING', 'jni': 'NOT_APPLICABLE_NO_NATIVE_CHANGE'}, indent=2) + '\n')
else:
    completed('candidate-12-html-regression')
    assert (evidence / 'PARITY.json').is_file()
    memory = work / 'memory'
    memory.mkdir()
    lint('memory-13-lint', [crate / 'proof/MemoryAgent.java', crate / 'proof/GapStorageMemory.java'])
    run('memory-14-compile', ['javac', '-Xlint:all', '-Werror', '-d', memory,
                            crate / 'proof/MemoryAgent.java', crate / 'proof/GapStorageMemory.java'])
    manifest = work / 'memory-manifest.mf'
    manifest.write_text('Premain-Class: proof.MemoryAgent\n\n')
    run('memory-15-agent', ['jar', '--create', '--file', work / 'memory-agent.jar',
                          '--manifest', manifest, '-C', memory, 'proof/MemoryAgent.class'])
    for state in ['baseline', 'candidate']:
        stage = work / state
        benchsrc = stage / 'bench-source/javax/swing/text'
        benchsrc.mkdir(parents=True)
        bench = stage / 'bench'
        bench.mkdir()
        shutil.copyfile(crate / 'proof/GapContentMarkBenchmark.java', benchsrc / 'GapContentMarkBenchmark.java')
        patch = 'java.desktop=' + str(stage / 'classes') + os.pathsep + str(stage / 'test-classes')
        lint(state + '-13-benchmark-lint', [benchsrc / 'GapContentMarkBenchmark.java'])
        run(state + '-14-benchmark-compile', ['javac', '-Xlint:all', '-Werror', *management,
                '--patch-module', patch + os.pathsep + str(stage / 'bench-source'), '-d', bench,
                benchsrc / 'GapContentMarkBenchmark.java'])
        base = ['java', '-Xms256m', '-Xmx256m', '-Djava.awt.headless=true', *management,
                '--patch-module', patch + os.pathsep + str(bench), 'javax.swing.text.GapContentMarkBenchmark']
        run(state + '-15-owned-storage', ['java', '-Xmx256m', '-Djava.awt.headless=true',
            '-javaagent:' + str(work / 'memory-agent.jar'), '--add-opens',
            'java.desktop/javax.swing.text=ALL-UNNAMED', '--patch-module', patch,
            '-cp', memory, 'GapStorageMemory'])
        run(state + '-15-probe-counts', [*base, 'counts'])
        for fork in range(3):
            run(state + '-16-measure-' + str(fork), base)
    assert (evidence / 'baseline-15-owned-storage.log').read_bytes() == (evidence / 'candidate-15-owned-storage.log').read_bytes()
