package com.synexia.convergence.repository;

import java.io.IOException;
import java.nio.file.Path;
import java.util.ArrayList;
import java.util.Comparator;
import java.util.List;
import java.util.Objects;
import java.util.TreeSet;

/** Immutable model and entry point for the all-module donor/public-code polish workspace. */
public final class RepositoryPublicPolishWorkspace {
    public enum Lane {
        PUBLIC_SURFACE_REVIEW,
        CONTRACT_TEST,
        IMPLEMENTATION_COMPLETION,
        INDEX_PRECOMPUTE,
        DONOR_REVIEW,
        CONTRACT_PRESERVING_REFACTOR
    }

    public record ApiTarget(
            String modulePath,
            String path,
            int line,
            String owner,
            String signature,
            RepositorySupersetAuditReport.ApiStrategy strategy,
            RepositoryApiCapabilityPlan.ImplementationState implementationState,
            RepositoryApiCapabilityPlan.VerificationState verificationState,
            List<Lane> lanes,
            boolean donorEvidenceAvailable,
            String bodySha256,
            String root) {

        public ApiTarget {
            modulePath = modulePath(modulePath);
            path = path(path);
            if (line < 1) throw new IllegalArgumentException("line");
            owner = text(owner, "owner", 1024);
            signature = text(signature, "signature", 4096);
            strategy = Objects.requireNonNull(strategy, "strategy");
            implementationState = Objects.requireNonNull(implementationState, "implementationState");
            verificationState = Objects.requireNonNull(verificationState, "verificationState");
            lanes = lanes(lanes);
            bodySha256 = sha(bodySha256, "bodySha256");
            root = root(root, apiMaterial(
                    modulePath, path, line, owner, signature, strategy, implementationState,
                    verificationState, lanes, donorEvidenceAvailable, bodySha256));
        }
    }

    public record ModuleTarget(
            String modulePath,
            String coordinate,
            String packaging,
            boolean rootReactorMember,
            int publicApiCount,
            int findingCount,
            List<Lane> lanes,
            String root) {

        public ModuleTarget {
            modulePath = modulePath(modulePath);
            coordinate = text(coordinate, "coordinate", 512);
            packaging = text(packaging, "packaging", 64);
            requireNonNegative(publicApiCount, "publicApiCount");
            requireNonNegative(findingCount, "findingCount");
            lanes = lanes(lanes);
            root = root(root, moduleMaterial(
                    modulePath, coordinate, packaging, rootReactorMember,
                    publicApiCount, findingCount, lanes));
        }
    }

    public record Workspace(
            String logicalRoot,
            String auditRoot,
            String donorEvidenceRoot,
            int donorEvidenceCount,
            List<ModuleTarget> modules,
            List<ApiTarget> apis,
            String root) {

        public Workspace {
            logicalRoot = text(logicalRoot, "logicalRoot", 256);
            auditRoot = sha(auditRoot, "auditRoot");
            donorEvidenceRoot = sha(donorEvidenceRoot, "donorEvidenceRoot");
            requireNonNegative(donorEvidenceCount, "donorEvidenceCount");
            modules = sortedModules(modules);
            apis = sortedApis(apis);
            root = root(root, workspaceMaterial(
                    logicalRoot, auditRoot, donorEvidenceRoot, donorEvidenceCount, modules, apis));
        }

        public long apiCount(Lane lane) {
            return apis.stream().filter(value -> value.lanes().contains(lane)).count();
        }

        public long moduleCount(Lane lane) {
            return modules.stream().filter(value -> value.lanes().contains(lane)).count();
        }

        public boolean sourceMutationAuthority() {
            return false;
        }

        public boolean donorSourceCopyAuthority() {
            return false;
        }

        public boolean promotionAuthority() {
            return false;
        }
    }

    private RepositoryPublicPolishWorkspace() {}

    public static Workspace build(Path repositoryRoot) throws IOException {
        return RepositoryPublicPolishWorkspaceCompiler.build(repositoryRoot);
    }

    public static Workspace compile(
            RepositorySupersetAuditReport report, String donorEvidenceRoot, int donorEvidenceCount) {
        return RepositoryPublicPolishWorkspaceCompiler.compile(
                report, donorEvidenceRoot, donorEvidenceCount);
    }

    private static String apiMaterial(
            String module,
            String path,
            int line,
            String owner,
            String signature,
            RepositorySupersetAuditReport.ApiStrategy strategy,
            RepositoryApiCapabilityPlan.ImplementationState implementation,
            RepositoryApiCapabilityPlan.VerificationState verification,
            List<Lane> lanes,
            boolean donor,
            String bodySha) {
        return String.join("\n", "PUBLIC-POLISH-API/1", module, path, Integer.toString(line),
                owner, signature, strategy.name(), implementation.name(), verification.name(),
                joinLanes(lanes), Boolean.toString(donor), bodySha);
    }

    private static String moduleMaterial(
            String path,
            String coordinate,
            String packaging,
            boolean rootMember,
            int apiCount,
            int findingCount,
            List<Lane> lanes) {
        return String.join("\n", "PUBLIC-POLISH-MODULE/1", path, coordinate, packaging,
                Boolean.toString(rootMember), Integer.toString(apiCount),
                Integer.toString(findingCount), joinLanes(lanes));
    }

    private static String workspaceMaterial(
            String logicalRoot,
            String auditRoot,
            String donorRoot,
            int donorCount,
            List<ModuleTarget> modules,
            List<ApiTarget> apis) {
        StringBuilder out = new StringBuilder("PUBLIC-POLISH-WORKSPACE/1\n")
                .append(logicalRoot).append('\n').append(auditRoot).append('\n')
                .append(donorRoot).append('\n').append(donorCount).append('\n');
        modules.forEach(value -> out.append("M\t").append(value.root()).append('\n'));
        apis.forEach(value -> out.append("A\t").append(value.root()).append('\n'));
        return out.toString();
    }

    private static List<ModuleTarget> sortedModules(List<ModuleTarget> values) {
        ArrayList<ModuleTarget> result = new ArrayList<>(Objects.requireNonNull(values, "modules"));
        result.sort(Comparator.comparing(ModuleTarget::modulePath).thenComparing(ModuleTarget::coordinate));
        return List.copyOf(result);
    }

    private static List<ApiTarget> sortedApis(List<ApiTarget> values) {
        ArrayList<ApiTarget> result = new ArrayList<>(Objects.requireNonNull(values, "apis"));
        result.sort(Comparator.comparing(ApiTarget::modulePath).thenComparing(ApiTarget::path)
                .thenComparingInt(ApiTarget::line).thenComparing(ApiTarget::signature));
        return List.copyOf(result);
    }

    static List<Lane> lanes(List<Lane> values) {
        TreeSet<Lane> result = new TreeSet<>(Comparator.comparing(Lane::name));
        result.addAll(Objects.requireNonNull(values, "lanes"));
        if (result.isEmpty()) throw new IllegalArgumentException("empty lanes");
        return List.copyOf(result);
    }

    static String joinLanes(List<Lane> values) {
        return values.stream().map(Enum::name).reduce((left, right) -> left + "," + right).orElse("");
    }

    static String modulePath(String value) {
        return ".".equals(value) ? "." : path(value);
    }

    static String path(String value) {
        String checked = text(value, "path", 4096);
        if (invalidPath(checked)) throw new IllegalArgumentException("path");
        return checked;
    }

    private static boolean invalidPath(String value) {
        return value.startsWith("/") || value.contains("\\") || value.equals("..")
                || value.startsWith("../") || value.endsWith("/..") || value.contains("/../")
                || java.util.Arrays.stream(value.split("/", -1))
                        .anyMatch(part -> part.isBlank() || ".".equals(part) || "..".equals(part));
    }

    private static String text(String value, String field, int max) {
        String checked = Objects.requireNonNull(value, field).strip();
        if (checked.isEmpty() || checked.length() > max || hasControl(checked)) {
            throw new IllegalArgumentException(field);
        }
        return checked;
    }

    private static boolean hasControl(String value) {
        return value.indexOf('\0') >= 0 || value.indexOf('\n') >= 0
                || value.indexOf('\r') >= 0 || value.indexOf('\t') >= 0;
    }

    private static String sha(String value, String field) {
        if (value == null || !value.matches("[0-9a-f]{64}")) throw new IllegalArgumentException(field);
        return value;
    }

    private static String root(String supplied, String material) {
        String expected = RepositoryHashing.sha256(material);
        String checked = supplied == null || supplied.isBlank() ? expected : sha(supplied, "root");
        if (!expected.equals(checked)) throw new IllegalArgumentException("PUBLIC_POLISH_ROOT_MISMATCH");
        return checked;
    }

    private static void requireNonNegative(int value, String field) {
        if (value < 0) throw new IllegalArgumentException(field);
    }
}
