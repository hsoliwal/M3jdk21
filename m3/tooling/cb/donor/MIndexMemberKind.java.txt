package com.synexia.mindex.classindex;

/** Frozen member-kind code stored in the primitive MIndexClass member table. */
public enum MIndexMemberKind {
    FIELD(0),
    METHOD(1),
    CONSTRUCTOR(2),
    RECORD_COMPONENT(3);

    private final int code;

    MIndexMemberKind(int code) {
        this.code = code;
    }

    public int code() {
        return code;
    }

    public static MIndexMemberKind fromCode(int code) {
        return switch (code) {
            case 0 -> FIELD;
            case 1 -> METHOD;
            case 2 -> CONSTRUCTOR;
            case 3 -> RECORD_COMPONENT;
            default -> throw new IllegalArgumentException("unknown member kind: " + code);
        };
    }
}
