package com.synexia.mindex.classindex;

import com.synexia.mindex.MIndexString;
import java.lang.reflect.Field;
import java.util.Objects;
import java.util.Optional;

/** Two-field indexed declared-field view with an explicit java.lang.reflect.Field boundary. */
public final class MIndexField {
    private final MIndexClassIndex owner;
    private final int row;

    MIndexField(MIndexClassIndex owner, int row) {
        this.owner = Objects.requireNonNull(owner, "owner");
        this.row = Objects.checkIndex(row, owner.memberSize());
        if (owner.memberKind(row) != MIndexMemberKind.FIELD) {
            throw new IllegalArgumentException("member row is not a field");
        }
    }

    public MIndexClassIndex owner() { return owner; }
    public int row() { return row; }
    public MIndexClass getDeclaringClass() { return owner.classAt(owner.memberDeclaringClassRow(row)); }
    public MIndexString name() { return owner.memberName(row); }
    public String getName() { return name().toString(); }
    public MIndexString descriptor() { return owner.memberDescriptor(row); }
    public int getModifiers() { return owner.memberModifiers(row); }
    public boolean isSynthetic() { return (owner.memberFlags(row) & MIndexMemberFlags.SYNTHETIC) != 0; }
    public boolean isEnumConstant() { return (owner.memberFlags(row) & MIndexMemberFlags.ENUM_CONSTANT) != 0; }
    public int annotationCount() { return owner.memberAnnotationCount(row); }
    public MIndexString annotationTypeNameAt(int ordinal) {
        return owner.memberAnnotationTypeName(row, ordinal);
    }
    public long structuralHash64() { return owner.memberStructuralHash64(row); }

    public Optional<Field> tryField() {
        return owner.resolveReflectiveElement(row)
                .filter(Field.class::isInstance)
                .map(Field.class::cast);
    }

    public Field toField() {
        return tryField().orElseThrow(
                () -> new IllegalStateException("JVM Field boundary is not live for row " + row));
    }

    @Override public String toString() { return getDeclaringClass().getName() + "#" + getName(); }
}
