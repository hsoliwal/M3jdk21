def load_module():
    path = Path(__file__).with_name("generate_recipe_crates.py")
    spec = importlib.util.spec_from_file_location("m3_generate_recipe_crates", path)
    if spec is None or spec.loader is None:
        raise RuntimeError(f"cannot load {path}")
    module = importlib.util.module_from_spec(spec)
    sys.modules[spec.name] = module
    spec.loader.exec_module(module)
    return module


class GenerateRecipeCratesTest(unittest.TestCase):
    def setUp(self) -> None:
        self.mod = load_module()

    @staticmethod
    def git(repo: Path, *args: str) -> None:
        subprocess.run(
            ("git", "-C", str(repo), *args),
            check=True,
            stdout=subprocess.PIPE,
            stderr=subprocess.PIPE,
            text=True,
        )

    @staticmethod
    def write(repo: Path, path: str, text: str) -> None:
        target = repo / path
        target.parent.mkdir(parents=True, exist_ok=True)
        target.write_text(text, encoding="utf-8")

    def test_verbatim_pairs_generate_hash_pinned_openrewrite_crate(self) -> None:
        with tempfile.TemporaryDirectory() as temp:
            root = Path(temp)
            repo = root / "repo"
            out = root / "out"
            subprocess.run(("git", "init", "-q", str(repo)), check=True)
            self.git(repo, "config", "user.email", "fixture@example.invalid")
            self.git(repo, "config", "user.name", "Fixture")

            modified = "src/java.base/share/classes/java/lang/A.java"
            added = "src/java.base/share/classes/java/lang/B.java"
            removed = "src/java.base/share/classes/java/lang/Removed.java"

            self.write(repo, modified, "package java.lang; final class A { int x() { return 21; } }\n")
            self.write(repo, removed, "package java.lang; final class Removed {}\n")
            self.git(repo, "add", ".")
            self.git(repo, "commit", "-q", "-m", "jdk21")
            self.git(repo, "tag", "jdk-21+35")

            self.write(repo, modified, "package java.lang; final class A { int x() { return 22; } }\n")
            self.write(repo, added, "package java.lang; final class B {}\n")
            (repo / removed).unlink()
            self.git(repo, "add", "-A")
            self.git(repo, "commit", "-q", "-m", "jdk22")
            self.git(repo, "tag", "jdk-22+36")

            candidates, exclusions = self.mod.candidates(
                repo, 22, selected=None, all_candidates=True
            )
            self.assertEqual(sorted([added, modified]), [row.path for row in candidates])
            self.assertEqual(
                [(removed, "TYPED_EXCLUSION_AUTOMATIC_REMOVAL")],
                exclusions,
            )

            crates = self.mod.materialize(out, 22, candidates, exclusions)
            self.assertEqual(["jdk22-0001"], crates)

            crate = (
                out
                / self.mod.RESOURCE_ROOT
                / "jdk22-0001"
            )
            with (crate / "manifest.tsv").open(
                "r", encoding="utf-8", newline=""
            ) as handle:
                rows = list(csv.reader(handle, delimiter="\t"))

            self.assertEqual(2, len(rows))
            self.assertEqual(modified, rows[0][0])
            self.assertEqual(64, len(rows[0][1]))
            self.assertEqual(added, rows[1][0])
            self.assertEqual("ABSENT", rows[1][1])
            self.assertEqual(64, len(rows[0][2]))
            self.assertEqual(64, len(rows[1][2]))
            self.assertTrue((crate / rows[0][3]).is_file())
            self.assertTrue((crate / rows[1][3]).is_file())

            yaml = (
                out
                / self.mod.YAML_ROOT
                / "m3-jdk22-candidate-backports.yml"
            ).read_text(encoding="utf-8")
            self.assertIn(
                "com.m3.rewrite.backport.M3Jdk21HashPinnedSnapshotRecipe",
                yaml,
            )
            self.assertIn("crateName: jdk22-0001", yaml)
            self.assertIn(
                "name: com.m3.generated.jdk22.CandidateBackports",
                yaml,
            )

            exclusions_text = (out / "EXCLUSIONS.tsv").read_text(encoding="utf-8")
            self.assertIn("TYPED_EXCLUSION_AUTOMATIC_REMOVAL", exclusions_text)

    def test_crate_size_one_emits_true_file_atomic_crates(self) -> None:
        with tempfile.TemporaryDirectory() as temp:
            out = Path(temp) / "out"
            first_text = "final class A {}\n"
            second_text = "final class B {}\n"
            candidates = [
                self.mod.Candidate(
                    path="src/java.base/share/classes/p/A.java",
                    status="MODIFIED",
                    before_sha256=self.mod._sha256_text("final class A { int x; }\n"),
                    after_sha256=self.mod._sha256_text(first_text),
                    after_text=first_text,
                ),
                self.mod.Candidate(
                    path="src/java.base/share/classes/p/B.java",
                    status="ADDED",
                    before_sha256="ABSENT",
                    after_sha256=self.mod._sha256_text(second_text),
                    after_text=second_text,
                ),
            ]

            crates = self.mod.materialize(
                out,
                22,
                candidates,
                exclusions=[],
                crate_size=1,
            )

            self.assertEqual(["jdk22-0001", "jdk22-0002"], crates)
            for ordinal, expected_path in enumerate(
                [candidate.path for candidate in candidates], 1
            ):
                crate = out / self.mod.RESOURCE_ROOT / f"jdk22-{ordinal:04d}"
                with (crate / "manifest.tsv").open(
                    "r", encoding="utf-8", newline=""
                ) as handle:
                    rows = list(csv.reader(handle, delimiter="\t"))
                self.assertEqual(1, len(rows))
                self.assertEqual(expected_path, rows[0][0])

            yaml = (
                out
                / self.mod.YAML_ROOT
                / "m3-jdk22-candidate-backports.yml"
            ).read_text(encoding="utf-8")
            self.assertIn("com.m3.generated.jdk22_0001", yaml)
            self.assertIn("com.m3.generated.jdk22_0002", yaml)

    @staticmethod
    def _ga_git_text(repo: Path, *args: str) -> str:
        return subprocess.run(
            ("git", "-C", str(repo), *args),
            check=True,
            stdout=subprocess.PIPE,
            stderr=subprocess.PIPE,
            text=True,
        ).stdout.strip()

    @staticmethod
    def _ga_store_manifest(fixture, manifest) -> None:
        fixture["reconciliation_file"].write_text(
            json.dumps(manifest, indent=2, sort_keys=True) + "\n",
            encoding="utf-8",
        )

    def _ga_fixture(self, root: Path):
        repo = root / "repo"
        subprocess.run(("git", "init", "-q", str(repo)), check=True)
        self.git(repo, "config", "user.email", "fixture@example.invalid")
        self.git(repo, "config", "user.name", "Fixture")
        self.git(repo, "config", "core.fileMode", "true")
        paths = {
            "old": "test/jdk/example/OldName.java",
            "new": "test/jdk/example/NewName.java",
            "transient": "src/jdk.jlink/share/classes/example/Transient.java",
            "modified": "src/jdk.jlink/share/classes/example/Changed.java",
            "same": "src/jdk.jlink/share/classes/example/Same.java",
            "excluded": "make/ga-helper.sh",
        }
        rename_text = "final class RenameFixture { int value() { return 1; } }\n"
        transient_text = "final class Transient { int value() { return 2; } }\n"
        same_text = "final class Same { int value() { return 21; } }\n"
        ga_text = "final class Changed { int value() { return 2400; } }\n"

        def commit(message):
            self.git(repo, "add", "-A")
            self.git(repo, "commit", "-q", "-m", message)
            return self._ga_git_text(repo, "rev-parse", "HEAD")

        self.write(repo, paths["modified"], "final class Changed { int value() { return 21; } }\n")
        self.write(repo, paths["same"], same_text)
        self.write(repo, paths["excluded"], "#!/bin/sh\necho 21\n")
        baseline = commit("baseline")
        self.git(repo, "tag", "jdk-21+35")

        self.write(repo, paths["old"], rename_text)
        self.write(repo, paths["same"], "final class Same { int value() { return 22; } }\n")
        implementation_parent = commit("preimplementation")

        self.git(repo, "mv", paths["old"], paths["new"])
        self.write(repo, paths["transient"], transient_text)
        self.write(repo, paths["modified"], "final class Changed { int value() { return 24; } }\n")
        self.write(repo, paths["same"], same_text)
        self.write(repo, paths["excluded"], "#!/bin/sh\necho 24\n")
        (repo / paths["excluded"]).chmod(0o755)
        implementation = commit("implementation rename and transient addition")

        (repo / paths["transient"]).unlink()
        deletion = commit("remove transient before GA")
        self.write(repo, paths["modified"], ga_text)
        donor = commit("GA includes cumulative followup")
        self.git(repo, "tag", "jdk-24+36")

        def blob(commit_sha, path):
            return self._ga_git_text(repo, "rev-parse", f"{commit_sha}:{path}")

        paths_file = root / "PATHS.txt"
        paths_file.write_text("".join(path + "\n" for path in sorted(paths.values())), encoding="utf-8")
        manifest = {
            "schema": "m3.ga-path-reconciliation/1",
            "release": 24,
            "catalogue_sha256": hashlib.sha256(paths_file.read_bytes()).hexdigest(),
            "declared_paths": len(paths),
            "baseline_ref": "jdk-21+35",
            "baseline_commit": baseline,
            "donor_ref": "jdk-24+36",
            "donor_commit": donor,
            "implementation_commit": implementation,
            "implementation_parent": implementation_parent,
            "entries": [
                {
                    "path": paths["old"],
                    "disposition": "RENAMED_PREIMAGE_ALREADY_SELECTED",
                    "transition_commit": implementation,
                    "transition_parent": implementation_parent,
                    "before_blob": blob(implementation_parent, paths["old"]),
                    "before_mode": "100644",
                    "replacement_path": paths["new"],
                    "transition_after_blob": blob(implementation, paths["new"]),
                    "ga_blob": blob(donor, paths["new"]),
                    "after_mode": "100644",
                },
                {
                    "path": paths["transient"],
                    "disposition": "TRANSIENT_DELETED_BEFORE_GA",
                    "transition_commit": deletion,
                    "transition_parent": implementation,
                    "before_blob": blob(implementation, paths["transient"]),
                    "before_mode": "100644",
                    "implementation_blob": blob(implementation, paths["transient"]),
                },
            ],
        }
        fixture = {
            "repo": repo,
            "paths": paths,
            "paths_file": paths_file,
            "reconciliation_file": root / "GA_RECONCILIATION.json",
            "manifest": manifest,
            "ga_text": ga_text,
            "rename_text": rename_text,
            "transient_text": transient_text,
        }
        self._ga_store_manifest(fixture, manifest)
        return fixture

    def _ga_select(self, fixture):
        return self.mod._ga_candidates(
            fixture["repo"],
            24,
            fixture["paths_file"],
            fixture["reconciliation_file"],
            include_text=True,
            include_native=False,
        )

    def _ga_reseal_catalogue(self, fixture, selected):
        fixture["paths_file"].write_text(
            "".join(path + "\n" for path in sorted(selected)), encoding="utf-8"
        )
        manifest = copy.deepcopy(fixture["manifest"])
        manifest["catalogue_sha256"] = hashlib.sha256(
            fixture["paths_file"].read_bytes()
        ).hexdigest()
        manifest["declared_paths"] = len(selected)
        return manifest

    def test_ga_reconciliation_accounts_for_both_histories_and_all_outcomes(self) -> None:
        with tempfile.TemporaryDirectory() as temp:
            fixture = self._ga_fixture(Path(temp))
            paths, manifest = fixture["paths"], fixture["manifest"]

            by_candidate = {candidate.path: candidate for candidate in candidates}

            self.assertEqual(fixture["rename_text"], by_candidate[paths["new"]].after_text)
            self.assertEqual(fixture["ga_text"], by_candidate[paths["modified"]].after_text)
            self.assertEqual(
                hashlib.sha256(fixture["ga_text"].encode("utf-8")).hexdigest(),

            self.assertEqual(3, len(exclusions))
            reasons = dict(exclusions)
            self.assertEqual(

            )
            self.assertEqual(
                "TYPED_EXCLUSION_GA_TRANSIENT_DELETED_BEFORE_GA", reasons[paths["transient"]]
            )
            self.assertTrue(reasons[paths["excluded"]].startswith("TYPED_EXCLUSION_FILE_MODE:"))
            self.assertEqual(sorted(paths.values()), [row["path"] for row in receipt])
            by_receipt = {row["path"]: row for row in receipt}
            expected = {
                paths["old"]: "HISTORY_ONLY", paths["transient"]: "HISTORY_ONLY",
                paths["new"]: "GENERATED", paths["modified"]: "GENERATED",
                paths["same"]: "SAME", paths["excluded"]: "EXCLUDED",
            }
            self.assertEqual(expected, {path: row["outcome"] for path, row in by_receipt.items()})
            self.assertEqual(paths["new"], by_receipt[paths["old"]]["related_path"])
            self.assertEqual("", by_receipt[paths["transient"]]["related_path"])
            receipt_keys = {
                "path", "outcome", "reason", "related_path", "baseline_commit", "donor_commit",
                "implementation_commit", "transition_commit", "manifest_sha256", "catalogue_sha256",
            }
            manifest_sha = hashlib.sha256(fixture["reconciliation_file"].read_bytes()).hexdigest()
            history = {entry["path"]: entry for entry in manifest["entries"]}
