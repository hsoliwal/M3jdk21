name: M3 JEP 467 Markdown doc-comment inventory

on:
  push:
    branches:
      - "m3/recover-jep467-current-master-20261005"
    paths:
      - "m3/backports/generate_recipe_crates.py"
      - "m3/backports/test_generate_recipe_crates.py"
      - "m3/backports/recipes/jep-467-markdown/**"
      - "m3/tooling/migration-recipes/**"
      - "m3/tooling/backport-dag/**"
      - ".github/workflows/m3-jep467-markdown-inventory.yml"
  pull_request:
    paths:
      - "m3/backports/generate_recipe_crates.py"
      - "m3/backports/test_generate_recipe_crates.py"
      - "m3/backports/recipes/jep-467-markdown/**"
      - "m3/tooling/migration-recipes/**"
      - "m3/tooling/backport-dag/**"
      - ".github/workflows/m3-jep467-markdown-inventory.yml"
  workflow_dispatch:

permissions:
  contents: read

jobs:
  inventory:
    runs-on: ubuntu-22.04
    timeout-minutes: 60
    steps:
      - uses: actions/checkout@v4
        with:
          fetch-depth: 0

      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: "21"
          cache: maven

      - name: Fetch exact OpenJDK 21 / JEP467 implementation / JDK23 GA
        shell: bash
        run: |
          set -euo pipefail
          mkdir "$RUNNER_TEMP/upstream-jdk"
          git -C "$RUNNER_TEMP/upstream-jdk" init -q
          git -C "$RUNNER_TEMP/upstream-jdk" remote add origin https://github.com/openjdk/jdk.git
          git -C "$RUNNER_TEMP/upstream-jdk" -c protocol.version=2 fetch \
            --filter=blob:none --no-tags origin \
            "refs/tags/jdk-21+35:refs/tags/jdk-21+35" \
            "refs/tags/jdk-23+37:refs/tags/jdk-23+37" \
            "0a58cffe88ba823e71fcdcca64b784ed04ca5398"
          printf 'UPSTREAM=%s\n' "$RUNNER_TEMP/upstream-jdk" >> "$GITHUB_ENV"

      - name: Prove implementation touched-path denominator
        shell: bash
        run: |
          set -euo pipefail
          mkdir -p target/jep467
          python3 - <<'PY'
          import subprocess
          from pathlib import Path

          repo = Path(__import__("os").environ["UPSTREAM"])
          expected = [
              line.strip()
              for line in Path("m3/backports/recipes/jep-467-markdown/PATHS.txt")
                  .read_text(encoding="utf-8").splitlines()
              if line.strip()
          ]
          actual = subprocess.check_output(
              ["git", "-C", str(repo), "diff-tree", "--no-commit-id", "--name-only", "-r",
               "0a58cffe88ba823e71fcdcca64b784ed04ca5398"],
              text=True,
          ).splitlines()
          if len(expected) != 251:
              raise SystemExit(f"expected 251 pinned paths, found {len(expected)}")
          if sorted(expected) != sorted(actual):
              missing = sorted(set(actual) - set(expected))
              extra = sorted(set(expected) - set(actual))
              raise SystemExit(f"JEP467 path denominator drift missing={missing} extra={extra}")
          print("JEP467_IMPLEMENTATION_PATHS=251")
          PY

      - name: Review implementation-to-GA follow-up delta
        shell: bash
        run: |
          set -euo pipefail
          python3 - <<'PY'
          import subprocess
          from pathlib import Path
          import os

          repo = Path(os.environ["UPSTREAM"])
          paths = [
              line.strip()
              for line in Path("m3/backports/recipes/jep-467-markdown/PATHS.txt")
                  .read_text(encoding="utf-8").splitlines()
              if line.strip()
          ]
          command = [
              "git", "-C", str(repo), "diff", "--name-status",
              "0a58cffe88ba823e71fcdcca64b784ed04ca5398",
              "jdk-23+37", "--", *paths,
          ]
          output = subprocess.check_output(command, text=True)
          target = Path("target/jep467/IMPLEMENTATION_TO_GA.tsv")
          target.write_text("status\tpath\n" + output, encoding="utf-8")
          changed = sum(1 for line in output.splitlines() if line.strip())
          Path("target/jep467/GA_FOLLOWUP_COUNT.txt").write_text(
              f"{changed}\n", encoding="utf-8"
          )
          print(f"JEP467_GA_FOLLOWUP_PATHS={changed}")
          PY

      - name: Unit-test file-atomic mixed crate generator
        shell: bash
        run: |
          set -euo pipefail
          cd m3/backports
          python3 -m unittest -v test_generate_recipe_crates.py

      - name: Generate candidate FILE atoms from JDK23 GA
        shell: bash
        run: |
          set -euo pipefail
          python3 m3/backports/generate_recipe_crates.py \
            --repo "$UPSTREAM" \
            --release 23 \
            --paths-file m3/backports/recipes/jep-467-markdown/PATHS.txt \
            --ga-reconciliation m3/backports/recipes/jep-467-markdown/GA_RECONCILIATION.json \
            --crate-size 1 \
            --include-text \
            --out target/jep467/file-atoms

      - name: Account for all 251 selected paths with pinned GA history
        shell: bash
        run: |
          set -euo pipefail
          python3 - <<'PY'
          import sys
          from pathlib import Path
          sys.path.insert(0, "m3/backports")
          from generate_recipe_crates import _verify_ga_accounting

          counts = _verify_ga_accounting(
              Path("target/jep467/file-atoms"),
              Path("m3/backports/recipes/jep-467-markdown/PATHS.txt"),
              Path("m3/backports/recipes/jep-467-markdown/GA_RECONCILIATION.json"),
          )
          if counts["selected"] != 251 or counts["history_only"] != 1:
              raise SystemExit("JEP467 declared-path/history denominator drift")
          if counts["generated"] + counts["exclusions"] != counts["selected"]:
              raise SystemExit(f"JEP467 accounting mismatch: {counts}")
          print(
              f"JEP467_ACCOUNTED generated={counts['generated']} "
              f"exclusions={counts['exclusions']} same={counts['same']} "
              f"history_only={counts['history_only']}"
          )
          PY

      - name: Verify M3 generator and recipe substrate
        shell: bash
        run: |
          set -euo pipefail
          mvn -B -ntp -f m3/pom.xml -pl tooling/migration-recipes -am \
            -Dtest=M3RecipeScopeDeclarationTest \
            -Dsurefire.failIfNoSpecifiedTests=false test

      - name: Build and install the complete DAG dependency reactor
        run: |
          mvn -B -ntp -f m3/pom.xml -pl tooling/backport-dag -am install

      - name: Validate evidence-bound JEP467 composition DAG
        run: |
          mvn -B -ntp -f m3/tooling/backport-dag/pom.xml \
            -Dexec.mainClass=com.m3.tooling.dag.M3PacketCollectionMain \
            -Dexec.args="validate $PWD/m3/backports/recipes/jep-467-markdown/collection.tsv" \
            exec:java

      - name: Project JEP467 DAG to Camel Airflow and Drools
        run: |
          set -euo pipefail
          mkdir -p target/jep467/projections
          mvn -B -ntp -f m3/tooling/backport-dag/pom.xml \
            -Dexec.mainClass=com.m3.tooling.dag.M3PacketCollectionMain \
            -Dexec.args="project $PWD/m3/backports/recipes/jep-467-markdown/collection.tsv $PWD/target/jep467/projections" \
            exec:java
          test -s target/jep467/projections/M3BackportRoutes.java
          test -s target/jep467/projections/m3_backport_dag.py
          test -s target/jep467/projections/m3_backport_rules.drl
          test -s target/jep467/projections/projection.tsv

      - name: Upload JEP467 inventory evidence
        uses: actions/upload-artifact@ea165f8d65b6e75b540449e92b4886f43607fa02
        with:
          name: jep467-inventory-${{ github.run_id }}
          path: target/jep467
          if-no-files-found: error
          retention-days: 30
