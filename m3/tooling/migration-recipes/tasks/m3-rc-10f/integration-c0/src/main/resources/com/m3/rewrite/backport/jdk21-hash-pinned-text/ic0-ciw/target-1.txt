name: M3 ClassVer actual framework proof
on:
  pull_request:
    paths:
      - 'm3/tooling/migration-recipes/**'
      - 'm3/tooling/module-packs/**'
      - '.github/workflows/m3-class-ver.yml'
      - 'm3/migration/classver.py'
      - 'm3/migration/classver-evidence.json'
      - 'm3/migration/test/test_classver.py'
      - 'm3/migration/recipe.py'
  workflow_dispatch:
permissions:
  contents: read
jobs:
  proof:
    runs-on: ubuntu-22.04
    timeout-minutes: 30
    steps:
      - uses: actions/checkout@11d5960a326750d5838078e36cf38b85af677262
        with:
          persist-credentials: false
          fetch-depth: 2
      - uses: actions/setup-java@cf277c60eb25467037889841efdb72551f06f6c3
        with:
          distribution: temurin
          java-version: '21'
      - name: Capture exact execution context
        shell: bash
        run: |
          set -euo pipefail
          mkdir m3-class-ver-proof
          git rev-parse HEAD > m3-class-ver-proof/commit.txt
          python3 -S -m unittest -v m3/migration/test/test_classver.py
          python3 -S m3/migration/classver.py > m3-class-ver-proof/whitespace.json
          java -version 2> m3-class-ver-proof/java.txt
          mvn -version > m3-class-ver-proof/maven.txt
      - name: Execute actual ClassVer parser and scheduler tests
        shell: bash
        run: |
          set -euo pipefail
          mvn -B -ntp -f m3/tooling/migration-recipes/pom.xml -Dtest=ClassVerTest test 2>&1 | tee m3-class-ver-proof/recipe.log
          python3 - <<'PY'
          import xml.etree.ElementTree as ET
          from pathlib import Path
          path = Path('m3/tooling/migration-recipes/target/surefire-reports/TEST-com.m3.rewrite.backport.ClassVerTest.xml')
          root = ET.parse(path).getroot()
          assert int(root.attrib['tests']) == 3, root.attrib
          assert all(int(root.attrib.get(key, '0')) == 0 for key in ('failures', 'errors', 'skipped')), root.attrib
          print('CLASS_VER_RECIPE_TESTS_PASS tests=3 skipped=0')
          PY
      - name: Execute complete unchanged recipe lifecycle and coverage gates
        shell: bash
        run: |
          set -euo pipefail
          mvn -B -ntp -f m3/tooling/migration-recipes/pom.xml clean verify 2>&1 | tee m3-class-ver-proof/recipes-verify.log
      - name: Execute complete unchanged module-pack lifecycle and coverage gates
        shell: bash
        run: |
          set -euo pipefail
          mvn -B -ntp -f m3/tooling/module-packs/pom.xml clean verify 2>&1 | tee m3-class-ver-proof/packs-verify.log
      - name: Execute actual Java21 VM and archive differential in both modes
        shell: bash
        run: |
          set -euo pipefail
          cp='m3/tooling/module-packs/kernel/target/classes:m3/tooling/module-packs/kernel/target/test-classes'
          java --add-modules jdk.jlink -cp "$cp" com.m3.tools.modulepack.M3ModuleClassVersionChecks "$RUNNER_TEMP/class-ver-jit" 2>&1 | tee m3-class-ver-proof/jit.log
          java -Xint --add-modules jdk.jlink -cp "$cp" com.m3.tools.modulepack.M3ModuleClassVersionChecks "$RUNNER_TEMP/class-ver-int" 2>&1 | tee m3-class-ver-proof/interpreter.log
          grep -Fx 'M3_CLASS_VERSION_PASS checks=639' m3-class-ver-proof/jit.log
          grep -Fx 'M3_CLASS_VERSION_PASS checks=639' m3-class-ver-proof/interpreter.log
      - name: Rebuild and run existing diagnostics image
        shell: bash
        run: |
          set -euo pipefail
          python3 -S m3/tooling/module-packs/build_pack.py --jdk "$JAVA_HOME" --out "$RUNNER_TEMP/class-ver-image" 2>&1 | tee m3-class-ver-proof/image.log
      - name: Preserve proof even when a gate fails
        if: always()
        uses: actions/upload-artifact@ea165f8d65b6e75b540449e92b4886f43607fa02
        with:
          name: class-ver-proof
          if-no-files-found: error
          path: |
            m3-class-ver-proof/
            m3/tooling/migration-recipes/target/surefire-reports/
            m3/tooling/migration-recipes/target/site/jacoco/
            m3/tooling/module-packs/kernel/target/surefire-reports/
            m3/tooling/module-packs/kernel/target/site/jacoco/
            m3/tooling/module-packs/recipes/target/surefire-reports/
            m3/tooling/module-packs/recipes/target/site/jacoco/
            ${{ runner.temp }}/class-ver-image/RECEIPT.json
            ${{ runner.temp }}/class-ver-image/logs/
