    @Test
    void aliasesAreCompleteStableFullSha256Identities() throws Exception {
        String original = "file-" + "long-".repeat(100) + "java";
        String expected = "a-" + HexFormat.of().formatHex(
                MessageDigest.getInstance("SHA-256").digest(original.getBytes(StandardCharsets.UTF_8)));
        assertEquals(expected, M3BackportPacketCollection.aliasOf(original));
        assertEquals(66, expected.length());
        assertNotEquals(expected, M3BackportPacketCollection.aliasOf(original + "-other"));
        assertEquals(66, M3BackportPacketCollection.aliasOf("a".repeat(1024)).length());
        for (String invalid : List.of("", "A", " a", "a ", "a/b", "a\n", "a".repeat(1025))) {
            assertThrows(IllegalArgumentException.class, () -> M3BackportPacketCollection.aliasOf(invalid));
        }
        assertThrows(NullPointerException.class, () -> M3BackportPacketCollection.aliasOf(null));
    }

    @Test
    void collectionRetainsAtomsEdgesEvidenceAndEveryGlobalTerminal() {
        M3BackportPacket left = packet("left", atom("a"), atom("b", "a"));
        M3BackportPacket right = packet("right", atom("c"), atom("unused"));
        M3BackportPacketCollection collection = collection(
                List.of(left, right), List.of(join("first", "b", "c"), join("last", "first")),
                "a", "b", "c", "unused", "first", "last");
        assertEquals(List.of(alias("a"), alias("b"), alias("c"), alias("unused"),
                        alias("first"), alias("last")),
                collection.atoms().stream().map(M3RecipeAtom::id).toList());
        assertEquals(List.of(alias("unused"), alias("last")),
                collection.terminalAtoms().stream().map(M3RecipeAtom::id).toList());
        assertEquals(4, collection.edgeCount());
        assertEquals(M3EditScope.LIBRARY_API, collection.maximumScope());
        M3RecipeDag dag = M3PacketDagComposer.composeCollection(collection);
        assertEquals(21, dag.size());
        assertCurrentLifecycle(dag);
        assertEquals(List.of("packet-" + alias("unused"), "packet-" + alias("last")),
                dag.require("recipe-junit").dependsOn());
        assertEquals(List.of("packet-" + alias("b"), "packet-" + alias("c")),
                dag.require("packet-" + alias("first")).dependsOn());
        assertEquals(1, dag.topologicalOrder().stream()
                .filter(node -> node.kind() == M3DagKind.PROMOTION).count());
        assertEquals("promote", dag.topologicalOrder().getLast().id());
        assertTrue(dag.require("promote").serialPromotion());
        assertEquals(M3EditScope.LIBRARY_API, dag.require("promote").scope());
    }

    @Test
    void collectionBudgetsAreExplicitWithoutChangingPacketBudget() {
        M3BackportPacket packet = packet("p", atom("a"));
        assertThrows(IllegalArgumentException.class,
                () -> collection(List.of(), List.of()));
        assertThrows(IllegalArgumentException.class,
                () -> collection(Collections.nCopies(17, packet), List.of(), "a"));
        assertThrows(IllegalArgumentException.class,
                () -> collection(List.of(packet), Collections.nCopies(17, join("j", "a")), "a", "j"));
        ArrayList<M3BackportPacket> partitions = new ArrayList<>();
        ArrayList<String> originals = new ArrayList<>();
        for (int i = 0; i < 16; i++) {
            String name = "atom-" + i;
            partitions.add(packet("part-" + i, atom(name)));
            originals.add(name);
        }
        assertEquals(16, collection(partitions, List.of(), originals.toArray(String[]::new)).packets().size());
        ArrayList<M3RecipeAtom> joins = new ArrayList<>();
        originals = new ArrayList<>(List.of("a"));
        for (int i = 0; i < 16; i++) {
            String name = "join-" + i;
            joins.add(join(name, i == 0 ? "a" : "join-" + (i - 1)));
            originals.add(name);
        }
        assertEquals(17, collection(List.of(packet), joins, originals.toArray(String[]::new)).atoms().size());
    }

    @Test
    void realCataloguesRetainEveryAtomDependencyAndEvidenceCell() throws Exception {
        for (String name : List.of("jep-467-markdown", "jep-493-runtime-image")) {
            Path descriptor = copyCatalogue(name, name);
            M3BackportPacketCollection collection = M3PacketCollectionLoader.load(descriptor);
            int count = name.equals("jep-467-markdown") ? 259 : 54;
            assertEquals(count, collection.atoms().size());
            assertEquals(count - 1, collection.edgeCount());
            assertEquals(name.equals("jep-467-markdown") ? 8 : 1, collection.packets().size());
            assertEquals(name.equals("jep-467-markdown") ? 1 : 0, collection.joins().size());
            assertTrue(collection.packets().stream().allMatch(packet -> packet.atoms().size() <= 256));
            assertEquals(sha256(Files.readAllBytes(descriptor)), collection.descriptorSHA256());
            Map<String, M3RecipeAtom> atoms = collection.atoms().stream()
                    .collect(Collectors.toMap(M3RecipeAtom::id, atom -> atom));
            Map<String, M3RecipeAtomEvidence> proofs = new LinkedHashMap<>();
            collection.evidence().forEach(rows -> proofs.putAll(rows.byAtomId()));
            collection.joinEvidence().forEach(row -> proofs.put(row.atomId(), row));
            List<String> packetRows = Files.readAllLines(descriptor.resolveSibling("packet.tsv"));
            List<String> evidenceRows = Files.readAllLines(descriptor.resolveSibling("atom-evidence.tsv"));
            assertThrows(IllegalArgumentException.class, () -> M3BackportPacketLoader.parse(
                    Files.readString(descriptor.resolveSibling("packet.tsv"))));
            assertThrows(IllegalArgumentException.class, () -> M3BackportPacketEvidenceLoader.parse(
                    Files.readString(descriptor.resolveSibling("atom-evidence.tsv"))));
            for (int i = 1; i < packetRows.size(); i++) {
                String[] original = packetRows.get(i).split("\t", -1);
                String[] proof = evidenceRows.get(i).split("\t", -1);
                String id = collection.aliases().get(original[1]);
                M3RecipeAtom atom = atoms.get(id);
                assertEquals(alias(original[1]), id);
                assertEquals(original[2], atom.scope().name());
                assertEquals(original[3], Boolean.toString(atom.scopePromotionApproved()));
                assertEquals(original[4], atom.workRef());
                assertEquals(original[5].isEmpty() ? List.of()
                                : List.of(original[5].split(",")).stream().map(collection.aliases()::get).toList(),
                        atom.dependsOn());
                assertEquals(new M3RecipeAtomEvidence(id, proof[2], proof[3], proof[4], proof[5], proof[6],
                        Boolean.parseBoolean(proof[7])), proofs.get(id));
            }
            assertEquals(count, atoms.size());
            assertEquals(count, proofs.size());
            M3RecipeDag dag = M3PacketDagComposer.composeCollection(collection);
            assertEquals(count + 15, dag.size());
            assertEquals(name.equals("jep-467-markdown") ? 274 : 69, dag.size());
            assertCurrentLifecycle(dag);
            assertEquals(1, dag.topologicalOrder().stream()
                    .filter(node -> node.kind() == M3DagKind.PROMOTION).count());
            assertEquals("promote", dag.topologicalOrder().getLast().id());
        }
    }

    @Test

        for (String name : List.of("jep-467-markdown", "jep-493-runtime-image")) {
            Path descriptor = copyCatalogue(name, "cli-" + name);
            List<String> before;
            try (var files = Files.list(descriptor.getParent())) {

            }
            M3PacketCollectionMain.main(new String[] {"validate", descriptor.toString()});

                assertEquals(before, files.map(path -> path.getFileName().toString()).sorted().toList());

            Path output = descriptor.resolveSibling("projection");
            M3PacketCollectionMain.main(new String[] {"project", descriptor.toString(), output.toString()});
            M3BackportPacketCollection collection = M3PacketCollectionLoader.load(descriptor);
            M3RecipeDag dag = M3PacketDagComposer.composeCollection(collection);
            Map<M3DagProjectionFormat, M3DagProjection> projections = M3DagProjector.all(dag, collection.collectionId());
            Set<String> names = new java.util.HashSet<>(Set.of("projection.tsv", "atom-evidence.tsv"));
            for (M3DagProjection projection : projections.values()) {
                names.add(projection.fileName());
                assertEquals(projection.content(), Files.readString(output.resolve(projection.fileName())));
            }
            try (var files = Files.list(output)) {
                assertEquals(names, files.map(path -> path.getFileName().toString()).collect(Collectors.toSet()));
            }
            assertEquals("packet_id\tdag_root\tnodes\n" + collection.collectionId() + "\t"
                            + M3DagSemanticRoot.of(dag) + "\t" + dag.size() + "\n",
                    Files.readString(output.resolve("projection.tsv")));
            assertEquals("collection_id\tdescriptor_sha256\tatoms\tedges\n" + collection.collectionId() + "\t"
                            + collection.descriptorSHA256() + "\t" + collection.atoms().size() + "\t"
                            + collection.edgeCount() + "\n",
                    Files.readString(output.resolve("atom-evidence.tsv")));
            assertThrows(IllegalArgumentException.class, () -> M3PacketCollectionMain.main(
                    new String[] {"project", descriptor.toString(), output.toString()}));
        }

    @Test
    void collectionRechecksBudgetAfterGenuineGrowthDuringDefensiveCopy() {
        ArrayList<M3BackportPacketEvidence> proofs = new ArrayList<>();
        M3BackportPacket extra = packet("part-16", atom("atom-16"));
        M3BackportPacketEvidence extraEvidence = evidence(extra);
        boolean[] grewDuringCopy = {false};
        ArrayList<M3BackportPacket> growing = new ArrayList<>() {
            @java.io.Serial
            private static final long serialVersionUID = 1L;

            @Override
            public Object[] toArray() {
                if (!grewDuringCopy[0]) {
                    add(extra);
                    proofs.add(extraEvidence);
                    grewDuringCopy[0] = true;
                }
                return super.toArray();
            }
        };
        LinkedHashMap<String, String> allAliases = new LinkedHashMap<>();
        for (int i = 0; i < 17; i++) {
            String original = "atom-" + i;
            allAliases.put(original, alias(original));
            if (i < 16) {
                M3BackportPacket packet = packet("part-" + i, atom(original));
                growing.add(packet);
                proofs.add(evidence(packet));
            }
        }
        assertEquals(16, growing.size());
        assertEquals(16, proofs.size());
        assertFalse(grewDuringCopy[0]);
        IllegalArgumentException failure = assertThrows(IllegalArgumentException.class,
                () -> new M3BackportPacketCollection("collection", growing, proofs,
                        List.of(), List.of(), allAliases, DESCRIPTOR_SHA));
        assertEquals("collection input budget", failure.getMessage());
        assertTrue(grewDuringCopy[0]);
        assertEquals(17, growing.size());
        assertEquals(17, proofs.size());
        assertEquals(extra, growing.get(16));
        assertEquals(extraEvidence, proofs.get(16));
    }

    private Path copyCatalogue(String catalogue, String fixtureName) throws Exception {
        // Maven runs this test from m3/tooling/backport-dag; no fallback or filesystem discovery.
        Path source = Path.of("../../backports/recipes").resolve(catalogue).normalize();
        Path destination = Files.createDirectory(temp.resolve(fixtureName));
        for (String name : List.of("packet.tsv", "atom-evidence.tsv", "collection-map.tsv", "collection.tsv")) {
            Files.copy(source.resolve(name), destination.resolve(name));
        }
        return destination.resolve("collection.tsv");
    }

    private static String sha256(byte[] data) throws Exception {
        return HexFormat.of().formatHex(MessageDigest.getInstance("SHA-256").digest(data));
    }

    private static void changeDescriptorCell(Path descriptor, int column, String value) throws Exception {
        List<String> lines = Files.readAllLines(descriptor);
        String[] row = lines.get(1).split("\t", -1);
        row[column] = value;
        Files.writeString(descriptor, lines.getFirst() + "\n" + String.join("\t", row) + "\n");
    }

    private static void reseal(Path descriptor, String file) throws Exception {
        int column = switch (file) {
            case "packet.tsv" -> 3;
            case "atom-evidence.tsv" -> 5;
            case "collection-map.tsv" -> 7;
            default -> throw new IllegalArgumentException(file);
        };
        changeDescriptorCell(descriptor, column, sha256(Files.readAllBytes(descriptor.resolveSibling(file))));
    }

    private static void changeInputCell(Path descriptor, String file, int row, int column, String value)
            throws Exception {
        Path input = descriptor.resolveSibling(file);
        List<String> lines = new ArrayList<>(Files.readAllLines(input));
        String[] cells = lines.get(row).split("\t", -1);
        cells[column] = value;
        lines.set(row, String.join("\t", cells));
        Files.writeString(input, String.join("\n", lines) + "\n");
        reseal(descriptor, file);
    }

    private static String alias(String original) {
        return M3BackportPacketCollection.aliasOf(original);
    }

    private static void assertCurrentLifecycle(M3RecipeDag dag) {
        M3RecipeDag canonical = M3RecipeDag.canonical();
        assertEquals(16, canonical.size());
        List<String> reviewPrefix = List.of("review-code-signal", "review-atom-pattern",
                "review-problem-planner", "review-jni-contract");
        assertEquals(reviewPrefix, dag.topologicalOrder().subList(0, 4).stream().map(M3DagNode::id).toList());
        for (String id : reviewPrefix) {
            assertEquals(canonical.require(id), dag.require(id));
            assertFalse(dag.require(id).mutating());
        }
        assertEquals(canonical.require("inventory"), dag.require("inventory"));
        List<String> tail = List.of("recipe-junit", "diff", "lint", "compile", "jtreg", "runtime", "promote");
        assertEquals(tail, dag.topologicalOrder().stream().map(M3DagNode::id).filter(tail::contains).toList());
        for (int i = 1; i < tail.size(); i++) {
            assertEquals(List.of(tail.get(i - 1)), dag.require(tail.get(i)).dependsOn());
        }
        assertEquals(1, dag.topologicalOrder().stream().filter(node -> node.kind() == M3DagKind.PROMOTION).count());
        assertTrue(dag.require("promote").serialPromotion());
        assertEquals("promote", dag.topologicalOrder().getLast().id());
    }

    private static Map<String, String> aliases(String... originals) {
        LinkedHashMap<String, String> result = new LinkedHashMap<>();
        for (String original : originals) {
            result.put(original, alias(original));
        }
        return result;
    }

    private static M3RecipeAtom atom(String original, String... dependencies) {
        return new M3RecipeAtom(alias(original), M3EditScope.FILE, false, "work/" + original,
                List.of(dependencies).stream().map(M3BackportPacketCollectionTest::alias).toList());
    }

        return new M3RecipeAtom(alias(original), M3EditScope.LIBRARY_API, true, "work/" + original,

    }

    private static M3RecipeAtomEvidence evidence(String atomId) {

    private static M3BackportPacketCollection collection(
            List<M3BackportPacket> packets, List<M3RecipeAtom> joins, String... originals) {

                packets.stream().map(M3BackportPacketCollectionTest::evidence).toList(), joins,
                joins.stream().map(atom -> evidence(atom.id())).toList(), aliases(originals), DESCRIPTOR_SHA);
    }
