package com.m3.rewrite.backport;

import com.fasterxml.jackson.annotation.JsonCreator;
import com.fasterxml.jackson.annotation.JsonProperty;
import java.util.List;
import java.util.Set;
import org.openrewrite.Option;
import org.openrewrite.Recipe;

/**
 * Explicit opt-in recipe for final JEP 485, Stream Gatherers, on Java 21.
 *
 * <p>The reviewed donor state is OpenJDK JDK 24 GA ({@code jdk-24+36}). The packet imports the
 * final, non-preview Gatherer API and implementation, the cumulative GA fixes, focused jtreg, and
 * only the stream-pipeline integration required by that API. JDK24's {@code MhUtil} dependency is
 * adapted locally because that unrelated helper does not exist in the Java 21 target.</p>
 *
 * <p>Recipe invocation requires explicit LIBRARY_API contract-change authority. The guard does
 * not remove existing Gatherer sources or establish the compatibility of a built JDK image.</p>
 */
public final class M3Jep485StreamGatherersBackportRecipe extends Recipe {
    public static final String OPT_IN_TOKEN = "JEP485";
    public static final String DONOR_TAG = "jdk-24+36";
    public static final String IMPLEMENTATION_COMMIT =
            "33b26f79a986d015abdcd84b89842adc0a4bde64";
    public static final String GRADUATION_COMMIT =
            "ef0dc2518e7636cc8a9ca580613ff5edeb4c19fd";
    public static final String MAP_CONCURRENT_FIX =
            "450636ae28b84ded083b6861c6cba85fbf87e16e";

    @Option(
            displayName = "Explicit opt-in token",
            description = "Must be JEP485 to select the explicit SE API extension recipe.",
            example = "JEP485",
            required = false)
    private final String optInToken;

    public M3Jep485StreamGatherersBackportRecipe() {
        this(null);
    }

    @JsonCreator
    public M3Jep485StreamGatherersBackportRecipe(@JsonProperty("optInToken") String optInToken) {
        this.optInToken = optInToken;
    }

    public boolean optedIn() {
        return OPT_IN_TOKEN.equals(optInToken);
    }

    @Override
    public String getDisplayName() {
        return "Backport final Stream Gatherers to M3JDK21";
    }

    @Override
    public String getDescription() {
        return "Replays the reviewed JDK24-GA Stream Gatherers API, stream-pipeline integration "
                + "and GA tests from exact sealed preimages after explicit JEP485 opt-in.";
    }

    @Override
    public Set<String> getTags() {
        return Set.of(
                "m3",
                "jdk21",
                "backport",
                "jep-485",
                "stream-gatherers",
                "java-base",
                "hash-pinned",
                "library-api",
                "explicit-contract-change",
                "opt-in-se-api",
                "jdk24-ga",
                "candidate-only");
    }

    @Override
    public int maxCycles() {
        return 1;
    }

    @Override
    public List<Recipe> getRecipeList() {
        if (!optedIn()) {
            return List.of();
        }
        return List.of(
                new M3Jdk21HashPinnedSnapshotRecipe("jdk24-jep485-stream-gatherers"));
    }
}
