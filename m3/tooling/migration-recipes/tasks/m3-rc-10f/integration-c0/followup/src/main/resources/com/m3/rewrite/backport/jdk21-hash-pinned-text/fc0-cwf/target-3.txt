"""Exercise Lane contexts from actual Maven outputs and selected source bodies.

All 19 historical postimages must already have been generated by the retained
Maven recipe. Each invocation runs the actual verifier on a private fixture.
"""
import hashlib
import importlib.util
import json
import os
from pathlib import Path
import stat
import subprocess
import sys
import tempfile
import unittest

CRATE = Path(__file__).resolve().parents[3]
ROOT = CRATE.parents[2]
MAP = 'm3/docs/name-mapping.json'
WORKFLOW = '.github/workflows/m3-lane28.yml'
NAMING = 'm3/docs/counterpart-naming.md'
LEGAL = 'src/java.base/share/legal/synexia.md'
CURRENT_MAP_SHA = 'cf6a75806908fee920ecbf7bbe548182f1d9db179e9cc554e36739b1c45a6e6c'
CURRENT_WORKFLOW_SHA = '1aa2a7a495a00d3a0754bcf776cf7d3ef3bd176134186bfe1ec4004bb8a74616'


class Ic0ContextTest(unittest.TestCase):
    @classmethod
    def setUpClass(cls):
        spec = importlib.util.spec_from_file_location('lane_context_sealed', ROOT / 'm3/migration/recipe.py')
        sealed = importlib.util.module_from_spec(spec)
        spec.loader.exec_module(sealed)
        cls.plan, cls.rows = sealed.sealed_plan(CRATE / 'plan.json')
        if len(cls.rows) != 19:
            raise AssertionError('Exact 19-output Lane plan required')
        cls.generated = {}
        for row in cls.rows:
            body = (CRATE / 'target/generated' / row['path']).read_bytes()
            if body != row['after']:
                raise AssertionError('Actual Maven-generated postimage differs: ' + row['path'])
            cls.generated[row['path']] = body
        cls.current = {name: (ROOT / name).read_bytes() for name in (MAP, WORKFLOW, NAMING, LEGAL)}
        for name, expected in ((MAP, CURRENT_MAP_SHA), (WORKFLOW, CURRENT_WORKFLOW_SHA)):
            if hashlib.sha256(cls.current[name]).hexdigest() != expected:
                raise AssertionError('Actual selected Ic0 body required: ' + name)
        source_paths = {CRATE / 'plan.json', CRATE / 'verify-plan.py', ROOT / 'm3/migration/recipe.py'}
        source_paths.update(ROOT / guard['path'] for guard in cls.plan['guards'])
        source_paths.update(p for p in (CRATE / 'src/main/resources').rglob('*') if p.is_file())
        cls.sources = {p.relative_to(ROOT).as_posix(): p.read_bytes() for p in source_paths}
        cls.original_inputs = {str(ROOT / p): b for p, b in cls.sources.items()}
        cls.original_inputs.update({str(ROOT / p): b for p, b in cls.current.items()})
        cls.original_inputs.update({str(CRATE / 'target/generated' / p): b for p, b in cls.generated.items()})

    @classmethod
    def tearDownClass(cls):
        for path, body in cls.original_inputs.items():
            if Path(path).read_bytes() != body:
                raise AssertionError('Receiving input changed: ' + path)

    def setUp(self):
        temporary = tempfile.TemporaryDirectory(prefix='m3-lane-context-')
        self.addCleanup(temporary.cleanup)
        self.root = Path(temporary.name)
        self.crate = self.root / 'm3/tooling/lane28'
        for name, body in self.sources.items():
            path = self.root / name
            path.parent.mkdir(parents=True, exist_ok=True)
            path.write_bytes(body)
        for name, body in self.generated.items():
            for path in (self.root / name, self.crate / 'target/generated' / name):
                path.parent.mkdir(parents=True, exist_ok=True)
                path.write_bytes(body)

    def snapshot(self):
        result = {}
        for parent, directories, files in os.walk(self.root, followlinks=False):
            for name in directories + files:
                path = Path(parent) / name
                relative = path.relative_to(self.root).as_posix()
                mode = stat.S_IMODE(path.lstat().st_mode)
                if path.is_symlink():
                    result[relative] = ('symlink', mode, os.readlink(path))
                elif path.is_dir():
                    result[relative] = ('directory', mode)
                else:
                    result[relative] = ('file', mode, path.read_bytes())
        return result

    def invoke(self, arguments=(), successful=False):
        before = self.snapshot()
        result = subprocess.run(
            [sys.executable, '-B', '-S', str(self.crate / 'verify-plan.py'), *arguments],
            cwd=self.root, capture_output=True, timeout=30, check=False)
        if successful:
            self.assertEqual(result.returncode, 0, result.stderr.decode(errors='replace'))
            self.assertEqual(result.stderr, b'')
            after = self.snapshot()
            custody = 'm3/tooling/lane28/target/custody.json'
            self.assertEqual(set(after) - set(before), {custody})
            self.assertEqual({k: v for k, v in after.items() if k != custody}, before)
            record = json.loads(after[custody][2])
            self.assertEqual(record['schema'], 'm3.lane28-custody/1')
            self.assertEqual(record['plan_sha256'], self.plan['plan_sha256'])
            self.assertEqual(record['generated_outputs'], 19)
            events = record['events']
            self.assertEqual([r['mode'] for r in events if 'mode' in r],
                             ['check', 'apply', 'apply', 'check', 'rollback', 'rollback'])
            self.assertEqual([(r['state'], r['writes']) for r in events if 'mode' in r],
                             [('before', 0), ('after', 19), ('after', 0),
                              ('after', 0), ('before', 19), ('before', 0)])
            self.assertEqual([r['refusal'] for r in events if 'refusal' in r],
                             ['occupied target, no writes', 'mixed pre/postimages, no writes',
                              'dependency drift, no writes'])
            return record
        self.assertNotEqual(result.returncode, 0, result.stdout.decode(errors='replace'))
        self.assertEqual(self.snapshot(), before, 'Refusal must preserve fixture bytes, modes and links')
        return result

    def composed(self):
        for name, body in self.current.items():
            (self.root / name).write_bytes(body)

    def mutate_mapping(self, mutation):
        document = json.loads(self.current[MAP])
        mutation(document)
        (self.root / MAP).write_text(json.dumps(document, indent=2) + '\n')

    def test_default_historical_context_and_replay(self):
        record = self.invoke(successful=True)
        self.assertNotIn('source_context', record)

    def test_named_historical_context_and_replay(self):
        record = self.invoke(['--context', 'historical'], successful=True)
        self.assertNotIn('source_context', record)

    def test_explicit_ic0_context_and_historical_replay(self):
        self.composed()
        record = self.invoke(['--context', 'ic0'], successful=True)
        self.assertEqual(record['source_context'], 'ic0')
        self.assertEqual(record['current_mapping_sha256'], CURRENT_MAP_SHA)
        self.assertEqual(record['current_workflow_sha256'], CURRENT_WORKFLOW_SHA)
        self.assertEqual(record['current_records'], 87)

    def test_default_refuses_new_workflow(self):
        self.composed()
        self.invoke()

    def test_historical_shared_document_append_contract_survives(self):
        for name in (MAP, NAMING, LEGAL):
            (self.root / name).write_bytes(self.current[name])
        self.invoke(successful=True)

    def test_ic0_requires_selected_mapping(self):
        (self.root / WORKFLOW).write_bytes(self.current[WORKFLOW])
        self.invoke(['--context', 'ic0'])

    def test_ic0_requires_selected_workflow(self):
        (self.root / MAP).write_bytes(self.current[MAP])
        self.invoke(['--context', 'ic0'])

    def test_unknown_context_is_refused(self):
        self.composed()
        result = self.invoke(['--context', 'future'])
        self.assertEqual(result.returncode, 2)

    def test_retained_record_mutation_is_refused(self):
        self.composed()
        self.mutate_mapping(lambda d: d['migration']['records'][0].update(status='active'))
        self.invoke(['--context', 'ic0'])

    def test_missing_retained_record_is_refused(self):
        self.composed()
        self.mutate_mapping(lambda d: d['migration']['records'].pop(0))
        self.invoke(['--context', 'ic0'])

    def test_duplicate_id_is_refused(self):
        self.composed()
        def duplicate(document):
            rows = document['migration']['records']
            rows[-1]['id'] = rows[0]['id']
        self.mutate_mapping(duplicate)
        self.invoke(['--context', 'ic0'])

    def test_foreign_pending_payload_is_refused(self):
        self.composed()
        self.mutate_mapping(lambda d: d['migration']['records'][-1].update(status='active'))
        self.invoke(['--context', 'ic0'])

    def test_top_level_authority_change_is_refused(self):
        self.composed()
        self.mutate_mapping(lambda d: d.update(unreviewed_authority=True))
        self.invoke(['--context', 'ic0'])

    def test_every_canonical_output_remains_required(self):
        self.composed()
        for name in self.generated:
            with self.subTest(path=name):
                path = self.root / name
                original = path.read_bytes()
                path.write_bytes(b'foreign bytes\n')
                self.invoke(['--context', 'ic0'])
                path.write_bytes(original)

    def test_every_generated_output_remains_exact(self):
        self.composed()
        for name, original in self.generated.items():
            with self.subTest(path=name):
                path = self.crate / 'target/generated' / name
                path.write_bytes(original + b'\ndrift\n')
                self.invoke(['--context', 'ic0'])
                path.write_bytes(original)

    def test_every_guard_remains_required(self):
        self.composed()
        for guard in self.plan['guards']:
            with self.subTest(path=guard['path']):
                path = self.root / guard['path']
                original = path.read_bytes()
                path.write_bytes(original + b'\n# drift\n')
                self.invoke(['--context', 'ic0'])
                path.write_bytes(original)

    def test_missing_generated_output_is_refused(self):
        self.composed()
        (self.crate / 'target/generated' / next(iter(self.generated))).unlink()
        self.invoke(['--context', 'ic0'])

    def test_missing_canonical_output_is_refused(self):
        self.composed()
        (self.root / next(iter(self.generated))).unlink()
        self.invoke(['--context', 'ic0'])

    def test_missing_guard_is_refused(self):
        self.composed()
        (self.root / self.plan['guards'][0]['path']).unlink()
        self.invoke(['--context', 'ic0'])

    def test_canonical_output_symlinks_are_refused(self):
        self.composed()
        saved = self.root / 'saved-output'
        for name in self.generated:
            with self.subTest(path=name):
                path = self.root / name
                path.rename(saved)
                path.symlink_to(saved)
                self.invoke(['--context', 'ic0'])
                path.unlink()
                saved.rename(path)

    def test_canonical_output_ancestor_symlink_is_refused(self):
        self.composed()
        original = self.root / '.github/workflows'
        moved = self.root / 'saved-workflows'
        original.rename(moved)
        original.symlink_to(moved, target_is_directory=True)
        self.invoke(['--context', 'ic0'])

    def test_unfinished_transactions_are_refused(self):
        self.composed()
        for name in ('.m3-migration.lock', '.m3-migration-journal.json'):
            with self.subTest(path=name):
                path = self.root / name
                path.write_bytes(b'pending transaction\n')
                self.invoke(['--context', 'ic0'])
                path.unlink()

    def test_shared_naming_and_notice_extensions_remain_admissible(self):
        self.composed()
        (self.root / NAMING).write_bytes(self.current[NAMING] + b'\nAdditional naming note\n')
        (self.root / LEGAL).write_bytes(b'Additional distribution notice\n' + self.current[LEGAL])
        self.invoke(['--context', 'ic0'], successful=True)


if __name__ == '__main__':
    unittest.main()
