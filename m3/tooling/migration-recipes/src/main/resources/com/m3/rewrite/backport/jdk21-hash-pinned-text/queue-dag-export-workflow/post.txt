name: M3 complete backport queue DAG export

on:
  pull_request:
    paths:
      - "m3/backports/BACKPORT_PASSES.tsv"
      - "m3/backports/BACKPORT_WORK_QUEUE.tsv"
      - "m3/backports/export_recipe_dag.py"
      - "m3/backports/test_export_recipe_dag.py"
      - "m3/docs/backport-recipe-dag-orchestration.md"
      - "m3/tooling/recipe-catalogue.tsv"
      - "m3/tooling/migration-recipes/**"
      - ".github/workflows/m3-backport-queue-dag-export-v2.yml"
  workflow_dispatch:

permissions:
  contents: read

jobs:
  queue-dag:
    runs-on: ubuntu-22.04
    timeout-minutes: 25
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: "21"
          cache: maven

      - name: 1 - Runner and diff probe
        shell: bash
        run: |
          set -euo pipefail
          java -version
          mvn -version
          git diff --check

      - name: 2 - Queue exporter unit proof
        shell: bash
        run: |
          set -euo pipefail
          (
            cd m3/backports
            python3 -m unittest -v test_export_recipe_dag.py
          )

      - name: 3 - Export complete candidate DAG
        shell: bash
        run: |
          set -euo pipefail
          mkdir -p target/backport-queue-dag
          python3 m3/backports/export_recipe_dag.py             --output target/backport-queue-dag/BACKPORT_RECIPE_DAG.tsv             --root-output target/backport-queue-dag/BACKPORT_RECIPE_DAG_ROOT.txt
          python3 - <<'PY'
          import csv
          from pathlib import Path

          queue = list(csv.DictReader(
              Path("m3/backports/BACKPORT_WORK_QUEUE.tsv").open(encoding="utf-8"),
              delimiter="\t"))
          rows = list(csv.DictReader(
              Path("target/backport-queue-dag/BACKPORT_RECIPE_DAG.tsv").open(encoding="utf-8"),
              delimiter="\t"))
          if len(rows) != len(queue) * 7:
              raise SystemExit(f"node count mismatch: queue={len(queue)} nodes={len(rows)}")
          apply = [row for row in rows if row["pass_id"] == "apply"]
          if len(apply) != len(queue):
              raise SystemExit("every queue item must have exactly one apply node")
          if any(row["manifest_mutation_authority"] != "false" for row in rows):
              raise SystemExit("queue evidence acquired mutation authority")
          if any(row["promotion_authority"] != "false" for row in rows):
              raise SystemExit("queue evidence acquired promotion authority")
          invalid = [
              row["identity"]
              for row in apply
              if row["execution_ready"] == "true"
              and row["proof_status"].upper() not in {"VERIFIED", "PASS", "GREEN"}
          ]
          if invalid:
              raise SystemExit(f"unverified apply nodes became ready: {invalid}")
          print(
              "M3_QUEUE_DAG_OK",
              f"items={len(queue)}",
              f"nodes={len(rows)}",
              f"ready={sum(row['execution_ready'] == 'true' for row in apply)}",
          )
          PY

      - name: 4 - OpenRewrite installer and scope proof
        shell: bash
        run: |
          set -euo pipefail
          mvn -B -ntp -f m3/pom.xml             -pl tooling/migration-recipes -am             -Dtest=M3BackportRecipeDagExportRecipeTest,M3RecipeScopeDeclarationTest             -Dsurefire.failIfNoSpecifiedTests=false             test

      - name: 5 - Upload queue DAG evidence
        uses: actions/upload-artifact@v4
        with:
          name: m3jdk21-complete-backport-queue-dag
          path: target/backport-queue-dag
          if-no-files-found: error
          retention-days: 30
