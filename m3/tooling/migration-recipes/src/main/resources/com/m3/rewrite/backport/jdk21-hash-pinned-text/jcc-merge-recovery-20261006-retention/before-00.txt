name: M3 retained capability audit

on:
  push:
    branches:
      - "m3/retained-capability-audit-*"
      - master
    paths:
      - "m3/history/**"
      - "m3/tooling/migration-recipes/src/main/resources/com/m3/rewrite/backport/**"
      - "src/**"
      - "test/**"
      - ".github/workflows/m3-retained-capability-audit.yml"
  pull_request:
    paths:
      - "m3/history/**"
      - "m3/tooling/migration-recipes/src/main/resources/com/m3/rewrite/backport/**"
      - "src/**"
      - "test/**"
      - ".github/workflows/m3-retained-capability-audit.yml"
  workflow_dispatch:

permissions:
  contents: read

jobs:
  retained-capabilities:
    runs-on: ubuntu-22.04
    timeout-minutes: 15
    steps:
      - uses: actions/checkout@v4
        with:
          fetch-depth: 0

      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: "21"
          cache: maven

      - name: Run exact Git-tree retention proof
        shell: bash
        env:
          HEAD_SHA: ${{ github.event.pull_request.head.sha || github.sha }}
        run: |
          set -euo pipefail
          OUT="$RUNNER_TEMP/m3-retention-${GITHUB_RUN_ID}-${GITHUB_RUN_ATTEMPT}"
          mvn -B -ntp -f m3/history/retention-pom.xml verify \
            -Dm3.repo="$GITHUB_WORKSPACE" \
            -Dm3.retentionCommit="$HEAD_SHA" \
            -Dm3.output="$OUT"
          cat "$OUT/SUMMARY.json"

      - name: Upload retention evidence
        if: always()
        uses: actions/upload-artifact@ea165f8d65b6e75b540449e92b4886f43607fa02
        with:
          name: retained-capability-${{ github.run_id }}
          path: ${{ runner.temp }}/m3-retention-${{ github.run_id }}-${{ github.run_attempt }}
          if-no-files-found: warn
          retention-days: 30
