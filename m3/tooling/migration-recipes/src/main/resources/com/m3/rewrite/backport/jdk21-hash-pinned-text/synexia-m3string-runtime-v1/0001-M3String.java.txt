package java.lang;

import java.util.Objects;
import jdk.internal.misc.Unsafe;

/**
 * JDK-internal M3String storage engine for ordinary {@link String} values.
 *
 * <p>The absolute representation invariant is:</p>
 *
 * <pre>
 * String wrapper
 *      -> canonical M3String storage
 *          -> OS-shared lexicon atom
 *          -> VM-local interned atom
 *          -> immutable tuple/range composition of those atoms
 * </pre>
 *
 * <p>Java arrays never resize. Logical concatenation/slicing/repeat changes only immutable
 * atom/range geometry. A contiguous Compact-String {@code byte[]} is a compatibility projection,
 * allocated only when a legacy Java/VM/JNI boundary requires it. Local scalar atoms can expose
 * their already-canonical Compact-String byte array directly; mapped lexicon atoms and joined
 * values materialize lazily.</p>
 */
final class M3String {
    static final byte EMPTY = 0;
    static final byte LOCAL = 1;
    static final byte LEXICON = 2;
    static final byte JOINED = 3;
    static final byte SHARED_LEXICON = 4; // Existing Synexia SYNARR01 UTF-16BE atoms

    private static final byte[] EMPTY_BYTES = new byte[0];
    private static final Unsafe UNSAFE = Unsafe.getUnsafe();
    private static volatile boolean ready;
    private static final ThreadLocal<Boolean> ADMITTING = new ThreadLocal<>();

    static boolean admissionEnabled() {
        return ready && ADMITTING.get() != Boolean.TRUE;
    }

    static void enterAdmission() { ADMITTING.set(Boolean.TRUE); }
    static void leaveAdmission() { ADMITTING.remove(); }

    private static final M3String EMPTY_STORAGE =
            new M3String(
                    EMPTY,
                    null,
                    null,
                    0L,
                    null,
                    new int[0],
                    new int[0],
                    0,
                    String.COMPACT_STRINGS ? String.LATIN1 : String.UTF16,
                    0L,
                    0,
                    M3StringPool.mix64(0L));

    /*
     * These field names/types are part of the M3JDK VM/JDK private contract.
     * HotSpot resolves them in java_lang_M3String.
     */
    private final byte storageKind;
    private final byte[] localValue;
    private final Object mappedOwner;
    private final long mappedAddress;
    private final M3String[] segments;
    private final int[] offsets;
    private final int[] ends;
    private final int length;
    private final byte coder;
    private final long canonicalId;
    private final int javaHash;
    private final long structuralHash64;

    /** Compatibility projection for non-local storage only. */
    private volatile byte[] materialized;

    /** Fixed-size facts retained by this canonical M3 String body, never by String wrappers. */
    private volatile Facts facts;

    private M3String(
            byte storageKind,
            byte[] localValue,
            Object mappedOwner,
            long mappedAddress,
            M3String[] segments,
            int[] offsets,
            int[] ends,
            int length,
            byte coder,
            long canonicalId,
            int javaHash,
            long structuralHash64) {
        this.storageKind = storageKind;
        this.localValue = localValue;
        this.mappedOwner = mappedOwner;
        this.mappedAddress = mappedAddress;
        this.segments = segments;
        this.offsets = offsets;
        this.ends = ends;
        this.length = length;
        this.coder = coder;
        this.canonicalId = canonicalId;
        this.javaHash = javaHash;
        this.structuralHash64 = structuralHash64;
    }

    static void activate(String lexiconFile) {
        if (ready) {
            return;
        }
        synchronized (M3String.class) {
            if (ready) {
                return;
            }
            ADMITTING.set(Boolean.TRUE);
            M3StringPool.initializeLexicon(lexiconFile);
            // Resolve allocation/class initialization before constructor admission.
            M3StringPool.internScalar(new byte[] {65}, String.LATIN1);
            ADMITTING.remove();
            ready = true;
        }
    }

    static boolean ready() {
        return ready;
    }

    static M3String admit(byte[] value, byte coder) {
        Objects.requireNonNull(value, "value");
        if (coder != String.LATIN1 && coder != String.UTF16) {
            throw new IllegalArgumentException("invalid String coder");
        }
        if ((value.length >> coder << coder) != value.length) {
            throw new IllegalArgumentException("misaligned String payload");
        }
        enterAdmission();
        try {
            return M3StringPool.internScalar(value, coder);
        } finally {
            leaveAdmission();
        }
    }

    static M3String emptyStorage() {
        return EMPTY_STORAGE;
    }

    static M3String localScalar(
            byte[] canonicalValue, byte coder, long canonicalId, long structuralHash64) {
        Objects.requireNonNull(canonicalValue, "canonicalValue");
        int length = canonicalValue.length >> coder;
        int hash =
                coder == String.LATIN1
                        ? StringLatin1.hashCode(canonicalValue)
                        : StringUTF16.hashCode(canonicalValue);
        return new M3String(
                LOCAL,
                canonicalValue,
                null,
                0L,
                null,
                null,
                null,
                length,
                coder,
                canonicalId,
                hash,
                structuralHash64);
    }

    static M3String lexiconScalar(
            Object mappedOwner,
            long mappedAddress,
            int length,
            int javaHash,
            long canonicalId,
            long structuralHash64, boolean bigEndian) {
        if (mappedAddress == 0L || length < 0) {
            throw new IllegalArgumentException("invalid mapped M3 atom");
        }
        return new M3String(
                bigEndian ? SHARED_LEXICON : LEXICON,
                null,
                Objects.requireNonNull(mappedOwner, "mappedOwner"),
                mappedAddress,
                null,
                null,
                null,
                length,
                String.COMPACT_STRINGS && mappedLatin1(mappedAddress, length, bigEndian) ? String.LATIN1 : String.UTF16,
                canonicalId,
                javaHash,
                structuralHash64);
    }

    static M3String joinedCanonical(
            M3String[] segments,
            int[] offsets,
            int[] lengths,
            byte coder,
            int logicalLength,
            long canonicalId,
            long structuralHash64) {
        Objects.requireNonNull(segments, "segments");
        Objects.requireNonNull(offsets, "offsets");
        Objects.requireNonNull(lengths, "lengths");
        if (segments.length != offsets.length || segments.length != lengths.length) {
            throw new IllegalArgumentException("M3 tuple lane lengths differ");
        }
        int[] ends = new int[segments.length];
        int total = 0;
        for (int index = 0; index < segments.length; index++) {
            M3String atom = Objects.requireNonNull(segments[index], "segment");
            if (!atom.isScalar()) {
                throw new IllegalArgumentException("joined M3 pieces must be scalar atoms");
            }
            Objects.checkFromIndexSize(offsets[index], lengths[index], atom.length);
            if (lengths[index] <= 0) {
                throw new IllegalArgumentException("empty M3 joined segment");
            }
            total = Math.addExact(total, lengths[index]);
            ends[index] = total;
        }
        if (total != logicalLength) {
            throw new IllegalArgumentException("M3 joined length mismatch");
        }
        M3String provisional =
                new M3String(
                        JOINED,
                        null,
                        null,
                        0L,
                        segments,
                        offsets,
                        ends,
                        logicalLength,
                        coder,
                        canonicalId,
                        0,
                        structuralHash64);
        int hash = provisional.computeJavaHash();
        return new M3String(
                JOINED,
                null,
                null,
                0L,
                segments,
                offsets,
                ends,
                logicalLength,
                coder,
                canonicalId,
                hash,
                structuralHash64);
    }

    static M3String join(String first, String second) {
        return join(new String[] {first, second});
    }

    static M3String join(String[] parts) {
        Objects.requireNonNull(parts, "parts");
        int capacity = 0;
        int totalLength = 0;
        byte resultCoder = String.LATIN1;

        for (String part : parts) {
            Objects.requireNonNull(part, "part");
            M3String storage = part.m3String();
            if (storage == null || storage.length == 0) {
                continue;
            }
            if ((long)totalLength + storage.length > Integer.MAX_VALUE) {
                throw new OutOfMemoryError("Required String length exceeds implementation limit");
            }
            totalLength += storage.length;
            resultCoder |= storage.coder;
            capacity = Math.addExact(capacity, storage.storageKind == JOINED
                    ? storage.segments.length : 1);
        }

        if (totalLength == 0) {
            return EMPTY_STORAGE;
        }

        M3String[] atoms = new M3String[capacity];
        int[] atomOffsets = new int[capacity];
        int[] atomLengths = new int[capacity];
        int count = 0;

        for (String part : parts) {
            M3String storage = part.m3String();
            if (storage == null || storage.length == 0) {
                continue;
            }
            if (storage.storageKind == JOINED) {
                for (int index = 0; index < storage.segments.length; index++) {
                    int previous = index == 0 ? 0 : storage.ends[index - 1];
                    int take = storage.ends[index] - previous;
                    count = addNormalized(
                            atoms,
                            atomOffsets,
                            atomLengths,
                            count,
                            storage.segments[index],
                            storage.offsets[index],
                            take);
                }
            } else {
                count = addNormalized(
                        atoms,
                        atomOffsets,
                        atomLengths,
                        count,
                        storage,
                        0,
                        storage.length);
            }
        }

        if (count == 1
                && atomOffsets[0] == 0
                && atomLengths[0] == atoms[0].length) {
            return atoms[0];
        }

        if (count != capacity) {
            atoms = java.util.Arrays.copyOf(atoms, count);
            atomOffsets = java.util.Arrays.copyOf(atomOffsets, count);
            atomLengths = java.util.Arrays.copyOf(atomLengths, count);
        }
        return M3StringPool.internJoin(
                atoms, atomOffsets, atomLengths, resultCoder, totalLength);
    }

    static M3String sliceOf(String source, int beginIndex, int endIndex) {
        Objects.requireNonNull(source, "source");
        M3String storage = source.m3String();
        if (storage == null) {
            return null;
        }
        return storage.slice(beginIndex, endIndex);
    }

    M3String slice(int beginIndex, int endIndex) {
        Objects.checkFromToIndex(beginIndex, endIndex, length);
        if (beginIndex == 0 && endIndex == length) {
            return this;
        }
        int newLength = endIndex - beginIndex;
        if (newLength == 0) {
            return EMPTY_STORAGE;
        }

        if (isScalar()) {
            return M3StringPool.internJoin(
                    new M3String[] {this},
                    new int[] {beginIndex},
                    new int[] {newLength},
                    rangeCoder(beginIndex, endIndex),
                    newLength);
        }

        int first = segmentAt(beginIndex);
        int last = segmentAt(endIndex - 1);
        int capacity = last - first + 1;
        M3String[] atoms = new M3String[capacity];
        int[] atomOffsets = new int[capacity];
        int[] atomLengths = new int[capacity];
        int count = 0;
        byte resultCoder = String.LATIN1;

        for (int segment = first; segment <= last; segment++) {
            int segmentStart = segment == 0 ? 0 : ends[segment - 1];
            int logicalStart = Math.max(beginIndex, segmentStart);
            int logicalEnd = Math.min(endIndex, ends[segment]);
            int take = logicalEnd - logicalStart;
            M3String atom = segments[segment];
            int atomOffset = offsets[segment] + logicalStart - segmentStart;
            resultCoder |= atom.coder;
            count = addNormalized(
                    atoms, atomOffsets, atomLengths, count, atom, atomOffset, take);
        }

        if (count == 1
                && atomOffsets[0] == 0
                && atomLengths[0] == atoms[0].length) {
            return atoms[0];
        }
        if (count != capacity) {
            atoms = java.util.Arrays.copyOf(atoms, count);
            atomOffsets = java.util.Arrays.copyOf(atomOffsets, count);
            atomLengths = java.util.Arrays.copyOf(atomLengths, count);
        }
        return M3StringPool.internJoin(
                atoms, atomOffsets, atomLengths, rangeCoder(beginIndex, endIndex), newLength);
    }

    M3String repeat(int count) {
        if (count < 0) {
            throw new IllegalArgumentException("count is negative: " + count);
        }
        if (count == 0 || length == 0) {
            return EMPTY_STORAGE;
        }
        if (count == 1) {
            return this;
        }
        int resultLength = Math.multiplyExact(length, count);
        int leafCount = storageKind == JOINED ? segments.length : 1;
        int capacity = Math.multiplyExact(leafCount, count);
        M3String[] atoms = new M3String[capacity];
        int[] atomOffsets = new int[capacity];
        int[] atomLengths = new int[capacity];
        int at = 0;
        for (int repetition = 0; repetition < count; repetition++) {
            if (storageKind == JOINED) {
                for (int segment = 0; segment < segments.length; segment++) {
                    int previous = segment == 0 ? 0 : ends[segment - 1];
                    at = addNormalized(
                            atoms,
                            atomOffsets,
                            atomLengths,
                            at,
                            segments[segment],
                            offsets[segment],
                            ends[segment] - previous);
                }
            } else {
                at = addNormalized(atoms, atomOffsets, atomLengths, at, this, 0, length);
            }
        }
        if (at != capacity) {
            atoms = java.util.Arrays.copyOf(atoms, at);
            atomOffsets = java.util.Arrays.copyOf(atomOffsets, at);
            atomLengths = java.util.Arrays.copyOf(atomLengths, at);
        }
        return M3StringPool.internJoin(
                atoms, atomOffsets, atomLengths, coder, resultLength);
    }

    private byte rangeCoder(int begin, int end) {
        if (!String.COMPACT_STRINGS) return String.UTF16;
        for (int i = begin; i < end; i++) if (charAt(i) > 255) return String.UTF16;
        return String.LATIN1;
    }

    int length() {
        return length;
    }

    byte coder() {
        return coder;
    }

    byte storageKind() {
        return storageKind;
    }

    long canonicalId() {
        return canonicalId;
    }

    long structuralHash64() {
        return structuralHash64;
    }

    boolean isScalar() {
        return storageKind != JOINED;
    }

    boolean isContiguousLocal() {
        return storageKind == LOCAL || storageKind == EMPTY;
    }

    byte[] compatibilityValue() {
        return storageKind == LOCAL ? localValue : EMPTY_BYTES;
    }

    char charAt(int index) {
        Objects.checkIndex(index, length);
        return switch (storageKind) {
            case EMPTY -> throw new StringIndexOutOfBoundsException(index);
            case LOCAL -> coder == String.LATIN1
                    ? StringLatin1.charAt(localValue, index)
                    : StringUTF16.charAt(localValue, index);
            case LEXICON, SHARED_LEXICON -> mappedChar(index);
            case JOINED -> {
                int segment = segmentAt(index);
                int previous = segment == 0 ? 0 : ends[segment - 1];
                yield segments[segment].charAt(offsets[segment] + index - previous);
            }
            default -> throw new InternalError("invalid M3 storage kind");
        };
    }

    int hashCodeValue() {
        return javaHash;
    }

    /** Fixed-size canonical text facts. Length-proportional analyses live in bounded side caches. */
    static final class Facts {
        static final int ASCII=1,LATIN1=1<<1,HAS_SURROGATE=1<<2;
        final int utf16Length,codePointCount,unpairedSurrogateCount,characterFlags;
        final int asciiUpperHash,asciiLowerHash,asciiTitleHash;
        final long prefix4,suffix4,bitSignal64,bigramSignal64,trigramSignal64;
        Facts(int l,int cp,int u,int f,int up,int lo,int ti,long p,long s,long b,long bg,long tg){
            utf16Length=l;codePointCount=cp;unpairedSurrogateCount=u;characterFlags=f;
            asciiUpperHash=up;asciiLowerHash=lo;asciiTitleHash=ti;prefix4=p;suffix4=s;
            bitSignal64=b;bigramSignal64=bg;trigramSignal64=tg;
        }
        boolean mayContain(Facts n){
            if(n.utf16Length>utf16Length)return false;
            if((bitSignal64&n.bitSignal64)!=n.bitSignal64)return false;
            if(n.utf16Length>=2&&(bigramSignal64&n.bigramSignal64)!=n.bigramSignal64)return false;
            return n.utf16Length<3||(trigramSignal64&n.trigramSignal64)==n.trigramSignal64;
        }
    }
    Facts facts(){Facts c=facts;if(c!=null)return c;synchronized(this){c=facts;if(c==null){c=computeFacts();facts=c;}return c;}}
    private Facts computeFacts(){
        if(length==0)return new Facts(0,0,0,Facts.ASCII|Facts.LATIN1,0,0,0,0L,0L,0L,0L,0L);
        int flags=Facts.ASCII|Facts.LATIN1,upper=0,lower=0,title=0,pairs=0,unpaired=0;boolean previousHigh=false;
        long prefix=0L,suffix=0L,bits=0L,bigrams=0L,trigrams=0L;char p2=0,p1=0;
        for(int i=0;i<length;i++){char u=charAt(i);if(u>0x7f)flags&=~Facts.ASCII;if(u>0xff)flags&=~Facts.LATIN1;if(Character.isSurrogate(u))flags|=Facts.HAS_SURROGATE;
            if(Character.isLowSurrogate(u)){if(previousHigh)pairs++;else unpaired++;}else if(previousHigh)unpaired++;previousHigh=Character.isHighSurrogate(u);
            if(i<4)prefix|=(long)u<<(48-(i<<4));suffix=(suffix<<16)|u;upper=31*upper+asciiUpper(u);lower=31*lower+asciiLower(u);title=31*title+(i==0?asciiUpper(u):asciiLower(u));
            bits=addUnitSignal(bits,u);if(i>=1)bigrams=addGramSignal(bigrams,((long)p1<<16)|u);if(i>=2)trigrams=addGramSignal(trigrams,((long)p2<<32)|((long)p1<<16)|u);p2=p1;p1=u;}
        if(previousHigh)unpaired++;return new Facts(length,length-pairs,unpaired,flags,upper,lower,title,prefix,suffix,bits,bigrams,trigrams);
    }
    private static char asciiUpper(char v){return v>='a'&&v<='z'?(char)(v-32):v;}
    private static char asciiLower(char v){return v>='A'&&v<='Z'?(char)(v+32):v;}
    private static long addUnitSignal(long s,char v){int f=asciiLower(v);long a=M3StringPool.mix64(Integer.toUnsignedLong(f)),b=M3StringPool.mix64(Integer.toUnsignedLong(f^0x9e37_79b9));return s|(1L<<((int)a&63))|(1L<<((int)b&63));}
    private static long addGramSignal(long s,long v){long a=M3StringPool.mix64(v),b=M3StringPool.mix64(v^0x9e3779b97f4a7c15L);return s|(1L<<((int)a&63))|(1L<<((int)b&63));}

    static boolean regionEquals(String l,int lo,String r,int ro,int n){Objects.requireNonNull(l);Objects.requireNonNull(r);String.checkBoundsOffCount(lo,n,l.length());String.checkBoundsOffCount(ro,n,r.length());M3String a=l.m3String(),b=r.m3String();if(a!=null&&a==b&&lo==ro)return true;for(int i=0;i<n;i++)if((a==null?l.charAt(lo+i):a.charAt(lo+i))!=(b==null?r.charAt(ro+i):b.charAt(ro+i)))return false;return true;}

    boolean contentEquals(String other) {
        Objects.requireNonNull(other, "other");
        M3String that = other.m3String();
        if (that == this) {
            return true;
        }
        if (other.length() != length) {
            return false;
        }
        for (int index = 0; index < length; index++) {
            if (charAt(index) != other.charAt(index)) {
                return false;
            }
        }
        return true;
    }

    /**
     * Copy one UTF-16 range into the caller-owned compatibility array without flattening the
     * canonical M3 representation first.
     *
     * <p>Joined values resolve their first owning segment once and then advance sequentially
     * through overlapping atom ranges. Mapped scalar values compute their starting address once
     * and advance linearly. This keeps {@link String#toCharArray()} and
     * {@link String#getChars(int, int, char[], int)} as one-allocation materialization boundaries.</p>
     */
    void getChars(int srcBegin, int srcEnd, char[] dst, int dstBegin) {
        String.checkBoundsBeginEnd(srcBegin, srcEnd, length);
        int count = srcEnd - srcBegin;
        Objects.checkFromIndexSize(dstBegin, count, dst.length);
        if (count == 0) {
            return;
        }

        switch (storageKind) {
            case LOCAL -> {
                if (coder == String.LATIN1) {
                    StringLatin1.getChars(localValue, srcBegin, srcEnd, dst, dstBegin);
                } else {
                    StringUTF16.getChars(localValue, srcBegin, srcEnd, dst, dstBegin);
                }
            }
            case LEXICON, SHARED_LEXICON -> {
                long address = mappedAddress + ((long) srcBegin << 1);
                int target = dstBegin;
                for (int remaining = count; remaining > 0;
                        remaining--, target++, address += 2L) {
                    int low = UNSAFE.getByte(address) & 0xff;
                    int high = UNSAFE.getByte(address + 1L) & 0xff;
                    dst[target] = storageKind == SHARED_LEXICON
                            ? (char) ((low << 8) | high)
                            : (char) (low | (high << 8));
                }
            }
            case JOINED -> {
                int logical = srcBegin;
                int target = dstBegin;
                int segment = segmentAt(logical);
                while (logical < srcEnd) {
                    int previous = segment == 0 ? 0 : ends[segment - 1];
                    int segmentEnd = ends[segment];
                    int take = Math.min(srcEnd, segmentEnd) - logical;
                    int atomBegin = offsets[segment] + logical - previous;
                    segments[segment].getChars(atomBegin, atomBegin + take, dst, target);
                    logical += take;
                    target += take;
                    segment++;
                }
            }
            case EMPTY -> throw new InternalError("non-empty range on empty M3String");
            default -> throw new InternalError("invalid M3 storage kind");
        }
    }

    void getBytes(byte[] dst,int srcPos,int dstBegin,byte dstCoder,int count){Objects.checkFromIndexSize(srcPos,count,length);Objects.checkFromIndexSize(dstBegin<<dstCoder,count<<dstCoder,dst.length);if(count==0)return;if(storageKind==LOCAL&&coder==dstCoder){System.arraycopy(localValue,srcPos<<coder,dst,dstBegin<<dstCoder,count<<coder);return;}switch(storageKind){case LOCAL->{for(int i=0;i<count;i++){char u=coder==String.LATIN1?StringLatin1.charAt(localValue,srcPos+i):StringUTF16.charAt(localValue,srcPos+i);if(dstCoder==String.LATIN1)dst[dstBegin+i]=(byte)u;else StringUTF16.putChar(dst,dstBegin+i,u);}}case LEXICON,SHARED_LEXICON->{long a=mappedAddress+((long)srcPos<<1);for(int i=0;i<count;i++,a+=2){int lo=UNSAFE.getByte(a)&255,hi=UNSAFE.getByte(a+1)&255;char u=storageKind==SHARED_LEXICON?(char)((lo<<8)|hi):(char)(lo|(hi<<8));if(dstCoder==String.LATIN1)dst[dstBegin+i]=(byte)u;else StringUTF16.putChar(dst,dstBegin+i,u);}}case JOINED->{int logical=srcPos,target=dstBegin,end=srcPos+count,segment=segmentAt(logical);while(logical<end){int previous=segment==0?0:ends[segment-1],segmentEnd=ends[segment],take=Math.min(end,segmentEnd)-logical,atomBegin=offsets[segment]+logical-previous;segments[segment].getBytes(dst,atomBegin,target,dstCoder,take);logical+=take;target+=take;segment++;}}case EMPTY->throw new InternalError("non-empty byte copy on empty M3String");default->throw new InternalError("invalid M3 String storage kind");}}

    byte[] materialize() {
        if (storageKind == LOCAL) {
            return localValue;
        }
        if (storageKind == EMPTY) {
            return EMPTY_BYTES;
        }
        byte[] cached = materialized;
        if (cached != null) {
            return cached;
        }
        int byteLength = length << coder;
        if ((byteLength >> coder) != length) {
            throw new OutOfMemoryError("Required length exceeds implementation limit");
        }
        byte[] created = new byte[byteLength];
        getBytes(created, 0, 0, coder, length);
        materialized = created;
        return created;
    }

    boolean localContentEquals(byte[] value, byte valueCoder) {
        if (storageKind != LOCAL
                || coder != valueCoder
                || localValue.length != value.length) {
            return false;
        }
        return java.util.Arrays.equals(localValue, value);
    }

    boolean joinGeometryEquals(
            M3String[] atoms, int[] atomOffsets, int[] atomLengths, byte expectedCoder) {
        if (storageKind != JOINED
                || coder != expectedCoder
                || segments.length != atoms.length) {
            return false;
        }
        for (int index = 0; index < segments.length; index++) {
            int previous = index == 0 ? 0 : ends[index - 1];
            if (segments[index] != atoms[index]
                    || offsets[index] != atomOffsets[index]
                    || ends[index] - previous != atomLengths[index]) {
                return false;
            }
        }
        return true;
    }

    private int computeJavaHash() {
        int hash = 0, previous = 0;
        for (int piece = 0; piece < segments.length; piece++) {
            M3String atom = segments[piece];
            int count = ends[piece] - previous;
            int atomHash = atom.javaHash;
            if (offsets[piece] != 0 || count != atom.length) {
                atomHash = 0;
                for (int i = 0; i < count; i++) atomHash = 31 * atomHash + atom.charAt(offsets[piece] + i);
            }
            int power = 1, base = 31;
            for (int remaining = count; remaining != 0; remaining >>>= 1) {
                if ((remaining & 1) != 0) power *= base;
                base *= base;
            }
            hash = hash * power + atomHash;
            previous = ends[piece];
        }
        return hash;
    }

    private static boolean mappedLatin1(long address, int length, boolean bigEndian) {
        for (int i = 0; i < length; i++) if (UNSAFE.getByte(address + ((long)i << 1) + (bigEndian ? 0 : 1)) != 0) return false;
        return true;
    }

    private char mappedChar(int index) {
        long address = mappedAddress + ((long) index << 1);
        int low = UNSAFE.getByte(address) & 0xff;
        int high = UNSAFE.getByte(address + 1L) & 0xff;
        return storageKind == SHARED_LEXICON ? (char)((low << 8) | high) : (char)(low | (high << 8));
    }

    private int segmentAt(int logicalIndex) {
        int low = 0;
        int high = ends.length - 1;
        int key = logicalIndex + 1;
        while (low <= high) {
            int middle = (low + high) >>> 1;
            int end = ends[middle];
            if (end < key) {
                low = middle + 1;
            } else if (middle > 0 && ends[middle - 1] >= key) {
                high = middle - 1;
            } else {
                return middle;
            }
        }
        throw new InternalError("invalid M3String segment coordinate");
    }

    private static int addNormalized(
            M3String[] atoms,
            int[] offsets,
            int[] lengths,
            int count,
            M3String atom,
            int offset,
            int length) {
        if (length == 0) {
            return count;
        }
        if (!atom.isScalar()) {
            throw new IllegalArgumentException("M3 join atom must be scalar");
        }
        Objects.checkFromIndexSize(offset, length, atom.length);
        if (count > 0
                && atoms[count - 1] == atom
                && offsets[count - 1] + lengths[count - 1] == offset) {
            lengths[count - 1] = Math.addExact(lengths[count - 1], length);
            return count;
        }
        atoms[count] = atom;
        offsets[count] = offset;
        lengths[count] = length;
        return count + 1;
    }
