on:
  pull_request:
    paths:
      - 'm3/ports/tornadovm-bulk/**'
      - 'm3/pom.xml'
      - 'm3/compatibility/synexia-tornadovm-bulk-adapter-20261009.tsv'
      - 'm3/docs/name-mapping.json'
      - 'm3/tooling/recipe-catalogue.tsv'
      - 'm3/tooling/migration-recipes/src/main/java/com/m3/rewrite/backport/M3TornadoBulkAdapter*.java'
      - 'm3/tooling/migration-recipes/src/test/java/com/m3/rewrite/backport/M3TornadoBulkAdapter*.java'
      - 'm3/tooling/migration-recipes/src/main/resources/com/m3/rewrite/backport/jdk21-hash-pinned/m3-tornadovm-bulk-adapter/**'
      - 'm3/tooling/migration-recipes/src/main/resources/com/m3/rewrite/backport/jdk21-hash-pinned-text/m3-tornadovm-bulk-adapter/**'
      - 'm3/tooling/migration-recipes/tasks/tornadovm-bulk-adapter/**'
      - '.github/workflows/m3-tornadovm-bulk-adapter.yml'
  workflow_dispatch:

permissions:
  contents: read

jobs:
  recipe-receiver:
    runs-on: ubuntu-24.04
    timeout-minutes: 20
    steps:
      - uses: actions/checkout@v5
      - uses: actions/setup-java@v5
        with:
          distribution: temurin
          java-version: '21'
          cache: maven
      - name: Receiver and mapping recipe tests
        run: >
          mvn -B -ntp -f m3/tooling/migration-recipes/pom.xml
          -Dtest=M3TornadoBulkAdapterPortRecipeTest,M3TornadoBulkAdapterMappingRecipeTest
          test
      - name: Exact receiver fixed point
        run: >
          mvn -B -ntp -f m3/tooling/migration-recipes/pom.xml
          -Drewrite.activeRecipes=com.m3.rewrite.backport.M3TornadoBulkAdapterPortRecipe
          rewrite:dryRun

  cpu-adapter:
    runs-on: ubuntu-24.04
    timeout-minutes: 30
    steps:
      - uses: actions/checkout@v5
      - uses: actions/setup-java@v5
        with:
          distribution: temurin
          java-version: '21'
          cache: maven
      - uses: actions/checkout@v5
        with:
          repository: hsoliwal/synexia-tornadovm
          ref: 3bf826714a48d7fdf346ffde8a0fc25c525572c5
          path: target/synexia-tornadovm
      - name: Install pinned RV32IM provider
        run: >
          mvn -B -ntp -f target/synexia-tornadovm/m3/cpu-gpu/pom.xml
          -DskipTests install
      - name: CPU adapter contracts
        run: >
          mvn -B -ntp -f m3/ports/tornadovm-bulk/pom.xml
          -Dm3.gpu=false test
