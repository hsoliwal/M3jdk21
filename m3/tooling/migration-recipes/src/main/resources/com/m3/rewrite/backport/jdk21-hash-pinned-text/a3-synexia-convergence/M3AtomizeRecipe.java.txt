package com.synexia.rewrite;

import com.fasterxml.jackson.annotation.JsonCreator;
import com.fasterxml.jackson.annotation.JsonProperty;
import java.util.List;
import java.util.Objects;
import java.util.Set;
import org.openrewrite.Recipe;

/**
 * Canonical machine-oriented Atomize recipe.
 *
 * <p>Inventory, structural/code-signal recognition, semantic normal form, relationship patterns
 * and repeated-residue fan-in run before physical extraction. Public donor/problem/recipe evidence
 * is reviewed next. The only behavior-changing child is the trusted-verifier-gated private
 * primitive extractor; unresolved or repeated shapes remain typed recipe backlog.</p>
 */
public final class M3AtomizeRecipe extends Recipe {
    private static final String SPECTRUM_REVIEW_TERM =
            "atom decomposition pattern relationship jdt jsparrow foss semantic operator "
                    + "branch loop nested-loop switch exception synchronization lambda stream "
                    + "record instanceof string regex binary-search sliding-window rolling-hash "
                    + "trie bfs dynamic-programming heap-top-k union-find two-pointers prefix-sum "
                    + "monotonic-stack dfs topological-sort dijkstra backtracking bit-manipulation "
                    + "interval-sweep kmp suffix-array fenwick-tree segment-tree";

    private final String sourceFilePattern;

    public M3AtomizeRecipe() {
        this("**/*.java");
    }

    @JsonCreator
    public M3AtomizeRecipe(
            @JsonProperty("sourceFilePattern") String sourceFilePattern) {
        this.sourceFilePattern = checked(sourceFilePattern);
    }

    @Override
    public String getDisplayName() {
        return "M3 Atomize";
    }

    @Override
    public String getDescription() {
        return "Deterministically decomposes Java into bounded semantic atoms after inventory, "
                + "signal/normal-form/relationship analysis, repeated-residue fan-in and FOSS/donor "
                + "review; extraction is trusted-verifier gated and never grants canonical promotion authority.";
    }

    @Override
    public Set<String> getTags() {
        return Set.of(
                "synexia",
                "m3",
                "machine-oriented-programming",
                "atomize",
                "recipe-first",
                "foss-first",
                "deterministic",
                "proof-gated",
                "candidate-only");
    }

    @Override
    public int maxCycles() {
        return 1;
    }

    @Override
    public boolean causesAnotherCycle() {
        return false;
    }

    @Override
    public List<Recipe> getRecipeList() {
        return List.of(
                new M3RecipeFirstJavaAtomInventoryRecipe(),
                new M3AtomStructureInventoryRecipe(),
                new M3SemanticCodeInventoryRecipe(),
                new M3CodeSplitJavaDonorReviewRecipe(sourceFilePattern),
                new M3CodeSplitRelationshipInventoryRecipe(),
                new M3CodeSignalTriggerRecipe(sourceFilePattern),
                new M3SignalChainRecipe(sourceFilePattern, 4, 4),
                new M3SemanticNormalFormRecipe(sourceFilePattern),
                new M3RepositorySemanticGraphRecipe(),
                new M3RelationshipPatternizerRecipe(),
                new M3AtomPatternFanInRecipe(sourceFilePattern, 2, 4, 4),
                new M3CartesianRecipeIdeaExpansionRecipe(
                        sourceFilePattern,
                        2,
                        4,
                        10_000,
                        4,
                        4),
                new M3ProblemRecipePlannerRecipe(20),
                new M3UniversalRecipeSupersetReviewRecipe(
                        sourceFilePattern,
                        SPECTRUM_REVIEW_TERM),
                new M3DonorBestOfBestSupersetReviewRecipe("ALL"),
                new M3PreparePrivatePrimitiveAtomsRecipe(sourceFilePattern),
                new M3ApplyVerifiedPrivatePrimitiveAtomsRecipe(sourceFilePattern));
    }

    public String getSourceFilePattern() {
        return sourceFilePattern;
    }

    public boolean llmFallbackAllowed() {
        return false;
    }

    public boolean canonicalPromotionAuthority() {
        return false;
    }

    private static String checked(String value) {
        String checked = Objects.toString(value, "").strip();
        if (checked.isEmpty()
                || checked.length() > 4096
                || checked.indexOf('\0') >= 0
                || checked.indexOf('\r') >= 0
                || checked.indexOf('\n') >= 0) {
            throw new IllegalArgumentException("sourceFilePattern");
        }
        return checked;
    }
}
