    public record Source(
            String id,
            URI uri,
            SourceKind kind,
            Admission admission,
            String licenseExpression,
            String revision,
            String path,
            String root) {
        public Source {
            id = token(id, "id");
            uri = Objects.requireNonNull(uri, "uri");
            kind = Objects.requireNonNull(kind, "kind");
            admission = Objects.requireNonNull(admission, "admission");
            licenseExpression = Objects.requireNonNullElse(licenseExpression, "");
            revision = Objects.requireNonNullElse(revision, "");
            path = Objects.requireNonNullElse(path, "");
            if (kind == SourceKind.PLATFORM_CATALOG
                    && admission != Admission.REFERENCE_ONLY) {
                throw new IllegalArgumentException("PLATFORM_CATALOG_MUST_BE_REFERENCE_ONLY");
            }
            if (admission == Admission.SOURCE_ADAPTATION_ALLOWED
                    && (licenseExpression.isBlank() || revision.isBlank() || path.isBlank())) {
                throw new IllegalArgumentException(
                        "ADAPTABLE_SOURCE_REQUIRES_LICENSE_REVISION_PATH");
            }
            final String expected = RepositoryHashing.sha256(String.join("\n",
                    "COMPETITIVE-DONOR-SOURCE/1",
                    id,
                    uri.toString(),
                    kind.name(),
                    admission.name(),
                    licenseExpression,
                    revision,
                    path));
            root = root == null || root.isBlank() ? expected : sha256(root, "root");
            if (!expected.equals(root)) {
                throw new IllegalArgumentException("SOURCE_ROOT_MISMATCH");
            }
        }
    }

    public record Shape(
            String id,
            Category category,
            String problemShape,
            Set<SynexiaTarget> targets,
            List<String> invariants,
            List<String> benchmarkAxes,
            String root) {
        public Shape {
            id = token(id, "id");
            category = Objects.requireNonNull(category, "category");
            problemShape = text(problemShape, "problemShape");
            targets = Set.copyOf(Objects.requireNonNull(targets, "targets"));
            if (targets.isEmpty()) {
                throw new IllegalArgumentException("TARGETS_REQUIRED");
            }
            final ArrayList<String> sortedInvariants =
                    new ArrayList<>(Objects.requireNonNull(invariants, "invariants"));
            sortedInvariants.replaceAll(value -> text(value, "invariant"));
            sortedInvariants.sort(String::compareTo);
            invariants = List.copyOf(sortedInvariants);
            final ArrayList<String> sortedAxes =
                    new ArrayList<>(Objects.requireNonNull(benchmarkAxes, "benchmarkAxes"));
            sortedAxes.replaceAll(value -> text(value, "benchmarkAxis"));
            sortedAxes.sort(String::compareTo);
            benchmarkAxes = List.copyOf(sortedAxes);
            final String expected = RepositoryHashing.sha256(String.join("\n",
                    "COMPETITIVE-DONOR-SHAPE/1",
                    id,
                    category.name(),
                    problemShape,
                    targets.stream().map(Enum::name).sorted().reduce("", (a, b) -> a + "," + b),
                    String.join(",", invariants),
                    String.join(",", benchmarkAxes)));
            root = root == null || root.isBlank() ? expected : sha256(root, "root");
            if (!expected.equals(root)) {
                throw new IllegalArgumentException("SHAPE_ROOT_MISMATCH");
            }
        }
    }

    public record Catalog(
            List<Source> sources,
            List<Shape> shapes,
            String root) {
        public Catalog {
            final ArrayList<Source> sortedSources =
                    new ArrayList<>(Objects.requireNonNull(sources, "sources"));
            sortedSources.sort(Comparator.comparing(Source::id));
            sources = List.copyOf(sortedSources);
            final ArrayList<Shape> sortedShapes =
                    new ArrayList<>(Objects.requireNonNull(shapes, "shapes"));
            sortedShapes.sort(Comparator.comparing(Shape::id));
            shapes = List.copyOf(sortedShapes);
            final StringBuilder material =
                    new StringBuilder("COMPETITIVE-DONOR-CATALOG/1\n");
            sources.forEach(source -> material.append(source.root()).append('\n'));
            shapes.forEach(shape -> material.append(shape.root()).append('\n'));
            final String expected = RepositoryHashing.sha256(material.toString());
            root = root == null || root.isBlank() ? expected : sha256(root, "root");
            if (!expected.equals(root)) {
                throw new IllegalArgumentException("CATALOG_ROOT_MISMATCH");
            }
        }

        public List<Shape> shapesFor(final SynexiaTarget target) {
            return shapes.stream().filter(shape -> shape.targets().contains(target)).toList();
        }

        public List<Shape> shapesFor(final Category category) {
            return shapes.stream().filter(shape -> shape.category() == category).toList();
        }
    }

    private CompetitiveProgrammingDonorCatalog() {}

    private static Shape shape(
            final String id,
            final Category category,
            final String problemShape,
            final Set<SynexiaTarget> targets,
            final List<String> invariants,
            final List<String> benchmarkAxes) {
        return new Shape(
                id,
                category,
                problemShape,
                targets,
                invariants,
                benchmarkAxes,
                "");
    }

    private static String token(final String value, final String field) {
        if (value == null || !value.matches("[a-z0-9][a-z0-9._-]{0,127}")) {
            throw new IllegalArgumentException(
                    "INVALID_" + field.toUpperCase(Locale.ROOT));
        }
        return value;
    }

    private static String text(final String value, final String field) {
        if (value == null || value.isBlank() || value.length() > 4096
                || value.indexOf('\0') >= 0) {
            throw new IllegalArgumentException(
                    "INVALID_" + field.toUpperCase(Locale.ROOT));
        }
        return value;
    }

    private static String hash(final String value) {
        try {
            return HexFormat.of().formatHex(
                    MessageDigest.getInstance("SHA-256")
                            .digest(value.getBytes(StandardCharsets.UTF_8)));
        } catch (NoSuchAlgorithmException impossible) {
            throw new IllegalStateException("SHA256_UNAVAILABLE", impossible);
        }
    }

    private static String sha256(final String value, final String field) {
        if (value == null || !value.matches("[0-9a-f]{64}")) {
            throw new IllegalArgumentException(
                    "INVALID_" + field.toUpperCase(Locale.ROOT));
        }
        return value;
    }
