name: M3 TornadoVM Bulk Adapter GPU

on:
  workflow_dispatch:

permissions:
  contents: read

jobs:
  physical-gpu-parity:
    runs-on: [self-hosted, linux, tornadovm-gpu]
    timeout-minutes: 60
    steps:
      - uses: actions/checkout@v5
      - uses: actions/setup-java@v5
        with:
          distribution: temurin
          java-version: '21'
      - uses: actions/checkout@v5
        with:
          repository: hsoliwal/synexia-tornadovm
          ref: 3bf826714a48d7fdf346ffde8a0fc25c525572c5
          path: target/synexia-tornadovm
      - name: Install pinned RV32IM provider
        run: >
          mvn -B -ntp -f target/synexia-tornadovm/m3/cpu-gpu/pom.xml
          -DskipTests install
      - name: Physical TornadoVM CPU-parity gate
        run: >
          mvn -B -ntp -f m3/ports/tornadovm-bulk/pom.xml
          -Dm3.gpu=true test
