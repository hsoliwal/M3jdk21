name: M3 Parallel Required JNI Bitcount

on:
  pull_request:
    paths:
      - 'm3/ports/parallel-bulk/src/main/java/com/m3/parallel/bulk/M3ParallelNativeBitCountProvider.java'
      - 'm3/ports/parallel-bulk/src/test/java/com/m3/parallel/bulk/M3ParallelNativeBitCount*.java'
      - 'm3/ports/parallel-bulk/native/**'
      - 'm3/ports/parallel-bulk/verification/verify-native-bitcount.sh'
      - 'm3/ports/parallel-bulk/verification/NATIVE_BITCOUNT.md'
      - 'm3/tooling/migration-recipes/src/main/resources/META-INF/rewrite/m3-parallel-native-bitcount.yml'
      - 'm3/tooling/migration-recipes/src/main/resources/com/m3/rewrite/backport/jdk21-hash-pinned/m3-required-native-bitcount-20261014/**'
      - 'm3/tooling/migration-recipes/src/main/resources/com/m3/rewrite/backport/jdk21-hash-pinned-text/m3-required-native-bitcount-20261014/**'
      - 'm3/tooling/migration-recipes/src/test/java/com/m3/rewrite/backport/M3ParallelNativeBitCountReceiverRecipeTest.java'
      - 'm3/tooling/migration-recipes/tasks/m3-parallel-native-bitcount-20261014/**'
      - '.github/workflows/m3-parallel-native-bitcount.yml'
  workflow_dispatch:

permissions:
  contents: read

jobs:
  actual-jni-parity:
    runs-on: ubuntu-24.04
    timeout-minutes: 20
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '21'
      - name: Compile actual JDK owners and call real C11 JNI
        run: bash m3/ports/parallel-bulk/verification/verify-native-bitcount.sh

  module-junit:
    runs-on: ubuntu-24.04
    timeout-minutes: 30
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '21'
      - name: Build source-pinned JUnit dependencies
        run: >
          mvn -B -ntp -f m3/vendor/third_party/pom.xml
          -pl junit-jupiter-5.10.2,junit-platform-launcher-1.10.2
          -am -DskipTests install
      - name: Install existing M3 parallel substrate
        run: mvn -B -ntp -f m3/ports/parallel/pom.xml -DskipTests install
      - name: Build required native library
        shell: bash
        run: |
          set -euo pipefail
          cc -shared -fPIC -std=c11 -Wall -Wextra -Wconversion -Wshadow -Wpedantic -Werror \
            -I"$JAVA_HOME/include" -I"$JAVA_HOME/include/linux" \
            -o "$RUNNER_TEMP/libm3bitcount.so" \
            m3/ports/parallel-bulk/native/m3_native_bitcount_jni.c
      - name: Run real JNI module JUnit
        run: >
          mvn -B -ntp -f m3/ports/parallel-bulk/pom.xml
          -Dtest=M3ParallelNativeBitCountProviderTest
          -Dm3.parallel.bitcount.native.path=${{ runner.temp }}/libm3bitcount.so test

  recipe-fixed-point:
    runs-on: ubuntu-24.04
    timeout-minutes: 30
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '21'
      - name: Bootstrap existing source-built OpenRewrite and JUnit
        run: >
          mvn -B -ntp -f m3/vendor/third_party/pom.xml
          -pl rewrite-java-21,rewrite-test,junit-jupiter-5.10.2,junit-platform-launcher-1.10.2
          -am -DskipTests install
      - name: Run exact hash-pinned receiver and fixed-point test
        run: >
          mvn -B -ntp -f m3/tooling/migration-recipes/pom.xml
          -Dtest=M3ParallelNativeBitCountReceiverRecipeTest test
      - name: Replay canonical source recipe without source edits
        run: >
          mvn -B -ntp -f m3/tooling/migration-recipes/pom.xml
          -Drewrite.activeRecipes=com.m3.rewrite.M3ParallelNativeBitCount
          rewrite:dryRun
