name: M3 A3 absorption proof

on:
  push:
    branches:
      - master
      - "m3/**"
    paths:
      - "m3/tooling/a3/**"
      - "m3/tooling/migration-recipes/**"
      - "m3/backports/JEP_CATALOGUE.tsv"
      - "m3/backports/UPSTREAM_CHANGE_SEEDS.tsv"
      - "m3/backports/COMMUNITY_CAPABILITY_CANDIDATES.tsv"
      - "m3/backports/ALGORITHM_CATALOGUE.tsv"
      - "m3/docs/a3.md"
      - "m3/docs/a3-alg.md"
      - "m3/docs/atom-pattern-admission-and-tooling-boundary.md"
      - "m3/pom.xml"
      - ".github/workflows/m3-a3.yml"
  pull_request:
    paths:
      - "m3/tooling/a3/**"
      - "m3/tooling/migration-recipes/**"
      - "m3/backports/JEP_CATALOGUE.tsv"
      - "m3/backports/UPSTREAM_CHANGE_SEEDS.tsv"
      - "m3/backports/COMMUNITY_CAPABILITY_CANDIDATES.tsv"
      - "m3/backports/ALGORITHM_CATALOGUE.tsv"
      - "m3/docs/a3.md"
      - "m3/docs/a3-alg.md"
      - "m3/docs/atom-pattern-admission-and-tooling-boundary.md"
      - "m3/pom.xml"
      - ".github/workflows/m3-a3.yml"
  workflow_dispatch:

permissions:
  contents: read

jobs:
  a3-proof:
    runs-on: ubuntu-22.04
    timeout-minutes: 35
    steps:
      - name: Fetch exact workflow head
        shell: bash
        env:
          HEAD_SHA: ${{ github.event.pull_request.head.sha || github.sha }}
        run: |
          set -euo pipefail
          test -n "$HEAD_SHA"
          curl --fail --location --retry 3             "https://codeload.github.com/${GITHUB_REPOSITORY}/tar.gz/${HEAD_SHA}"             -o "$RUNNER_TEMP/repo.tar.gz"
          mkdir -p "$RUNNER_TEMP/repo"
          tar -xzf "$RUNNER_TEMP/repo.tar.gz" -C "$RUNNER_TEMP/repo" --strip-components=1
          printf 'REPO=%s\n' "$RUNNER_TEMP/repo" >> "$GITHUB_ENV"

      - name: Prepare pinned JDK 21
        shell: bash
        run: |
          set -euo pipefail
          curl --fail --location --retry 3             'https://github.com/adoptium/temurin21-binaries/releases/download/jdk-21%2B35/OpenJDK21U-jdk_x64_linux_hotspot_21_35.tar.gz'             -o "$RUNNER_TEMP/jdk21.tar.gz"
          printf '%s  %s\n'             '82f64c53acaa045370d6762ebd7441b74e6fda14b464d54d1ff8ca941ec069e6'             "$RUNNER_TEMP/jdk21.tar.gz" | sha256sum --check -
          tar -xzf "$RUNNER_TEMP/jdk21.tar.gz" -C "$RUNNER_TEMP"
          printf 'JAVA_HOME=%s\n' "$RUNNER_TEMP/jdk-21+35" >> "$GITHUB_ENV"
          printf '%s\n' "$RUNNER_TEMP/jdk-21+35/bin" >> "$GITHUB_PATH"

      - name: Build A3 and retained recipe dependencies
        shell: bash
        run: |
          set -euo pipefail
          java -version
          mvn -version
          mvn -B -ntp -f "$REPO/m3/pom.xml"             -pl tooling/a3 -am clean verify

      - name: Inventory whole JDK source and test corpus
        shell: bash
        run: |
          set -euo pipefail
          mvn -B -ntp -f "$REPO/m3/tooling/a3/pom.xml" exec:java             -Dexec.mainClass=com.m3.a3.A3             -Dexec.args="inv --root $REPO --out m3/build/a3/inventory.tsv"
          test -s "$REPO/m3/build/a3/inventory.tsv"
          JAVA_ROWS="$(awk -F '\t' 'NR > 1 && $5 == "JAVA" {n++} END {print n+0}' "$REPO/m3/build/a3/inventory.tsv")"
          NATIVE_ROWS="$(awk -F '\t' 'NR > 1 && $5 == "NATIVE" {n++} END {print n+0}' "$REPO/m3/build/a3/inventory.tsv")"
          test "$JAVA_ROWS" -gt 1000
          test "$NATIVE_ROWS" -gt 100
          printf 'A3 inventory: Java=%s Native=%s\n' "$JAVA_ROWS" "$NATIVE_ROWS" >> "$GITHUB_STEP_SUMMARY"

      - name: Validate algorithm evidence catalogue
        shell: bash
        run: |
          set -euo pipefail
          mvn -B -ntp -f "$REPO/m3/tooling/a3/pom.xml" exec:java             -Dexec.mainClass=com.m3.a3.A3             -Dexec.args="alg --root $REPO --out m3/build/a3/algorithms.tsv"
          ALG="$REPO/m3/build/a3/algorithms.tsv"
          test -s "$ALG"
          ALG_ROWS="$(awk -F '\t' 'NR > 1 {n++} END {print n+0}' "$ALG")"
          STRICT_ROWS="$(awk -F '\t' 'NR > 1 && ($4 == "admitted" || $4 == "candidate" || $4 == "candidate-adapted") {n++} END {print n+0}' "$ALG")"
          STRICT_TRI="$(awk -F '\t' 'NR > 1 && ($4 == "admitted" || $4 == "candidate" || $4 == "candidate-adapted") && $7 == "true" {n++} END {print n+0}' "$ALG")"
          test "$ALG_ROWS" -ge 7
          test "$STRICT_ROWS" -eq "$STRICT_TRI"
          awk -F '\t' 'NR > 1 && $9 != "false" {bad++} END {exit bad ? 1 : 0}' "$ALG"
          printf 'A3 algorithms: rows=%s strict-triplatform=%s\n' "$ALG_ROWS" "$STRICT_TRI" >> "$GITHUB_STEP_SUMMARY"

      - name: Plan every current JEP JBS community and algorithm row
        shell: bash
        run: |
          set -euo pipefail
          mvn -B -ntp -f "$REPO/m3/tooling/a3/pom.xml" exec:java             -Dexec.mainClass=com.m3.a3.A3             -Dexec.args="plan --root $REPO --out m3/build/a3/plan.tsv"
          PLAN="$REPO/m3/build/a3/plan.tsv"
          test -s "$PLAN"
          JEP_ROWS="$(awk -F '\t' 'NR > 1 && $1 == "JEP" {n++} END {print n+0}' "$PLAN")"
          JBS_ROWS="$(awk -F '\t' 'NR > 1 && $1 == "JBS" {n++} END {print n+0}' "$PLAN")"
          CAP_ROWS="$(awk -F '\t' 'NR > 1 && $1 == "CAP" {n++} END {print n+0}' "$PLAN")"
          ALG_ROWS="$(awk -F '\t' 'NR > 1 && $1 == "ALG" {n++} END {print n+0}' "$PLAN")"
          test "$JEP_ROWS" -ge 82
          test "$JBS_ROWS" -ge 13
          test "$CAP_ROWS" -ge 10
          test "$ALG_ROWS" -ge 7
          printf 'A3 plan: JEP=%s JBS=%s CAP=%s ALG=%s\n' "$JEP_ROWS" "$JBS_ROWS" "$CAP_ROWS" "$ALG_ROWS" >> "$GITHUB_STEP_SUMMARY"

      - name: Expand real tree paths into serial FILE work atoms
        shell: bash
        run: |
          set -euo pipefail
          QUEUE="$REPO/m3/build/a3/work-queue.tsv"
          WORK="$REPO/m3/build/a3/work.tsv"
          cat > "$QUEUE" <<'TSV'
          order	release	commit	jbs_ids	subject	domain	inventory_disposition	risk	scope_floor	proof_lane	recipe_strategy	priority	compatibility_state	next_action	paths
          0	27	1111111111111111111111111111111111111111	JDK-A3-WORK	A3 serial work integration	core-libs	candidate	HIGH	MULTI_MODULE	CORE_LIBRARY	MIXED_PACKET_OPENREWRITE_PLUS_VERBATIM	35	PENDING_COMPATIBILITY_PROOF	PROVE_CORE_LIBRARY_BACKPORT	src/java.base/share/classes/java/lang/String.java,src/hotspot/share/runtime/os.cpp,make/autoconf/flags.m4
          TSV

          mvn -B -ntp -f "$REPO/m3/tooling/a3/pom.xml" exec:java \
            -Dexec.mainClass=com.m3.a3.A3 \
            -Dexec.args="work --root $REPO --inventory m3/build/a3/inventory.tsv --queue m3/build/a3/work-queue.tsv --out m3/build/a3/work.tsv"

          test -s "$WORK"
          grep -F $'src/java.base/share/classes/java/lang/String.java\tPRESENT\tJAVA\t' "$WORK"
          grep -F $'\tA3_JAVA_ATOMIZE_PATTERNIZE\tFILE' "$WORK"
          grep -F $'src/hotspot/share/runtime/os.cpp\tPRESENT\tNATIVE\t' "$WORK"
          grep -F $'\tNATIVE_ATOM_REVIEW\tFILE' "$WORK"
          grep -F $'make/autoconf/flags.m4\tOUTSIDE_A3\tOTHER\tUNSCANNED\tOUTSIDE_A3_REVIEW\tFILE' "$WORK"
          grep -F $'\tMULTI_MODULE\t' "$WORK"
          printf 'A3 serial FILE work rows=%s\n' "$(($(wc -l < "$WORK") - 1))" >> "$GITHUB_STEP_SUMMARY"

      - name: Prove guarded fixed-point candidate application
        shell: bash
        run: |
          set -euo pipefail
          FIX="$RUNNER_TEMP/a3-fixture"
          SRC="$FIX/src/java.base/share/classes/example/Sample.java"
          mkdir -p "$(dirname "$SRC")"
          cat > "$SRC" <<'JAVA'
          package example;
          final class Sample {
              private static int compute(int a, int b) {
                  return (a + b) * 31;
              }
          }
          JAVA
          cp "$SRC" "$RUNNER_TEMP/Sample.before"

          mvn -B -ntp -f "$REPO/m3/tooling/a3/pom.xml" exec:java             -Dexec.mainClass=com.m3.a3.A3             -Dexec.args="apply --root $FIX --out m3/build/a3/apply --file src/java.base/share/classes/example/Sample.java"

          cmp "$SRC" "$RUNNER_TEMP/Sample.before"
          CANDIDATE="$FIX/m3/build/a3/apply/candidate/src/java.base/share/classes/example/Sample.java"
          RECEIPT="$FIX/m3/build/a3/apply/receipt.tsv"
          grep -F 'm3$pureIntAtom' "$CANDIDATE"
          grep -F 'M3-IOP: PURE_INT_EXPRESSION' "$CANDIDATE"
          grep -F $'\ttrue\ttrue' "$RECEIPT"
          cp "$RECEIPT" "$REPO/m3/build/a3/apply-receipt.tsv"

      - name: Upload A3 proof
        uses: actions/upload-artifact@ea165f8d65b6e75b540449e92b4886f43607fa02
        with:
          name: m3-a3-proof-${{ github.run_id }}
          path: |
            ${{ env.REPO }}/m3/build/a3/inventory.tsv
            ${{ env.REPO }}/m3/build/a3/algorithms.tsv
            ${{ env.REPO }}/m3/build/a3/plan.tsv
            ${{ env.REPO }}/m3/build/a3/work.tsv
            ${{ env.REPO }}/m3/build/a3/apply-receipt.tsv
          if-no-files-found: error
          retention-days: 30
