import os
from pathlib import Path
import subprocess
import tempfile
import unittest
from unittest.mock import patch
from inventory import scan
from recipe import Refusal
class InventoryTests(unittest.TestCase):
    def setUp(self):
        self.temp=tempfile.TemporaryDirectory();self.root=Path(self.temp.name)
        self.git('init','-q');self.git('config','user.email','fixture@example.invalid');self.git('config','user.name','Fixture')
        for path,text in {'m/pom.xml':'<project/>','m/src/MIndexX.java':'class MIndexX {}','m/native/helper.c':'int helper;','m/test/resource.txt':'resource','other/pom.xml':'<project/>','other/A.java':'class A {}','consumer/pom.xml':'<project/>','consumer/B.java':'MIndexX ref;','README.md':'root'}.items():
            p=self.root/path;p.parent.mkdir(parents=True,exist_ok=True);p.write_text(text)
        self.git('add','.');self.git('commit','-qm','fixture');self.pin=self.git('rev-parse','HEAD').strip()
    def tearDown(self):self.temp.cleanup()
    def git(self,*args):return subprocess.check_output(['git','-C',str(self.root),*args],stderr=subprocess.PIPE,text=True)
    def test_names_behavior_and_resources(self):
        result=scan(self.root,self.pin);self.assertTrue(result['file_accounting_complete']);self.assertFalse(result['inventory_exhaustive'])
        self.assertIn('m/native/helper.c',result['candidate_paths']);self.assertIn('consumer/B.java',result['candidate_paths'])
        self.assertNotIn('other/A.java',result['candidate_paths']);self.assertEqual(result['tracked_entries'],9)
    def test_worktree_changes_do_not_change_pin(self):
        first=scan(self.root,self.pin);(self.root/'m/src/MIndexX.java').write_text('modified')
        self.assertEqual(scan(self.root,self.pin),first)
    def test_exact_pin_required(self):
        with self.assertRaises(Refusal):scan(self.root,'HEAD')
    def test_symlink_not_followed(self):
        os.symlink('/missing-secret',self.root/'m/MIndexLink');self.git('add','.');self.git('commit','-qm','link')
        result=scan(self.root,self.git('rev-parse','HEAD').strip())
        link=next(x for x in result['entries'] if x['path']=='m/MIndexLink');self.assertEqual(link['mode'],'120000');self.assertEqual(link['disposition'],'excluded')
    def test_replay_is_deterministic(self):self.assertEqual(scan(self.root,self.pin),scan(self.root,self.pin))
    def test_whole_repository_has_no_name_filter(self):
        result=scan(self.root,self.pin,'repository')
        self.assertEqual(result['tracked_entries'],len(result['candidate_paths']))
        self.assertIn('other/A.java',result['candidate_paths'])
        self.assertIn('README.md',result['candidate_paths'])
        self.assertTrue(result['all_tracked_entries_selected'])
        self.assertFalse(result['inventory_exhaustive'])
        self.assertFalse(result['dependency_attribution_complete'])
        self.assertFalse(result['semantic_inventory_executed'])
        self.assertFalse(result['production_admitted'])
        self.assertTrue(all(r['disposition']=='partial' for r in result['entries']))
    def test_scope_changes_selection_not_source_identity(self):
        family=scan(self.root,self.pin);whole=scan(self.root,self.pin,'repository')
        identity=lambda r:[(x['path'],x['git_blob'],x['sha256'],x['module']) for x in r['entries']]
        self.assertEqual(identity(family),identity(whole));self.assertEqual(family['tree'],whole['tree'])
        self.assertFalse(family['all_tracked_entries_selected'])
        self.assertEqual(family['selection_scope'],'family')
    def test_whole_scope_keeps_non_java_binary_and_dormant_files(self):
        for path,data in {'off/profile/pom.xml':b'<project/>','off/native/a.c':b'int a;',
                          'off/schema/data.bin':b'\0\xff','off/templates/X.java.in':b'class X {}'}.items():
            p=self.root/path;p.parent.mkdir(parents=True,exist_ok=True);p.write_bytes(data)
        self.git('add','.');self.git('commit','-qm','non Java obligations')
        result=scan(self.root,self.git('rev-parse','HEAD').strip(),'repository')
        self.assertEqual(len(result['entries']),13)
        self.assertEqual(result['tracked_entries'],len(result['candidate_paths']))
        self.assertTrue(all(x['sha256'] for x in result['entries']))
    def test_whole_scope_keeps_symlink_and_submodule_blocked(self):
        os.symlink('/missing-secret',self.root/'outside')
        self.git('add','outside')
        self.git('update-index','--add','--cacheinfo','160000',self.pin,'vendor/submodule')
        self.git('commit','-qm','indirect obligations')
        result=scan(self.root,self.git('rev-parse','HEAD').strip(),'repository')
        by_path={x['path']:x for x in result['entries']}
        for path in ['outside','vendor/submodule']:
            self.assertEqual(by_path[path]['disposition'],'blocked')
            self.assertIn(path,result['candidate_paths']);self.assertIn(path,result['bounded_scan_gaps'])
        self.assertIsNone(by_path['vendor/submodule']['sha256'])
        self.assertIsNotNone(by_path['outside']['sha256'])
    def test_oversize_remains_hashed_and_blocked(self):
        with patch('inventory.MAX_TEXT',1):result=scan(self.root,self.pin,'repository')
        self.assertEqual(result['tracked_entries'],len(result['candidate_paths']))
        self.assertTrue(all(x['sha256'] and x['disposition']=='blocked' for x in result['entries']))
    def test_repository_scope_is_exact_pin_and_deterministic(self):
        first=scan(self.root,self.pin,'repository')
        (self.root/'other/A.java').write_text('dirty')
        (self.root/'untracked').write_text('not admitted')
        self.assertEqual(first,scan(self.root,self.pin,'repository'))
    def test_unknown_scope_refused(self):
        with self.assertRaises(Refusal):scan(self.root,self.pin,'everything-is-ready')
if __name__=='__main__':unittest.main()
