name: MIndex JDK mapped String backing
on:
  pull_request:
    paths:
      - 'src/java.base/share/classes/jdk/internal/mindex/**'
      - 'test/jdk/java/lang/String/MIndexMappedStringBackingTest.java'
      - 'doc/mindex-string-backing.md'
      - '.github/workflows/mindex-string-backing.yml'
  workflow_dispatch:

permissions:
  contents: read

jobs:
  jdk-backing:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '20'
      - name: Install build prerequisites
        run: sudo apt-get update && sudo apt-get install -y autoconf
      - name: Configure
        run: bash configure --with-debug-level=fastdebug --with-jvm-variants=server
      - name: Build java.base
        run: make java.base-java-only
      - name: Run MIndex String backing jtreg
        run: make test TEST=test/jdk/java/lang/String/MIndexMappedStringBackingTest.java
