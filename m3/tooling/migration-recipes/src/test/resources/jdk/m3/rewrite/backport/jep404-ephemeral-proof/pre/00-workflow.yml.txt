name: M3 JEP 404 Generational Shenandoah inventory

on:
  push:
    branches:
      - "m3/jep404-*"
    paths:
      - "m3/backports/file_delta_inventory.py"
      - "m3/backports/generate_recipe_crates.py"
      - "m3/backports/selected_path_inventory.py"
      - "m3/backports/test_selected_path_inventory.py"
      - "m3/backports/test_jep404_inventory_policy.py"
      - "m3/backports/recipes/jep-404-generational-shenandoah/**"
      - ".github/workflows/m3-jep404-generational-shenandoah-inventory.yml"
  pull_request:
    paths:
      - "m3/backports/file_delta_inventory.py"
      - "m3/backports/generate_recipe_crates.py"
      - "m3/backports/selected_path_inventory.py"
      - "m3/backports/test_selected_path_inventory.py"
      - "m3/backports/test_jep404_inventory_policy.py"
      - "m3/backports/recipes/jep-404-generational-shenandoah/**"
      - ".github/workflows/m3-jep404-generational-shenandoah-inventory.yml"
  workflow_dispatch:

permissions:
  contents: read

jobs:
  inventory:
    runs-on: ubuntu-22.04
    timeout-minutes: 70
    env:
      IMPLEMENTATION_COMMIT: 28ae281b42cd00f471e275db544a5d23a42df59c
    steps:
      - uses: actions/checkout@v5
        with:
          persist-credentials: false

      - uses: actions/setup-java@v5
        with:
          distribution: temurin
          java-version: "21"
          cache: maven

      - name: 1 - Diff hygiene
        shell: bash
        run: |
          set -euo pipefail
          git diff --check

      - name: 2 - Prove reusable selected-path inventory and JEP404 policy
        shell: bash
        run: |
          set -euo pipefail
          python3 -m py_compile             m3/backports/file_delta_inventory.py             m3/backports/generate_recipe_crates.py             m3/backports/selected_path_inventory.py
          cd m3/backports
          python3 -m unittest -v             test_selected_path_inventory.py             test_jep404_inventory_policy.py

      - name: 3 - Fetch pinned Java21 and Java24 GA donor history
        shell: bash
        run: |
          set -euo pipefail
          mkdir "$RUNNER_TEMP/upstream-jdk"
          git -C "$RUNNER_TEMP/upstream-jdk" init -q
          git -C "$RUNNER_TEMP/upstream-jdk" remote add origin https://github.com/openjdk/jdk.git
          git -C "$RUNNER_TEMP/upstream-jdk" -c protocol.version=2 fetch             --filter=blob:none --no-tags origin             "refs/tags/jdk-21+35:refs/tags/jdk-21+35"             "refs/tags/jdk-24+36:refs/tags/jdk-24+36"
          git -C "$RUNNER_TEMP/upstream-jdk" cat-file -e "$IMPLEMENTATION_COMMIT^{commit}"
          git -C "$RUNNER_TEMP/upstream-jdk" merge-base --is-ancestor             "$IMPLEMENTATION_COMMIT" "jdk-24+36"
          printf 'UPSTREAM=%s\n' "$RUNNER_TEMP/upstream-jdk" >> "$GITHUB_ENV"

      - name: 4 - Compare Java21 against integration commit and final JDK24 GA
        shell: bash
        run: |
          set -euo pipefail
          mkdir -p "$RUNNER_TEMP/jep404"
          python3 m3/backports/file_delta_inventory.py             --repo "$UPSTREAM"             --release 24             --donor-ref "$IMPLEMENTATION_COMMIT"             --out "$RUNNER_TEMP/jep404/integration-delta.tsv"
          python3 m3/backports/file_delta_inventory.py             --repo "$UPSTREAM"             --release 24             --out "$RUNNER_TEMP/jep404/ga-delta.tsv"

      - name: 5 - Join exact 228-path denominator to final GA state
        shell: bash
        run: |
          set -euo pipefail
          python3 m3/backports/selected_path_inventory.py             --paths m3/backports/recipes/jep-404-generational-shenandoah/INTEGRATION_PATHS.txt             --integration-delta "$RUNNER_TEMP/jep404/integration-delta.tsv"             --ga-delta "$RUNNER_TEMP/jep404/ga-delta.tsv"             --out "$RUNNER_TEMP/jep404/selected-paths.tsv"
          python3 - <<'PY'
          import csv
          import os
          from collections import Counter
          from pathlib import Path

          root = Path(os.environ["RUNNER_TEMP"]) / "jep404"
          with (root / "selected-paths.tsv").open(encoding="utf-8", newline="") as handle:
              rows = list(csv.DictReader(handle, delimiter="\t"))
          assert len(rows) == 228, len(rows)
          assert len({row["path"] for row in rows}) == 228
          counts = Counter(row["disposition"] for row in rows)
          admitted = [
              row["path"]
              for row in rows
              if row["disposition"] == "MATERIALIZE_GA_POSTIMAGE"
          ]
          (root / "MATERIALIZE_PATHS.txt").write_text(
              "".join(path + "\n" for path in admitted),
              encoding="utf-8",
          )
          print("JEP404_SELECTED " + " ".join(f"{k}={v}" for k, v in sorted(counts.items())))
          print(f"JEP404_MATERIALIZE_PATHS {len(admitted)}")
          if not admitted:
              raise SystemExit("JEP404 has no final-GA materialization paths")
          PY

      - name: 6 - Generate one-file Java native and text recipe evidence
        shell: bash
        run: |
          set -euo pipefail
          python3 m3/backports/generate_recipe_crates.py             --repo "$UPSTREAM"             --release 24             --paths-file "$RUNNER_TEMP/jep404/MATERIALIZE_PATHS.txt"             --crate-size 1             --include-native             --include-text             --out "$RUNNER_TEMP/jep404/file-atoms"

      - name: 7 - Prove generator accounting without applying candidate
        shell: bash
        run: |
          set -euo pipefail
          python3 - <<'PY'
          import csv
          import os
          from pathlib import Path

          root = Path(os.environ["RUNNER_TEMP"]) / "jep404"
          selected = {
              line.strip()
              for line in (root / "MATERIALIZE_PATHS.txt").read_text(encoding="utf-8").splitlines()
              if line.strip()
          }
          with (root / "file-atoms" / "CRATES.tsv").open(encoding="utf-8", newline="") as handle:
              crates = list(csv.DictReader(handle, delimiter="\t"))
          with (root / "file-atoms" / "EXCLUSIONS.tsv").open(encoding="utf-8", newline="") as handle:
              exclusions = list(csv.DictReader(handle, delimiter="\t"))

          generated = sum(int(row["target_count"]) for row in crates)
          assert all(int(row["target_count"]) == 1 for row in crates), "JEP404 requires FILE crates"
          if generated + len(exclusions) != len(selected):
              raise SystemExit(
                  f"JEP404 generator accounting drift generated={generated} "
                  f"exclusions={len(exclusions)} selected={len(selected)}"
              )
          print(
              f"JEP404_RECIPE_EVIDENCE selected={len(selected)} "
              f"generated={generated} typed_exclusions={len(exclusions)}"
          )
          PY
          test -z "$(git status --porcelain=v1 -uall)"

      - name: 8 - Verify existing M3 backport control plane
        shell: bash
        run: |
          set -euo pipefail
          mvn -B -ntp -f m3/backports/pom.xml verify

      - name: 9 - Upload JEP404 inventory and ephemeral recipe evidence
        if: always()
        uses: actions/upload-artifact@ea165f8d65b6e75b540449e92b4886f43607fa02
        with:
          name: jep404-inventory-${{ github.run_id }}
          path: ${{ runner.temp }}/jep404
          if-no-files-found: warn
          retention-days: 30
