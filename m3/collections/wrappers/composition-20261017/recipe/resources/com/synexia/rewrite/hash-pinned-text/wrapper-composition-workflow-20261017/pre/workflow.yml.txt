name: M3JDK21 stored-representative filtered wrappers
on:
  pull_request:
    paths:
      - '.github/workflows/m3jdk-stored-representative-20261017.yml'
      - 'm3/collections/wrappers/**'
      - 'm3/collections/src/main/java/com/m3/collections/M3CollectionViews.java'
  workflow_dispatch:
permissions:
  contents: read
jobs:
  exact-head-recipe-and-java-jni:
    runs-on: ubuntu-24.04
    timeout-minutes: 35
    steps:
      - uses: actions/checkout@v5
        with:
          ref: ${{ github.event.pull_request.head.sha || github.sha }}
          persist-credentials: false
          fetch-depth: 1
          sparse-checkout: |
            m3/collections
      - uses: actions/setup-java@v5
        with:
          distribution: temurin
          java-version: '21'
      - name: Prove predecessor and successor recipes using the real OpenRewrite SDK
        shell: bash
        run: |
          set -euo pipefail
          mvn -B -ntp -f m3/collections/wrappers/recipe/pom.xml test \
            -Dtest=M3WrapperContractsRecipeTest,M3StoredRepresentativeRecipeTest \
            -Dm3.wrapper.sdk.output="$RUNNER_TEMP/m3-wrapper-sdk" \
            -Dm3.stored.sdk.output="$RUNNER_TEMP/m3-stored-sdk"
      - name: Verify SDK emissions and exact installed source, Java21 and unchanged JNI ABI
        shell: bash
        run: |
          set -euo pipefail
          mkdir -p "$RUNNER_TEMP/proof"/{classes,baseline,headers}
          python3 - <<'PY'
          import hashlib,os
          from pathlib import Path
          root=Path.cwd()
          temp=Path(os.environ['RUNNER_TEMP'])
          r=root/'m3/collections/wrappers/recipe/src/main/resources/com/synexia/rewrite/hash-pinned-java/m3-stored-representative-20261017'
          prev=root/'m3/collections/wrappers/recipe/src/main/resources/com/synexia/rewrite/hash-pinned-java/m3-wrapper-contracts-20261017'
          rows=[line.split('\t') for line in (r/'manifest.tsv').read_text().splitlines()]
          assert len(rows)==2 and all(len(row)==4 for row in rows)
          h=lambda p:hashlib.sha256(Path(p).read_bytes()).hexdigest()
          sdk=temp/'m3-stored-sdk'
          assert (sdk/'SDK_RECEIPT.tsv').is_file()
          assert (temp/'m3-wrapper-sdk'/'SDK_RECEIPT.tsv').is_file()
          for path,before,after,template in rows:
              assert h(root/path)==after,(path,'installed')
              assert h(sdk/path)==after,(path,'SDK')
              assert h(r/template)==after,(path,'template')
              if before!='ABSENT':
                  assert h(r/'pre'/template)==before
                  assert h(prev/template)==before
          print('M3_STORED_REPRESENTATIVE_SDK_RECEIVER_SEAL_PASS targets=2')
          PY
          root=m3/collections
          src=$root/src/main/java/com/m3/collections
          tests=$root/wrappers/src/test/java/com/m3/collections
          proof="$RUNNER_TEMP/proof"
          javac --release 21 -proc:none -Xlint:all -Werror -h "$proof/headers" -d "$proof/classes" \
            "$src/M3CollectionViews.java" "$src/M3FilteredMaps.java" \
            "$tests/M3StoredRepresentativeProof.java" \
            "$tests/M3WrapperContractsProof.java" \
            "$tests/M3DeepFilterCompositionProof.java" \
            "$tests/M3DeepFilterNativeProof.java"
          java -ea -Xss256k -Xmx256m -cp "$proof/classes" com.m3.collections.M3StoredRepresentativeProof | tee "$proof/new-tiered.log"
          java -ea -Xint -Xss256k -Xmx256m -cp "$proof/classes" com.m3.collections.M3StoredRepresentativeProof | tee "$proof/new-interpreter.log"
          java -ea -Xss256k -Xmx256m -cp "$proof/classes" com.m3.collections.M3WrapperContractsProof | tee "$proof/old-wrapper.log"
          java -ea -Xint -Xss256k -Xmx256m -cp "$proof/classes" com.m3.collections.M3WrapperContractsProof | tee "$proof/old-wrapper-int.log"
          java -ea -Xss256k -Xmx256m -Dm3.deep.depth=65536 -cp "$proof/classes" com.m3.collections.M3DeepFilterCompositionProof | tee "$proof/deep.log"
          java -ea -Xint -Xss256k -Xmx256m -Dm3.deep.depth=8192 -cp "$proof/classes" com.m3.collections.M3DeepFilterCompositionProof | tee "$proof/deep-int.log"
          cp "$root/wrappers/recipe/src/main/resources/com/synexia/rewrite/hash-pinned-java/m3-stored-representative-20261017/pre/M3CollectionViews.java.txt" "$proof/baseline/M3CollectionViews.java"
          javac --release 21 -proc:none -Xlint:all -Werror -d "$proof/baseline" "$proof/baseline/M3CollectionViews.java" "$src/M3FilteredMaps.java" "$tests/M3StoredRepresentativeProof.java"
          if java -ea -Xss256k -Xmx256m -cp "$proof/baseline" com.m3.collections.M3StoredRepresentativeProof >"$proof/red.log" 2>&1; then
            echo 'Predecessor unexpectedly passed failing regression'; exit 1
          fi
          grep -q 'search stored representative not external probe' "$proof/red.log"
          javap -public -s -cp "$proof/baseline" com.m3.collections.M3CollectionViews > "$proof/public-before.txt"
          javap -public -s -cp "$proof/classes" com.m3.collections.M3CollectionViews > "$proof/public-after.txt"
          diff -u "$proof/public-before.txt" "$proof/public-after.txt"
          java_home="$(dirname "$(dirname "$(readlink -f "$(command -v javac)")")")"
          gcc -std=c17 -O2 -Wall -Wextra -Werror -pedantic -fPIC -shared \
            -fsanitize=undefined -fno-sanitize-recover=undefined \
            -I"$java_home/include" -I"$java_home/include/linux" \
            "$root/wrappers/direct-projection-20261017/m3_map_cursor_jni.c" \
            "$root/wrappers/deep-filter-20261017/native/m3_deep_filter_jni.c" \
            -o "$proof/libm3_stored_representative_jni.so"
          java -ea -Xss256k -Xmx256m -Xcheck:jni \
            -Dm3.deep.native.library="$proof/libm3_stored_representative_jni.so" \
            -cp "$proof/classes" com.m3.collections.M3DeepFilterNativeProof | tee "$proof/jni.log"
          grep -q 'checks=86866 seeds=1200' "$proof/new-tiered.log"
          grep -q 'checks=86866 seeds=1200' "$proof/new-interpreter.log"
          grep -q 'checks=156 depth=65536' "$proof/jni.log"
          echo 'M3_STORED_REPRESENTATIVE_EXACT_HEAD_PASS'
      - uses: actions/upload-artifact@v4
        if: always()
        with:
          name: m3-stored-representative-proof
          path: |
            ${{ runner.temp }}/proof/
            m3/collections/wrappers/recipe/target/surefire-reports/
          retention-days: 7
