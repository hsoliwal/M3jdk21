  # Modified 2026-10-17: add bounded wrapper recipe replay and Java/JNI receiving checks.
  wrapper-reconciliation:
    name: Wrapper recipe fixed point and Java 21 JNI
    runs-on: ubuntu-22.04
    timeout-minutes: 30
    steps:
      - uses: actions/checkout@11d5960a326750d5838078e36cf38b85af677262
        with:
          persist-credentials: false
      - name: Prepare pinned Java 21 compiler
        shell: bash
        run: |
          curl --fail --location --retry 3 'https://github.com/adoptium/temurin21-binaries/releases/download/jdk-21.0.12.1%2B1/OpenJDK21U-jdk_x64_linux_hotspot_21.0.12.1_1.tar.gz' -o "$RUNNER_TEMP/m3-jdk21.tar.gz"

      - name: Verify mirrored wrapper recipe with the actual SDK
        run: mvn -B -ntp -f m3/collections/wrappers/recipe/pom.xml verify
      - name: Require a fixed point on the committed target tree
        shell: bash
        run: |
          export M3_WRAPPER_RECEIPT="$RUNNER_TEMP/m3-collections-wrapper-fixed-point.tsv"
          mvn -B -ntp \
            -f m3/collections/wrappers/recipe/synexia-openrewrite-recipes/recipes/m3-purpose-wrapper-receiver-20261017/master-reconciliation/pom.xml \
            exec:java \
            -Dexec.mainClass=com.synexia.rewrite.M3CollectionWrapperReconciliationApply \
            -Dexec.args="\"$GITHUB_WORKSPACE\" \"$M3_WRAPPER_RECEIPT\""
          python3 - <<'PY'
          import csv
          import os
          from pathlib import Path

          receipt = Path(os.environ['M3_WRAPPER_RECEIPT']).resolve()
          root = Path(os.environ['GITHUB_WORKSPACE']).resolve()
          assert not receipt.is_relative_to(root), 'receipt must be outside the target tree'
          with receipt.open(newline='', encoding='utf-8') as stream:
              rows = list(csv.DictReader(stream, delimiter='\t'))
          assert rows, 'missing SDK source receipt rows'
          assert len({row['target_path'] for row in rows}) == len(rows), 'duplicate SDK target rows'
          for row in rows:
              assert row['proof_scope'] == 'SDK_EXECUTION', row['target_path']
              assert row['first_pass_changes'] == row['second_pass_changes'] == '0', row['target_path']
              assert row['observed_before_sha256'] == row['sealed_after_sha256'] == row['actual_after_sha256'], row['target_path']
          print(f'M3_WRAPPER_COMMITTED_FIXED_POINT_PASS targets={len(rows)}')
          PY
      - name: Verify wrapper contracts and required Linux JNI
        run: mvn -B -ntp -f m3/collections/wrappers/pom.xml -Pjni-linux verify
      - name: Retain wrapper recipe and Java JNI evidence

        uses: actions/upload-artifact@ea165f8d65b6e75b540449e92b4886f43607fa02
