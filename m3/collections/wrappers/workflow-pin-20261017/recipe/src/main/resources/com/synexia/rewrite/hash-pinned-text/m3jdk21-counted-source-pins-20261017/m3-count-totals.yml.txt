name: M3 inline count totals
on:
  push:
    branches: [aix/m3-inline-count-totals-20261016]
  pull_request:
    paths:
      - 'm3/collections/**'
      - '.github/workflows/m3-count-totals.yml'
permissions:
  contents: read
concurrency:
  group: m3-count-totals-${{ github.head_ref || github.ref_name }}
  cancel-in-progress: true
jobs:
  recipe-and-java-jni:
    runs-on: ubuntu-24.04
    timeout-minutes: 25
    env:
      PYTHONOPTIMIZE: '0'
    steps:
      - uses: actions/checkout@11d5960a326750d5838078e36cf38b85af677262
        with:
          ref: ${{ github.event.pull_request.head.sha || github.sha }}
          persist-credentials: false
          sparse-checkout: m3/collections
      - uses: actions/setup-java@cf277c60eb25467037889841efdb72551f06f6c3
        with:
          distribution: temurin
          java-version: '21'
      - name: Execute retained predecessor and refinement Maven recipes
        shell: bash
        run: |
          set -euo pipefail
          mkdir -p proof/deps proof/core
          { git rev-parse HEAD; uname -m; java -version 2>&1; mvn -version; gcc --version; } > proof/identity.txt
          test "$(git hash-object m3/collections/counts/recipe/pinned-engine/com/synexia/rewrite/M3HashPinnedJavaSnapshotRecipe.java)" = d58ca50dfaeb60f581b64f43545e688aca5c6530
          test "$(git hash-object m3/collections/counts/recipe/pinned-engine/com/synexia/rewrite/M3HashPinnedTextSnapshotRecipe.java)" = 8bd6f399b298f364a41d53391ba2e72da4c5c635
          mvn -B -ntp -f m3/collections/counts/recipe/pom.xml -Dm3.counted.target="$GITHUB_WORKSPACE" test 2>&1 | tee proof/predecessor-recipe.log
          mvn -B -ntp -f m3/collections/counts/totals/recipe/pom.xml -Dm3.totals.target="$GITHUB_WORKSPACE" test 2>&1 | tee proof/totals-recipe.log
          python3 - <<'PY'
          from pathlib import Path
          import xml.etree.ElementTree as E
          for owner in ['m3/collections/counts/recipe','m3/collections/counts/totals/recipe']:
              files=list(Path(owner+'/target/surefire-reports').glob('TEST-*.xml'))
              totals={k:sum(int(E.parse(p).getroot().attrib.get(k,0)) for p in files) for k in ('tests','failures','errors','skipped')}
              assert totals=={'tests':8,'failures':0,'errors':0,'skipped':0},(owner,totals)
              print(owner,totals)
          PY
          git diff --exit-code -- m3/collections
      - name: Build current core and resolve existing interoperability dependencies
        shell: bash
        run: |
          set -euo pipefail
          find m3/collections/src/main/java -name '*.java' | sort > proof/core-sources.txt
          javac --release 21 -Xlint:all,-module -Werror -d proof/core @proof/core-sources.txt
          jar --create --file proof/m3-collections.jar -C proof/core .
          JAR="$GITHUB_WORKSPACE/proof/m3-collections.jar"
          (cd "$RUNNER_TEMP" && mvn -B -ntp org.apache.maven.plugins:maven-install-plugin:3.1.3:install-file -Dfile="$JAR" -DgroupId=com.m3 -DartifactId=m3-collections -Dversion=0.1.0-SNAPSHOT -Dpackaging=jar -DgeneratePom=true)
          mvn -B -ntp -f m3/collections/interop/pom.xml verify org.apache.maven.plugins:maven-dependency-plugin:3.8.1:copy-dependencies -DincludeScope=runtime -DoutputDirectory="$GITHUB_WORKSPACE/proof/deps" 2>&1 | tee proof/interop.log
          curl --fail --location --retry 3 https://repo.maven.apache.org/maven2/org/junit/platform/junit-platform-console-standalone/1.12.2/junit-platform-console-standalone-1.12.2.jar -o proof/junit.jar
          echo '329bd10288875a74d04c9ca6b7c9889c265ddf87b21a9ea42a7f7392f391472a  proof/junit.jar' | sha256sum --check -
      - name: Run all retained count proofs plus scalar JNI and allocation diagnostics
        shell: bash
        run: |
          set -euo pipefail
          python3 m3/collections/counts/totals/verify.py --root "$GITHUB_WORKSPACE" --deps proof/deps --junit proof/junit.jar 2>&1 | tee proof/verification.log
          git diff --exit-code -- m3/collections
          find proof m3/collections/counts/totals -type f ! -name SHA256SUMS -print0 | sort -z | xargs -0 sha256sum > proof/SHA256SUMS
      - name: Preserve actual source and evidence including failures
        if: always()
        uses: actions/upload-artifact@ea165f8d65b6e75b540449e92b4886f43607fa02
        with:
          name: m3-count-totals-${{ github.event.pull_request.head.sha || github.sha }}
          retention-days: 30
          path: |
            proof/
            m3/collections/counts/recipe/target/surefire-reports/
            m3/collections/counts/target/verification/evidence/
            m3/collections/counts/totals/
            m3/collections/src/main/java/com/m3/collections/M3CountTotals.java
            m3/collections/src/main/native/collections/m3_count_totals.c
