jobs:
  recipe-java-jni-contracts:
    runs-on: ubuntu-24.04
    timeout-minutes: 30
    env:
      PYTHONOPTIMIZE: '0'
    steps:
      - uses: actions/checkout@11d5960a326750d5838078e36cf38b85af677262
        with:
          ref: ${{ github.event.pull_request.head.sha || github.sha }}
          persist-credentials: false
          sparse-checkout: |
            m3/collections
      - uses: actions/setup-java@cf277c60eb25467037889841efdb72551f06f6c3
        with:
          distribution: temurin
          java-version: '21'
      - name: Identify exact head and require sealed recipe provider tree
        shell: bash
        run: |
          set -euo pipefail
          mkdir -p proof/deps proof/core
          { git rev-parse HEAD; uname -m; java -version 2>&1; mvn -version; gcc --version; } > proof/identity.txt
          test "$(git rev-parse HEAD:m3/collections/counts/recipe)" = 45d5b77bc9b18e02922eb0e69ef246a8bdff05c4
      - name: Execute all eight Maven OpenRewrite proofs and exact output replay
        shell: bash
        run: |
          set -euo pipefail
          mvn -B -ntp -f m3/collections/counts/recipe/pom.xml test 2>&1 | tee proof/recipe.log
          cp -R m3/collections/counts/recipe/target/surefire-reports proof/recipe-reports
          python3 - <<'PY'
          from pathlib import Path
          import xml.etree.ElementTree as E
          reports = list(Path('proof/recipe-reports').glob('TEST-*.xml'))
          assert reports, 'missing actual recipe execution reports'
          totals = {k: sum(int(E.parse(p).getroot().attrib.get(k, 0)) for p in reports) for k in ('tests', 'failures', 'errors', 'skipped')}
          assert totals == {'tests': 8, 'failures': 0, 'errors': 0, 'skipped': 0}, totals
          print('RECIPE_EXECUTED', totals)
          PY
          mvn -B -ntp -f m3/collections/counts/recipe/pom.xml -Dm3.counted.target="$GITHUB_WORKSPACE" test 2>&1 | tee proof/replay.log
          git diff --exit-code -- m3/collections
      - name: Compile current core and resolve the existing optional library closure
        shell: bash
        run: |
          set -euo pipefail
          find m3/collections/src/main/java -name '*.java' | sort > proof/core-sources.txt
          javac --release 21 -Xlint:all,-module -Werror -d proof/core @proof/core-sources.txt
          jar --create --file proof/m3-collections.jar -C proof/core .
          JAR="$GITHUB_WORKSPACE/proof/m3-collections.jar"
          (cd "$RUNNER_TEMP" && mvn -B -ntp org.apache.maven.plugins:maven-install-plugin:3.1.3:install-file -Dfile="$JAR" -DgroupId=com.m3 -DartifactId=m3-collections -Dversion=0.1.0-SNAPSHOT -Dpackaging=jar -DgeneratePom=true)
          mvn -B -ntp -f m3/collections/interop/pom.xml verify org.apache.maven.plugins:maven-dependency-plugin:3.8.1:copy-dependencies -DincludeScope=runtime -DoutputDirectory="$GITHUB_WORKSPACE/proof/deps" 2>&1 | tee proof/interop-maven.log
          curl --fail --location --retry 3 https://repo.maven.apache.org/maven2/org/junit/platform/junit-platform-console-standalone/1.12.2/junit-platform-console-standalone-1.12.2.jar -o proof/junit-platform-console-standalone-1.12.2.jar
          echo '329bd10288875a74d04c9ca6b7c9889c265ddf87b21a9ea42a7f7392f391472a  proof/junit-platform-console-standalone-1.12.2.jar' | sha256sum --check -
      - name: Run Java JNI differential suites and real allocation-failure negative controls
        shell: bash
        run: |
          set -euo pipefail
          python3 m3/collections/counts/verify.py --root "$GITHUB_WORKSPACE" --deps proof/deps --junit proof/junit-platform-console-standalone-1.12.2.jar 2>&1 | tee proof/verification.log
          git diff --exit-code -- m3/collections
          find proof m3/collections/counts/target/verification -type f ! -name SHA256SUMS -print0 | sort -z | xargs -0 sha256sum > proof/SHA256SUMS
      - name: Preserve actual reports and failures with the exact source identity
        if: always()
        uses: actions/upload-artifact@ea165f8d65b6e75b540449e92b4886f43607fa02
        with:
          name: m3-counted-qualified-${{ github.event.pull_request.head.sha || github.sha }}
