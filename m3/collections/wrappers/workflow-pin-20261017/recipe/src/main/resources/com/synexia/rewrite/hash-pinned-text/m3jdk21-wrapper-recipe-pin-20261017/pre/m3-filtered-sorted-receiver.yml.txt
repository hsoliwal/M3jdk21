name: M3 filtered and sorted wrapper contracts
on:
  push:
    branches: [aix/m3-filtered-sorted-receiver-20261016]
  pull_request:
    paths:
      - 'm3/collections/**'
      - '.github/workflows/m3-filtered-sorted-receiver.yml'
permissions:
  contents: read
concurrency:
  group: m3-filtered-sorted-${{ github.head_ref || github.ref_name }}
  cancel-in-progress: true
jobs:
  recipe-and-java-jni:
    runs-on: ubuntu-24.04-arm
    timeout-minutes: 30
    env:
      PYTHONOPTIMIZE: '0'
    steps:
      - uses: actions/checkout@11d5960a326750d5838078e36cf38b85af677262
        with:
          ref: ${{ github.event.pull_request.head.sha || github.sha }}
          persist-credentials: false
          sparse-checkout: m3/collections
      - uses: actions/setup-java@cf277c60eb25467037889841efdb72551f06f6c3
        with:
          distribution: temurin
          java-version: '21'
      - name: Identify exact source and canonical recipe mirror
        shell: bash
        run: |
          set -euo pipefail
          mkdir -p proof/deps proof/core
          { git rev-parse HEAD; uname -m; java -version 2>&1; mvn -version; gcc --version; } > proof/identity.txt
          test "$(git rev-parse HEAD:m3/collections/wrappers/recipe)" = 53fb42686982de4906dc73c2d8c7ca4d687dc4a1
          test "$(git hash-object m3/collections/wrappers/verify.py)" = 5c813e8aa766ae9bd90a8c3a7e54412a8b5281c8
          test "$(git rev-parse HEAD:m3/collections/wrappers/proof)" = c901d303b718fb435b24a8fe1d42f40016a89fcf
          cp m3/collections/wrappers/canonical-source.json proof/
      - name: Execute retained and new Maven OpenRewrite suites
        shell: bash
        run: |
          set -euo pipefail
          for recipe in counts counts/totals bimap wrappers; do
            label="${recipe//\//-}"
            mvn -B -ntp -f "m3/collections/$recipe/recipe/pom.xml" test 2>&1 | tee "proof/$label-recipe.log"
            cp -R "m3/collections/$recipe/recipe/target/surefire-reports" "proof/$label-recipe-reports"
          done
          python3 - <<'PY'
          from pathlib import Path
          import xml.etree.ElementTree as E
          for name, count in [('counts',8),('counts-totals',8),('bimap',9),('wrappers',9)]:
              reports=list(Path('proof/'+name+'-recipe-reports').glob('TEST-*.xml'))
              assert reports, 'missing '+name+' reports'
              actual={k:sum(int(E.parse(p).getroot().attrib.get(k,0)) for p in reports) for k in ('tests','failures','errors','skipped')}
              assert actual=={'tests':count,'failures':0,'errors':0,'skipped':0}, (name,actual)
              print(name,actual)
          PY
          mvn -B -ntp -f m3/collections/wrappers/recipe/pom.xml -Dm3.wrappers.target="$GITHUB_WORKSPACE" test 2>&1 | tee proof/wrappers-replay.log
          git diff --exit-code -- m3/collections
      - name: Compile current core and resolve existing optional library closure
        shell: bash
        run: |
          set -euo pipefail
          find m3/collections/src/main/java -name '*.java' | sort > proof/core-sources.txt
          javac --release 21 -Xlint:all,-module -Werror -d proof/core @proof/core-sources.txt
          jar --create --file proof/m3-collections.jar -C proof/core .
          JAR="$GITHUB_WORKSPACE/proof/m3-collections.jar"
          (cd "$RUNNER_TEMP" && mvn -B -ntp org.apache.maven.plugins:maven-install-plugin:3.1.3:install-file -Dfile="$JAR" -DgroupId=com.m3 -DartifactId=m3-collections -Dversion=0.1.0-SNAPSHOT -Dpackaging=jar -DgeneratePom=true)
          mvn -B -ntp -f m3/collections/interop/pom.xml verify org.apache.maven.plugins:maven-dependency-plugin:3.8.1:copy-dependencies -DincludeScope=runtime -DoutputDirectory="$GITHUB_WORKSPACE/proof/deps" 2>&1 | tee proof/interop-maven.log
          curl --fail --location --retry 3 https://repo.maven.apache.org/maven2/org/junit/platform/junit-platform-console-standalone/1.12.2/junit-platform-console-standalone-1.12.2.jar -o proof/junit-platform-console-standalone-1.12.2.jar
          echo '329bd10288875a74d04c9ca6b7c9889c265ddf87b21a9ea42a7f7392f391472a  proof/junit-platform-console-standalone-1.12.2.jar' | sha256sum --check -
      - name: Verify retained Java JNI and new wrapper runtime tests
        shell: bash
        run: |
          set -euo pipefail
          python3 m3/collections/wrappers/verify.py --root "$GITHUB_WORKSPACE" --deps proof/deps --junit proof/junit-platform-console-standalone-1.12.2.jar --phase java 2>&1 | tee proof/wrappers-java.log
      - name: Verify bound native navigation in tiered interpreter and C1 modes
        shell: bash
        run: |
          set -euo pipefail
          python3 m3/collections/wrappers/verify.py --root "$GITHUB_WORKSPACE" --deps proof/deps --junit proof/junit-platform-console-standalone-1.12.2.jar --phase native 2>&1 | tee proof/wrappers-native.log
      - name: Finish negative controls API package and source qualification
        shell: bash
        run: |
          set -euo pipefail
          python3 m3/collections/wrappers/verify.py --root "$GITHUB_WORKSPACE" --deps proof/deps --junit proof/junit-platform-console-standalone-1.12.2.jar --phase finish 2>&1 | tee proof/wrappers-finish.log
          git diff --exit-code -- m3/collections
          find proof m3/collections/wrappers m3/collections/counts/target/verification m3/collections/counts/totals/target/evidence m3/collections/bimap/target/verification -type f ! -name SHA256SUMS -print0 | sort -z | xargs -0 sha256sum > proof/SHA256SUMS
      - name: Preserve exact source and evidence including failures
        if: always()
        uses: actions/upload-artifact@ea165f8d65b6e75b540449e92b4886f43607fa02
        with:
          name: m3-filtered-sorted-${{ github.event.pull_request.head.sha || github.sha }}
          retention-days: 30
          path: |
            proof/
            m3/collections/wrappers/
            m3/collections/counts/target/verification/
            m3/collections/counts/totals/target/evidence/
            m3/collections/bimap/target/verification/
            m3/collections/src/main/java/com/m3/collections/M3CollectionViews.java
            m3/collections/src/main/java/com/m3/collections/M3FilteredMaps.java
            m3/collections/src/test/native/collections/run_search_jni.c
