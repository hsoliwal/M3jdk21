package com.m3.collections;

import java.util.Arrays;
import java.util.Random;

/** Direct parity of the original Java page owner with the already-merged JNI page export. */
public final class M3NativePageDirectContract {
    private static long checks;
    private static long digest = 0xcbf29ce484222325L;

    private M3NativePageDirectContract() { }

    private static void check(boolean value, String name) {
        checks++;
        if (!value) throw new AssertionError(name + " at check=" + checks);
    }

    private static void mix(int value) {
        digest = (digest ^ (value & 0xffff_ffffL)) * 0x100000001b3L;
    }

    private static void equal(int[] expected, int[] actual, String name) {
        check(Arrays.equals(expected, actual), name);
        for (int value : actual) mix(value);
    }

    private static void exercise(int[] rawOffsets, int[] rawStates, boolean shareInputs,
            int first, int length, int required, int forbidden, int skip, int max,
            int outputIndex, int targetOffset) {
        int[] offsetsA = rawOffsets.clone();
        int[] statesA = shareInputs ? offsetsA : rawStates.clone();
        int[] offsetsB = rawOffsets.clone();
        int[] statesB = shareInputs ? offsetsB : rawStates.clone();
        int[] outA = switch (outputIndex) {
            case 0 -> new int[Math.max(3, length + 5)];
            case 1 -> offsetsA;
            case 2 -> statesA;
            default -> throw new IllegalArgumentException("test configuration");
        };
        int[] outB = switch (outputIndex) {
            case 0 -> new int[Math.max(3, length + 5)];
            case 1 -> offsetsB;
            case 2 -> statesB;
            default -> throw new IllegalArgumentException("test configuration");
        };
        if (outputIndex == 0) {
            Arrays.fill(outA, -51);
            Arrays.fill(outB, -51);
        }
        int javaCount = -1;
        int nativeCount = -1;
        Throwable javaFailure = null;
        Throwable nativeFailure = null;
        try {
            javaCount = M3MaskViewportKernel.projectPage(offsetsA, statesA,
                    first, length, required, forbidden, skip, max, outA, targetOffset);
        } catch (RuntimeException failure) {
            javaFailure = failure;
        }
        try {
            nativeCount = M3MaskViewportNative.projectPage(offsetsB, statesB,
                    first, length, required, forbidden, skip, max, outB, targetOffset);
        } catch (RuntimeException failure) {
            nativeFailure = failure;
        }
        check(javaFailure == null ? nativeFailure == null
                : nativeFailure != null && nativeFailure.getClass() == javaFailure.getClass(),
                "exception parity");
        if (javaFailure == null) {
            check(nativeCount == javaCount, "return count");
            mix(nativeCount);
        } else {
            mix(javaFailure.getClass().getName().hashCode());
        }
        equal(outA, outB, "target cells + fences");
        equal(offsetsA, offsetsB, "original offsets and aliases");
        if (!shareInputs) equal(statesA, statesB, "original states and aliases");
    }

    private static void randomized() {
        Random rng = new Random(0x3a7baefa292719L);
        for (int trial = 0; trial < 25_000; trial++) {
            int size = switch (trial % 16) {
                case 0 -> 0;
                case 1 -> 1;
                case 2 -> 2;
                case 3 -> 127;
                case 4 -> 128;
                case 5 -> 129;
                case 6 -> 255;
                case 7 -> 257;
                default -> rng.nextInt(200);
            };
            int[] offsets = new int[size];
            int[] states = new int[size];
            for (int i = 0; i < size; i++) {
                offsets[i] = rng.nextInt(100000);
                states[i] = switch ((trial + i) % 9) {
                    case 0 -> 0;
                    case 1 -> 1;
                    case 2 -> Integer.MIN_VALUE;
                    case 3 -> Integer.MAX_VALUE;
                    case 4 -> -1;
                    case 5 -> 2;
                    default -> rng.nextInt();
                };
            }
            int first = rng.nextInt(size + 1);
            int len = rng.nextInt(size - first + 1);
            int required = switch (trial % 8) {
                case 0 -> 0;
                case 1 -> 1;
                case 2 -> Integer.MIN_VALUE;
                case 3 -> 0xff;
                default -> rng.nextInt();
            };
            int forbidden = switch (trial % 11) {
                case 0 -> required;
                case 1 -> 0;
                case 2 -> 1;
                default -> rng.nextInt();
            };
            int skip = switch (trial % 9) {
                case 0 -> Integer.MAX_VALUE;
                case 1 -> 0;
                case 2 -> 1;
                case 3 -> 127;
                case 4 -> 128;
                default -> rng.nextInt(40);
            };
            int max = switch (trial % 7) {
                case 0 -> Integer.MAX_VALUE;
                case 1 -> 0;
                case 2 -> 1;
                case 3 -> 128;
                default -> rng.nextInt(32);
            };
            boolean shared = trial % 17 == 0;
            int output = trial % 3;
            int capacity = output == 0 ? Math.max(3, len + 5) : size;
            int pos = capacity == 0 ? 0 : rng.nextInt(capacity + 1);
            exercise(offsets, states, shared, first, len, required, forbidden, skip, max, output, pos);
        }
    }

    private static void negativeInputs() {
        int[] ids = {1, 2, 3, -4, 5};
        int[] masks = {0, 1, 0, 1, 0};
        exercise(ids, masks, false, 0, 5, 1, 0, Integer.MAX_VALUE, 0, 0, 0);
        exercise(ids, masks, false, 0, 5, 1, 1, Integer.MAX_VALUE, 0, 0, 0);
        exercise(ids, masks, false, 0, 5, 0, 0, Integer.MAX_VALUE, 0, 1, 0);
        exercise(new int[]{1, 2, 3}, new int[]{0, 1}, false, 0, 1, 0, 0, 0, 1, 0, 0);
        exercise(new int[]{1, 2, 3}, new int[]{0, 1, 0}, false, Integer.MAX_VALUE, 1,
                0, 0, 0, 1, 0, 0);
        exercise(new int[]{1, 2, 3}, new int[]{0, 1, 0}, false, 0, 3,
                0, 0, -1, 1, 0, 0);
        exercise(new int[]{1, 2, 3}, new int[]{0, 1, 0}, false, 0, 3,
                0, 0, 0, -1, 0, 0);
    }

    public static void main(String[] args) {
        if (args.length != 1) throw new IllegalArgumentException("native library path");
        System.load(args[0]);
        randomized();
        negativeInputs();
        System.out.println("M3_NATIVE_PAGE_DIRECT_PASS checks=" + checks
                + " digest=" + Long.toUnsignedString(digest, 16));
    }
}
