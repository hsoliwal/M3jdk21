package com.m3.collections.interop;

import static org.junit.jupiter.api.Assertions.*;
import com.google.common.collect.ArrayListMultimap;
import com.google.common.collect.HashMultimap;
import com.m3.collections.M3LinkedHashMap;
import com.m3.collections.M3LinkedHashSet;
import com.m3.collections.M3LinkedList;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.Comparator;
import java.util.LinkedHashMap;
import java.util.LinkedHashSet;
import java.util.List;
import java.util.Map;
import java.util.Objects;
import java.util.Set;
import java.util.function.Function;
import java.util.function.Supplier;
import java.util.stream.Collector;
import java.util.stream.Collectors;
import java.util.stream.IntStream;
import org.junit.jupiter.api.Test;
import org.springframework.util.AutoPopulatingList;
import org.springframework.util.MultiValueMap;

final class M3LibraryClosureTest {
    private static List<Integer> data() { return IntStream.range(0, 4096).map(i -> (i * 71) % 103).boxed().toList(); }
    private static Object outcome(Supplier<?> operation) {
        try { return operation.get(); } catch (RuntimeException failure) { return failure.getClass(); }
    }

    @Test void listSequentialParallelAndNullOwnership() {
        var expected = new ArrayList<>(data()); expected.add(null);
        var sequential = expected.stream().collect(M3LibraryCollectors.toList());
        var parallel = expected.parallelStream().collect(M3LibraryCollectors.toList());
        assertEquals(expected, sequential); assertEquals(expected, parallel);
        assertInstanceOf(M3LinkedList.class, sequential); assertInstanceOf(M3LinkedList.class, parallel);
        parallel.clear(); assertEquals(expected, sequential);
    }

    @Test void setSequentialParallelOrderAndNull() {
        var input = new ArrayList<>(data()); input.add(null); input.add(null);
        var expected = new LinkedHashSet<>(input);
        var actual = input.parallelStream().collect(M3LibraryCollectors.toSet());
        assertInstanceOf(M3LinkedHashSet.class, actual); assertEquals(expected, actual);
        assertEquals(new ArrayList<>(expected), new ArrayList<>(actual));
    }

    @Test void uniqueMapOrderedParallelResult() {
        var input = IntStream.range(0, 4096).boxed().toList();
        var actual = input.parallelStream().collect(M3LibraryCollectors.toMap(Function.identity(), i -> i + 1));
        assertInstanceOf(M3LinkedHashMap.class, actual);
        assertEquals(input, new ArrayList<>(actual.keySet()));
        assertEquals(input.stream().collect(Collectors.toMap(Function.identity(), i -> i + 1)), actual);
    }

    @Test void duplicatesRejectedAcrossParallelPartitions() {
        var input = IntStream.range(0, 4096).map(i -> i % 127).boxed().toList();
        assertThrows(IllegalStateException.class, () -> input.stream().collect(M3LibraryCollectors.toMap(Function.identity(), i -> 1)));
        assertThrows(IllegalStateException.class, () -> input.parallelStream().collect(M3LibraryCollectors.toMap(Function.identity(), i -> 1)));
    }

    @Test void mergingMapCountsAndDeletionMatchJdk() {
        var expected = data().stream().collect(Collectors.toMap(Function.identity(), i -> 1, Integer::sum, LinkedHashMap::new));
        var actual = data().parallelStream().collect(M3LibraryCollectors.toMap(Function.identity(), i -> 1, Integer::sum));
        assertEquals(expected, actual); assertEquals(new ArrayList<>(expected.keySet()), new ArrayList<>(actual.keySet()));
        var deleting = List.of(1, 1, 2).stream().collect(M3LibraryCollectors.toMap(Function.identity(), i -> i, (a, b) -> null));
        assertEquals(Map.of(2, 2), deleting);
    }

    @Test void mapNullPoliciesAndCallbackExceptions() {
        var map = List.of(1).stream().collect(M3LibraryCollectors.toMap(i -> (String) null, i -> i));
        assertEquals(1, map.get(null).intValue());
        assertThrows(NullPointerException.class, () -> List.of(1).stream().collect(M3LibraryCollectors.toMap(i -> i, i -> null)));
        assertThrows(IllegalStateException.class, () -> List.of(1, 2).stream().collect(M3LibraryCollectors.toMap(i -> (String) null, i -> i)));
        RuntimeException marker = new IllegalArgumentException("mapper marker");
        assertSame(marker, assertThrows(RuntimeException.class,
                () -> List.of(1).stream().collect(M3LibraryCollectors.toMap(i -> { throw marker; }, i -> i))));
    }

    @Test void guavaListParallelValuesAndFirstKeyOrder() {
        var expected = ArrayListMultimap.<Integer, Integer>create(); data().forEach(i -> expected.put(i % 7, i));
        var actual = data().parallelStream().collect(M3LibraryCollectors.toGuavaListMultimap(i -> i % 7, Function.identity()));
        assertEquals(expected, actual);
        assertEquals(data().stream().map(i -> i % 7).distinct().toList(), new ArrayList<>(actual.keySet()));
        var nulls = Arrays.asList(1, null, 2, null).parallelStream()
                .collect(M3LibraryCollectors.toGuavaListMultimap(i -> "a", Function.identity()));
        assertEquals(Arrays.asList(1, null, 2, null), nulls.get("a"));
    }

    @Test void guavaSetParallelDuplicates() {
        var expected = HashMultimap.<Integer, Integer>create(); data().forEach(i -> expected.put(i % 7, i));
        var actual = data().parallelStream().collect(M3LibraryCollectors.toGuavaSetMultimap(i -> i % 7, Function.identity()));
        assertEquals(expected, actual); assertEquals(expected.size(), actual.size());
    }

    @Test void guavaSortedSetParallelComparator() {
        var actual = data().parallelStream().collect(M3LibraryCollectors.toGuavaSortedSetMultimap(i -> i % 7, Function.identity(), Comparator.reverseOrder()));
        for (Integer key : actual.keySet()) assertEquals(
                data().stream().filter(i -> i % 7 == key).distinct().sorted(Comparator.reverseOrder()).toList(),
                new ArrayList<>(actual.get(key)));
        assertEquals(103, actual.size());
    }

    @Test void eclipseListAndSetResultsStayMutableAndIndependent() {
        var list = data().parallelStream().collect(M3LibraryCollectors.toEclipseList());
        var set = data().parallelStream().collect(M3LibraryCollectors.toEclipseSet());
        assertEquals(data(), list); assertEquals(new LinkedHashSet<>(data()), set);
        assertEquals(new ArrayList<>(new LinkedHashSet<>(data())), new ArrayList<>(set));
        list.clear(); assertEquals(103, set.size());
    }

    @Test void eclipseMapUsesUniqueM3Accumulation() {
        var input = IntStream.range(0, 4096).boxed().toList();
        var actual = input.parallelStream().collect(M3LibraryCollectors.toEclipseMap(Function.identity(), i -> i + 1));
        assertEquals(input, new ArrayList<>(actual.keySet())); assertEquals(4096, actual.size());
        assertThrows(IllegalStateException.class, () -> List.of(1, 1).parallelStream()
                .collect(M3LibraryCollectors.toEclipseMap(Function.identity(), Function.identity())));
    }

    @Test void springParallelCollectorUsesM3ListsAndCopies() {
        var actual = data().parallelStream().collect(M3LibraryCollectors.toSpringMultiValueMap(i -> i % 7, Function.identity()));
        assertInstanceOf(M3SpringMultiValueMap.class, actual);
        assertEquals(7, actual.size()); assertEquals(4096, actual.values().stream().mapToInt(List::size).sum());
        for (Integer key : actual.keySet()) {
            assertInstanceOf(M3LinkedList.class, actual.get(key));
            assertEquals(data().stream().filter(i -> i % 7 == key).toList(), actual.get(key));
        }
        assertInstanceOf(M3LinkedHashMap.class, actual.toSingleValueMap());
    }

    @Test void springCreatedListsAndFactoryWiring() {
        var actual = M3LibraryCollections.<String, Integer>springMultiValueMap();
        assertInstanceOf(M3SpringMultiValueMap.class, actual);
        actual.add("a", 1); actual.set("b", 2); actual.addAll("c", List.of(3, 4)); actual.addAll("empty", List.of());
        for (List<Integer> values : actual.values()) assertInstanceOf(M3LinkedList.class, values);
        assertTrue(actual.containsKey("empty")); assertEquals(4, actual.size());
        assertEquals(Map.of("a", 1, "b", 2, "c", 3), actual.toSingleValueMap());
    }

    @Test void springBorrowedListsAreNotSilentlyCopied() {
        var actual = new M3SpringMultiValueMap<String, Integer>();
        var caller = new ArrayList<>(List.of(1)); actual.put("a", caller);
        assertSame(caller, actual.get("a")); actual.add("a", 2); assertEquals(List.of(1, 2), caller);
        actual.put("fixed", List.of(9));
        assertThrows(UnsupportedOperationException.class, () -> actual.add("fixed", 10));
        actual.set("a", 7); assertEquals(List.of(1, 2), caller); assertEquals(List.of(7), actual.get("a"));
        assertInstanceOf(M3LinkedList.class, actual.get("a"));
    }

    @Test void springNullEmptyAbsentAndCopyViewDistinctions() {
        var actual = new M3SpringMultiValueMap<String, Integer>();
        actual.put("null-list", null); actual.addAll("empty", List.of()); actual.add("null-value", null);
        actual.addIfAbsent("null-list", 99); assertNull(actual.get("null-list"));
        assertFalse(actual.toSingleValueMap().containsKey("null-list"));
        assertFalse(actual.toSingleValueMap().containsKey("empty"));
        assertTrue(actual.toSingleValueMap().containsKey("null-value"));
        actual.add("a", 1); var copy = actual.toSingleValueMap(); var live = actual.asSingleValueMap();
        actual.set("a", 2); assertEquals(1, copy.get("a").intValue()); assertEquals(2, live.get("a").intValue());
        actual.add("null-list", 3); assertInstanceOf(M3LinkedList.class, actual.get("null-list"));
    }

    @Test void springInvalidBulkInputDoesNotInstallAnAbsentKey() {
        var actual = new M3SpringMultiValueMap<String, Integer>();
        assertThrows(NullPointerException.class, () -> actual.addAll("absent", (List<Integer>) null));
        assertFalse(actual.containsKey("absent"));
        actual.add("present", 1);
        assertThrows(NullPointerException.class, () -> actual.addAll("present", (List<Integer>) null));
        assertEquals(List.of(1), actual.get("present"));
    }

    @Test void springSingleValueBridgeMatchesDonorExceptionsAndMutations() {
        var expected = MultiValueMap.<String, Integer>fromSingleValue(new LinkedHashMap<>());
        var actual = M3LibraryCollections.<String, Integer>springSingleValueMap();
        expected.add("a", 1); actual.add("a", 1);
        assertEquals(outcome(() -> { expected.add("a", 2); return null; }), outcome(() -> { actual.add("a", 2); return null; }));
        assertEquals(outcome(() -> expected.put("b", List.of(1, 2))), outcome(() -> actual.put("b", List.of(1, 2))));
        expected.set("a", null); actual.set("a", null); assertEquals(expected, actual);
        assertEquals(outcome(() -> expected.put("c", List.of())), outcome(() -> actual.put("c", List.of())));
        assertEquals(expected, actual);
    }

    @Test void springAutoPopulatingListPreservesLazyFactoryCalls() {
        List<Integer> aCalls = new ArrayList<>(), eCalls = new ArrayList<>();
        var expected = new AutoPopulatingList<Integer>(new ArrayList<>(), (AutoPopulatingList.ElementFactory<Integer>) i -> { eCalls.add(i); return i * 10; });
        var actual = M3LibraryCollections.<Integer>springAutoPopulatingList(i -> { aCalls.add(i); return i * 10; });
        for (int index : new int[] {5, 0, 5, 2, 1}) assertEquals(expected.get(index), actual.get(index));
        assertEquals(eCalls, aCalls); assertEquals(expected, actual);
        expected.set(2, null); actual.set(2, null); assertEquals(expected.get(2), actual.get(2));
        assertEquals(eCalls, aCalls);
        assertEquals(outcome(() -> expected.get(-1)), outcome(() -> actual.get(-1)));
    }

    @Test void collectorCharacteristicsDoNotClaimConcurrentMutation() {
        List<Collector<Integer, ?, ?>> collectors = List.of(
                M3LibraryCollectors.toList(), M3LibraryCollectors.toSet(),
                M3LibraryCollectors.toMap(Function.identity(), Function.identity()),
                M3LibraryCollectors.toMap(Function.identity(), i -> 1, Integer::sum),
                M3LibraryCollectors.toGuavaListMultimap(i -> i % 7, Function.identity()),
                M3LibraryCollectors.toGuavaSetMultimap(i -> i % 7, Function.identity()),
                M3LibraryCollectors.toGuavaSortedSetMultimap(i -> i % 7, Function.identity(), Comparator.naturalOrder()),
                M3LibraryCollectors.toEclipseList(), M3LibraryCollectors.toEclipseSet(),
                M3LibraryCollectors.toEclipseMap(Function.identity(), Function.identity()),
                M3LibraryCollectors.toSpringMultiValueMap(i -> i % 7, Function.identity()));
        for (var collector : collectors) {
            assertFalse(collector.characteristics().contains(Collector.Characteristics.CONCURRENT));
            assertFalse(collector.characteristics().contains(Collector.Characteristics.UNORDERED));
        }
    }

    @Test void mapperAndComparatorNullsAreRefusedBeforeCollection() {
        assertThrows(NullPointerException.class, () -> M3LibraryCollectors.toMap(null, Function.identity()));
        assertThrows(NullPointerException.class, () -> M3LibraryCollectors.toMap(Function.identity(), null));
        assertThrows(NullPointerException.class, () -> M3LibraryCollectors.toMap(Function.identity(), Function.identity(), null));
        assertThrows(NullPointerException.class, () -> M3LibraryCollectors.toGuavaSortedSetMultimap(Function.identity(), Function.identity(), null));
        assertThrows(NullPointerException.class, () -> M3LibraryCollectors.toSpringMultiValueMap(null, Function.identity()));
    }
}
