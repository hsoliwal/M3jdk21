cmake_minimum_required(VERSION 3.16)
project(m3_keyed LANGUAGES CXX)
find_package(JNI REQUIRED)
add_library(m3_keyed SHARED m3_keyed.cpp)
target_compile_features(m3_keyed PRIVATE cxx_std_17)
target_include_directories(m3_keyed PRIVATE ${JNI_INCLUDE_DIRS})
option(M3_KEYED_UBSAN "Enable undefined-behavior checks" OFF)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(m3_keyed PRIVATE -Wall -Wextra -Werror)
  if(M3_KEYED_UBSAN)
    target_compile_options(m3_keyed PRIVATE -fsanitize=undefined -fno-sanitize-recover=all)
    target_link_options(m3_keyed PRIVATE -fsanitize=undefined -fno-sanitize-recover=all)
  endif()
endif()
