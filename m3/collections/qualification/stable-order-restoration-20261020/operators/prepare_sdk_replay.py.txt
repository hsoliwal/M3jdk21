from pathlib import Path

def sha(data):
    return hashlib.sha256(data).hexdigest()


def blob(data):
    return hashlib.sha1(b'blob ' + str(len(data)).encode() + b'\0' + data).hexdigest()


def write(path, text):
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(text)


copied = {}
for rel in ('pom.xml', 'DEPENDENCIES.md', 'src/test/java/com/synexia/rewrite/M3TaskRecipeTest.java'):
    source = OLD / rel
    destination = HOST / rel
    destination.parent.mkdir(parents=True, exist_ok=True)
    shutil.copyfile(source, destination)
    copied[rel] = {'sha256': sha(source.read_bytes()), 'git_blob': blob(source.read_bytes()),
                   'source': f'https://github.com/hsoliwal/com.synexia/blob/{DONOR}/'
                             'synexia-openrewrite-recipes/recipe-crates/m3-wrapper-convergence-20261018/recipe-host/' + rel}

targets = {}
for selected in PASSES:
    source = OLD / 'passes' / selected
    shutil.copytree(source, HOST / 'passes' / selected, dirs_exist_ok=True)
    for path in source.rglob('*'):
        if path.is_file():
            rel = path.relative_to(OLD).as_posix()
            copied[rel] = {'sha256': sha(path.read_bytes()), 'git_blob': blob(path.read_bytes()),
                           'source': f'https://github.com/hsoliwal/com.synexia/blob/{DONOR}/'
                                     'synexia-openrewrite-recipes/recipe-crates/m3-wrapper-convergence-20261018/recipe-host/' + rel}
    crate = HOST / 'passes' / selected / 'resources/com/synexia/rewrite/hash-pinned-java' / ('m3-wrapper-' + selected)
    lines = [line for line in (crate / 'manifest.tsv').read_text().splitlines()
             if line and not line.startswith('#')]
    assert len(lines) == 1
    target, before, after, template = lines[0].split('\t')
    assert sha((crate / template).read_bytes()) == after
    target_rel = target.removeprefix('m3/collections/')
    current = subprocess.run(['git', '-C', str(REPOSITORY), 'show', TREE + ':' + target_rel],
                             capture_output=True, check=False)
    if before == 'ABSENT':
        assert current.returncode != 0 and b'does not exist' in current.stderr, current.stderr
    else:
        assert current.returncode == 0 and sha(current.stdout) == before
        assert (crate / 'pre' / template).read_bytes() == current.stdout
    targets[target] = {'pass': selected, 'before_sha256': before, 'after_sha256': after}

engine_lines = ['# canonical_base=' + BASE,

             'M3ScopeInference.java', 'M3EditScope.java'):

                 else f'https://github.com/hsoliwal/com.synexia/blob/{BASE}/' + rel)
    engine_lines.append('\t'.join((rel, blob(data), sha(data), 'Apache-2.0', reference)))
    engines[rel] = {'sha256': sha(data), 'git_blob': blob(data), 'reference': reference}
write(HOST / 'ENGINE_SOURCES.tsv', '\n'.join(engine_lines) + '\n')

    'source_custody_inventory': 'evidence/SOURCE_INVENTORY.json',

}, indent=2, sort_keys=True) + '\n')
