package com.synexia.rewrite;

import static org.junit.jupiter.api.Assertions.assertEquals;
import static org.junit.jupiter.api.Assertions.assertFalse;
import static org.junit.jupiter.api.Assertions.assertNotNull;
import static org.junit.jupiter.api.Assertions.assertThrows;
import static org.junit.jupiter.api.Assertions.assertTrue;

import java.io.IOException;
import java.nio.charset.StandardCharsets;
import java.nio.file.Path;
import java.util.List;
import java.util.Map;
import java.util.TreeMap;
import org.junit.jupiter.api.Test;
import org.openrewrite.InMemoryExecutionContext;
import org.openrewrite.Parser;
import org.openrewrite.SourceFile;
import org.openrewrite.internal.InMemoryLargeSourceSet;
import org.openrewrite.java.JavaParser;

/** Original four-source native M3 JDK-generic collection recipe contract. */
final class M3NativeGenericDequePriority20261018RecipeTest {
    private static final String NAME = "m3-native-generic-deque-priority-20261018";
    private static final String ROOT = "/com/synexia/rewrite/hash-pinned-java/" + NAME + "/";
    private static final List<String> PATHS = List.of(
            "m3/collections/src/main/java/com/m3/collections/M3ArrayDeque.java",
            "m3/collections/src/main/java/com/m3/collections/M3PriorityQueue.java",
            "m3/collections/src/test/java/com/m3/collections/M3GenericReferenceOwnersProof.java",
            "m3/collections/src/test/java/com/m3/collections/M3GenericReferenceOwnersTest.java");

    private static String template(String name) {
        try (var stream = M3NativeGenericDequePriority20261018RecipeTest.class
                .getResourceAsStream(ROOT + name)) {
            assertNotNull(stream, "missing canonical template: " + name);
            return new String(stream.readAllBytes(), StandardCharsets.UTF_8);
        } catch (IOException failure) {
            throw new IllegalStateException(failure);
        }
    }
    private static InMemoryExecutionContext context() {
        return new InMemoryExecutionContext(failure -> { throw new AssertionError(failure); });
    }
    private static List<SourceFile> parse(String path, String source) {
        return JavaParser.fromJavaVersion().build().parseInputs(
                List.of(Parser.Input.fromString(Path.of(path), source)), null, context()).toList();
    }

    @Test void exactFourOwnersFromEmptyReachFixedPoint() {
        var recipe = new M3HashPinnedJavaSnapshotRecipe(NAME);
        assertEquals(PATHS, recipe.targetPaths());
        var pass = recipe.run(new InMemoryLargeSourceSet(List.of()), context(), 1);
        assertEquals(4, pass.getChangeset().getAllResults().size());
        Map<String, SourceFile> outputs = new TreeMap<>();
        for (var change : pass.getChangeset().getAllResults()) {
            SourceFile after = change.getAfter();
            assertNotNull(after);
            outputs.put(after.getSourcePath().toString().replace('\\', '/'), after);
        }
        for (String path : PATHS) {
            SourceFile output = outputs.get(path);
            assertNotNull(output, path);
            assertEquals(template(Path.of(path).getFileName() + ".txt"), output.printAll(), path);
        }
        String deque = outputs.get(PATHS.get(0)).printAll();
        String queue = outputs.get(PATHS.get(1)).printAll();
        assertTrue(deque.contains("implements Deque<E>"));
        assertTrue(queue.contains("implements Cloneable, Serializable"));
        assertTrue(queue.contains("long[] identities"));
        var again = recipe.run(
                new InMemoryLargeSourceSet(List.copyOf(outputs.values())), context(), 1);
        assertTrue(again.getChangeset().getAllResults().isEmpty());
    }

    @Test void foreignOccupiedSourceMustRefuse() {
        var recipe = new M3HashPinnedJavaSnapshotRecipe(NAME);
        String original = template("M3ArrayDeque.java.txt");
        String changed = original.replace(
                "public final class M3ArrayDeque", "public class M3ArrayDeque");
        assertFalse(changed.equals(original));
        assertThrows(RuntimeException.class, () -> recipe.run(
                new InMemoryLargeSourceSet(parse(PATHS.get(0), changed)), context(), 1));
    }
}
