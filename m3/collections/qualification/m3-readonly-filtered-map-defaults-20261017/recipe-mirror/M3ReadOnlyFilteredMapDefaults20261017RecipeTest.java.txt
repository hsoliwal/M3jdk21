package com.synexia.rewrite;

import static org.junit.jupiter.api.Assertions.assertEquals;
import static org.junit.jupiter.api.Assertions.assertNotNull;
import static org.junit.jupiter.api.Assertions.assertThrows;
import static org.junit.jupiter.api.Assertions.assertTrue;

import java.io.IOException;
import java.nio.charset.StandardCharsets;
import java.nio.file.Path;
import java.util.List;
import java.util.Map;
import java.util.TreeMap;
import org.junit.jupiter.api.Test;
import org.openrewrite.InMemoryExecutionContext;
import org.openrewrite.Parser;
import org.openrewrite.SourceFile;
import org.openrewrite.internal.InMemoryLargeSourceSet;
import org.openrewrite.java.JavaParser;

/** Real OpenRewrite SDK fixed-point/refusal tests for exactly three admitted Java files. */
final class M3ReadOnlyFilteredMapDefaults20261017RecipeTest {
    private static final String CRATE = "m3-readonly-filtered-map-defaults-20261017";
    private static final String ROOT = "/com/synexia/rewrite/hash-pinned-java/" + CRATE + "/";
    private static final String OWNER =
            "m3/collections/src/main/java/com/m3/collections/M3FilteredViews.java";
    private static final String ORACLE =
            "m3/collections/src/test/java/com/m3/collections/M3ReadOnlyMapDefaultsContract.java";
    private static final String JUNIT =
            "m3/collections/src/test/java/com/m3/collections/M3ReadOnlyMapDefaultsTest.java";

    private static String read(String name) {
        try (var input = M3ReadOnlyFilteredMapDefaults20261017RecipeTest.class
                .getResourceAsStream(ROOT + name)) {
            assertNotNull(input, "missing recipe source " + name);
            return new String(input.readAllBytes(), StandardCharsets.UTF_8);
        } catch (IOException e) {
            throw new IllegalStateException(e);
        }
    }

    private static InMemoryExecutionContext context() {
        return new InMemoryExecutionContext(failure -> { throw new AssertionError(failure); });
    }

    private static List<SourceFile> parse(String path, String value) {
        var input = Parser.Input.fromString(Path.of(path), value);
        return JavaParser.fromJavaVersion().build().parseInputs(
                List.of(input), null, context()).toList();
    }

    @Test void generatedJavaSourcesMatchRecipeAndSecondPassIsFixedPoint() {
        var recipe = new M3HashPinnedJavaSnapshotRecipe(CRATE);
        assertEquals(List.of(OWNER, ORACLE, JUNIT), recipe.targetPaths());
        var initial = new InMemoryLargeSourceSet(
                parse(OWNER, read("pre/M3FilteredViews.java.txt")));
        var first = recipe.run(initial, context(), 1);
        assertEquals(3, first.getChangeset().getAllResults().size());
        Map<String, SourceFile> outputs = new TreeMap<>();
        for (var change : first.getChangeset().getAllResults()) {
            SourceFile value = change.getAfter();
            assertNotNull(value);
            outputs.put(value.getSourcePath().toString().replace('\\', '/'), value);
        }
        assertEquals(read("M3FilteredViews.java.txt"), outputs.get(OWNER).printAll());
        assertEquals(read("M3ReadOnlyMapDefaultsContract.java.txt"), outputs.get(ORACLE).printAll());
        assertEquals(read("M3ReadOnlyMapDefaultsTest.java.txt"), outputs.get(JUNIT).printAll());
        assertTrue(outputs.get(OWNER).printAll().contains(
                "Collections.unmodifiableNavigableMap(new NavigableMapView<>(source, predicate))"));
        var second = recipe.run(
                new InMemoryLargeSourceSet(List.copyOf(outputs.values())), context(), 1);
        assertTrue(second.getChangeset().getAllResults().isEmpty(), "second pass is fixed point");
    }

    @Test void driftAndMissingSourceRefuseBeforeAnyReplacement() {
        var recipe = new M3HashPinnedJavaSnapshotRecipe(CRATE);
        String original = read("pre/M3FilteredViews.java.txt");
        String mutated = original.replace("return new SortedMapView<>(source, predicate);",
                "return null; // candidate source drift");
        assertTrue(!mutated.equals(original));
        assertThrows(RuntimeException.class, () -> recipe.run(
                new InMemoryLargeSourceSet(parse(OWNER, mutated)), context(), 1));
        assertThrows(RuntimeException.class, () -> recipe.run(
                new InMemoryLargeSourceSet(List.of()), context(), 1));
    }

    @Test void occupiedTestTargetWithForeignBytesRefuses() {
        var recipe = new M3HashPinnedJavaSnapshotRecipe(CRATE);
        List<SourceFile> files = new java.util.ArrayList<>(
                parse(OWNER, read("pre/M3FilteredViews.java.txt")));
        files.addAll(parse(ORACLE, read("M3ReadOnlyMapDefaultsContract.java.txt")
                .replace("PASS M3ReadOnlyMapDefaultsContract", "unexpected-mutant")));
        assertThrows(RuntimeException.class, () -> recipe.run(
                new InMemoryLargeSourceSet(files), context(), 1));
    }
}
