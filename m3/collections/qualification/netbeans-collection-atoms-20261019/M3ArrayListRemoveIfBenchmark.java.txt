package com.m3.collections;

import com.sun.management.ThreadMXBean;
import java.lang.management.GarbageCollectorMXBean;
import java.lang.management.ManagementFactory;
import java.util.Arrays;
import java.util.List;
import java.util.Locale;
import java.util.Random;
import java.util.function.Predicate;

/** Measures removeIf only; input list construction happens outside measured spans. */
public final class M3ArrayListRemoveIfBenchmark {
    private static volatile long checksum;
    private static final ThreadMXBean THREADS =
            (ThreadMXBean) ManagementFactory.getThreadMXBean();
    private static final long THREAD_ID = Thread.currentThread().threadId();
    private static final List<GarbageCollectorMXBean> GCS = ManagementFactory.getGarbageCollectorMXBeans();

    private M3ArrayListRemoveIfBenchmark() { }

    private static final class MaskPredicate implements Predicate<Object> {
        private final boolean[] removed;
        private int cursor;
        private MaskPredicate(boolean[] removed) { this.removed = removed; }
        @Override public boolean test(Object ignored) { return removed[cursor++]; }
    }

    private static long gcCount() {
        long result = 0;
        for (GarbageCollectorMXBean collector : GCS) result += Math.max(0, collector.getCollectionCount());
        return result;
    }

    private static boolean[] mask(int length, String shape) {
        boolean[] bits = new boolean[length];
        Random random = new Random(0x4e425f42454e4348L + length);
        for (int i = 0; i < length; i++) {
            bits[i] = switch (shape) {
                case "none" -> false;
                case "all" -> true;
                case "last" -> i == length - 1;
                case "middle_block" -> i >= length / 4 && i < 3 * length / 4;
                case "sparse_1024" -> i % 1024 == 0;
                case "alternating" -> (i & 1) == 0;
                case "random_half" -> random.nextBoolean();
                default -> throw new IllegalArgumentException(shape);
            };
        }
        return bits;
    }

    @SuppressWarnings("unchecked")
    private static M3ArrayList<Object>[] batch(Object[] input, int count) {
        M3ArrayList<Object>[] lists = (M3ArrayList<Object>[]) new M3ArrayList<?>[count];
        List<Object> source = Arrays.asList(input);
        for (int i = 0; i < count; i++) lists[i] = new M3ArrayList<>(source);
        return lists;
    }

    private static void runBatch(M3ArrayList<Object>[] lists, MaskPredicate predicate) {
        long sum = 0;
        for (M3ArrayList<Object> list : lists) {
            predicate.cursor = 0;
            boolean changed = list.removeIf(predicate);
            sum += list.size() + (changed ? 1 : 0);
            if (!list.isEmpty()) sum += list.get(0) == null ? 1 : 2;
        }
        checksum = sum;
    }

    private static double median(double[] values) {
        Arrays.sort(values);
        return values[values.length / 2];
    }

    public static void main(String[] args) {
        if (!THREADS.isCurrentThreadCpuTimeSupported() || !THREADS.isThreadAllocatedMemorySupported()) {
            throw new IllegalStateException("CPU/allocation counters unavailable");
        }
        THREADS.setThreadCpuTimeEnabled(true);
        THREADS.setThreadAllocatedMemoryEnabled(true);
        String variant = args.length == 0 ? "unspecified" : args[0];
        System.out.println("variant\tlength\tshape\toperations\tcpu_ns_per_op_median\twall_ns_per_op_median\tallocated_bytes_per_op_median\tmeasured_gc_events\tsurvivors\tunchanged_prefix\tsurvivor_spans_after_first_reject\tbaseline_ref_assignments_model\trun_copy_refs_model");
        for (int length : new int[]{64, 8192, 65536}) {
            for (String shape : new String[]{"none", "all", "last", "middle_block", "sparse_1024", "alternating", "random_half"}) {
                Object[] input = new Object[length];
                Object marker = new Object();
                Arrays.fill(input, marker);
                boolean[] bits = mask(length, shape);
                MaskPredicate predicate = new MaskPredicate(bits);
                int count = length == 64 ? 512 : 48;
                for (int warmup = 0; warmup < 12; warmup++) runBatch(batch(input, count), predicate);
                double[] cpu = new double[7];
                double[] wall = new double[7];
                double[] alloc = new double[7];
                long collections = 0;
                for (int round = 0; round < 7; round++) {
                    M3ArrayList<Object>[] lists = batch(input, count);
                    long gcBefore = gcCount();
                    long allocatedBefore = THREADS.getThreadAllocatedBytes(THREAD_ID);
                    long cpuBefore = THREADS.getCurrentThreadCpuTime();
                    long start = System.nanoTime();
                    runBatch(lists, predicate);
                    wall[round] = (System.nanoTime() - start) / (double) count;
                    cpu[round] = (THREADS.getCurrentThreadCpuTime() - cpuBefore) / (double) count;
                    alloc[round] = (THREADS.getThreadAllocatedBytes(THREAD_ID) - allocatedBefore) / (double) count;
                    collections += gcCount() - gcBefore;
                }
                int rejected = 0;
                int first = length;
                for (int i = 0; i < length; i++) {
                    if (bits[i]) { rejected++; first = Math.min(first, i); }
                }
                int survivors = length - rejected;
                int spans = 0;
                for (int i = first; i < length; i++) {
                    if (!bits[i] && (i == 0 || bits[i - 1])) spans++;
                }
                int baselineAssignments = rejected == 0 ? 0 : survivors;
                int runReferences = rejected == 0 ? 0 : survivors - first;
                System.out.printf(Locale.ROOT,
                        "%s\t%d\t%s\t%d\t%.3f\t%.3f\t%.3f\t%d\t%d\t%d\t%d\t%d\t%d%n",
                        variant, length, shape, count * 7, median(cpu), median(wall), median(alloc),
                        collections, survivors, first, spans, baselineAssignments, runReferences);
            }
        }
        if (checksum == Long.MIN_VALUE) throw new AssertionError();
    }
}
