name: M3 indexed object position Java JNI receiver
on:
  pull_request:
    paths:
      - 'm3/collections/qualification/m3jdk21-indexed-object-position-20261018/**'
      - 'm3/collections/src/main/java/com/m3/collections/M3IndexedListSnapshot.java'
      - 'm3/collections/src/test/java/com/m3/collections/M3IndexedSnapshotPositionProof.java'
      - 'm3/collections/src/test/java/com/m3/collections/M3IndexedSnapshotPositionTest.java'
      - '.github/workflows/m3-indexed-object-position-20261018.yml'
  workflow_dispatch:
jobs:
  primitive-position-parity:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '21'
      - name: Check three sealed source/test outputs and their mirrored templates
        run: |
          set -eu
          Q=m3/collections/qualification/m3jdk21-indexed-object-position-20261018/recipe
          awk -F '\t' '{print $3 "  " $1}' "$Q/manifest.tsv" | sha256sum -c -
          awk -F '\t' -v q="$Q" '{print $3 "  " q "/" $4}' "$Q/manifest.tsv" | sha256sum -c -
          cmp "$Q/M3IndexedListSnapshot.java.txt" m3/collections/src/main/java/com/m3/collections/M3IndexedListSnapshot.java
      - name: Strict Java21 core plus standalone JDK reference-object contracts
        run: |
          set -eu
          PKG=m3/collections/src/main/java/com/m3/collections
          TST=m3/collections/src/test/java/com/m3/collections
          OUT="$RUNNER_TEMP/m3-positions/classes"
          mkdir -p "$OUT"
          javac --release 21 -proc:none -Xlint:all,-serial -Werror -d "$OUT" \
            "$PKG/M3IndexMaskCompactor.java" "$PKG/M3IndexedListSnapshot.java" \
            "$PKG/M3ArrayList.java" "$PKG/M3GenericCollections.java" \
            "$TST/M3IndexedSnapshotPositionProof.java" "$TST/M3GenericArrayListSelfTest.java"
          java -cp "$OUT" com.m3.collections.M3IndexedSnapshotPositionProof
          java -cp "$OUT" com.m3.collections.M3GenericArrayListSelfTest
      - name: Build real existing C17 JNI and run native-required reference projection
        run: |
          set -eu
          OUT="$RUNNER_TEMP/m3-positions/classes"
          BIN="$RUNNER_TEMP/m3-positions/native"
          mkdir -p "$BIN"
          cc -std=c17 -O1 -g -Wall -Wextra -Werror -fPIC -shared \
            -fsanitize=undefined -fno-sanitize-recover=all \
            -I"$JAVA_HOME/include" -I"$JAVA_HOME/include/linux" \
            m3/collections/src/main/native/collections/m3_index_mask_compactor.c \
            -o "$BIN/libm3_collections_mask.so"
          java -Xcheck:jni -Djava.library.path="$BIN" -cp "$OUT" \
            com.m3.collections.M3IndexedSnapshotPositionProof --native
          java -Xcheck:jni -Djava.library.path="$BIN" -cp "$OUT" \
            com.m3.collections.M3GenericArrayListSelfTest
      - name: Owning Maven JUnit (promotion gate)
        run: mvn -B -ntp -f m3/collections/pom.xml -Dtest=M3IndexedSnapshotPositionTest test
