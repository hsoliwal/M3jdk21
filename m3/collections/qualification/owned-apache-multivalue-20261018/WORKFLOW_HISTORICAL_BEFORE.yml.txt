name: Owned M3 Apache multivalue contracts
on:
  pull_request:
    paths: ['m3/collections/**', '.github/workflows/m3-owned-apache-multivalue.yml']
  workflow_dispatch:
permissions:
  contents: read
jobs:
  contracts:
    runs-on: ubuntu-latest
    timeout-minutes: 15
    steps:
      - uses: actions/checkout@v4
        with:
          fetch-depth: 1
          sparse-checkout: m3/collections
      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '21'
      - name: Compile exact M3 sources and run owned contracts
        shell: bash
        run: |
          set -euo pipefail
          M3_TASK_OUT="$RUNNER_TEMP/m3-apache-multivalue"
          mkdir -p "$M3_TASK_OUT/core" "$M3_TASK_OUT/classes"
          mvn -B -ntp -Dmaven.repo.local="$M3_TASK_OUT/m2" dependency:get -Dartifact=org.apache.commons:commons-collections4:4.5.0
          M3_PROVIDER="$M3_TASK_OUT/m2/org/apache/commons/commons-collections4/4.5.0/commons-collections4-4.5.0.jar"
          python -I -S -B m3/collections/qualification/owned-apache-multivalue-20261018/verify.py --jar "$M3_PROVIDER" --args "$M3_TASK_OUT/owned.args"
          find m3/collections/src/main/java -name '*.java' | sort > "$M3_TASK_OUT/core.args"
          javac --release 21 -encoding UTF-8 -proc:none -Xlint:all,-module -Werror -d "$M3_TASK_OUT/core" @"$M3_TASK_OUT/core.args"
          javac --release 21 -encoding UTF-8 -proc:none -Xlint:all,-module -Werror -cp "$M3_TASK_OUT/core:$M3_PROVIDER" -d "$M3_TASK_OUT/classes" @"$M3_TASK_OUT/owned.args"
          java -Xmx512m -cp "$M3_TASK_OUT/classes:$M3_TASK_OUT/core:$M3_PROVIDER" com.m3.collections.apache.M3MultiValuedMapSelfTest
          java -Xmx512m -cp "$M3_TASK_OUT/classes:$M3_TASK_OUT/core:$M3_PROVIDER" ApacheM3MultiValueCandidateProbe
