on:
  pull_request:
    paths:
      - AGENTS.md
      - m3/collections/src/main/java/com/m3/collections/M3IndexedMultiValueMap.java
      - m3/collections/src/test/java/com/m3/collections/M3IndexedMultiValueOccurrenceSelfTest.java
      - m3/collections/src/test/java/com/m3/collections/M3IndexedMultiValueOccurrenceTest.java
      - m3/collections/qualification/m3-five-library-occurrences-20261018/**
      - .github/workflows/m3-five-library-occurrences-20261018.yml
  workflow_dispatch:
jobs:
  integrated-array-owner:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '21'
      - name: Run actual canonical OpenRewrite recipe Maven JUnit
        run: |
          set -eu
          mvn -B -ntp -f m3/collections/qualification/m3-five-library-occurrences-20261018/recipe/pom.xml verify
      - name: Build M3JDK21 collection module and run real JUnit
        run: |
          set -eu
          mvn -B -ntp -f m3/collections/pom.xml test -Dtest=M3IndexedMultiValueOccurrenceTest -Dsurefire.failIfNoSpecifiedTests=true
      - name: Exact real Java21 M3 owner and native-index C17 differential
        run: |
          set -eu
          CLS="$RUNNER_TEMP/m3-multivalue-classes"
          NATIVE="$RUNNER_TEMP/m3-multivalue-native"
          mkdir -p "$CLS" "$NATIVE"
          find m3/collections/src/main/java -name '*.java' -print > "$RUNNER_TEMP/m3-multivalue-main-sources.txt"
          javac --release 21 -proc:none -Xlint:all,-module -Werror -d "$CLS" @"$RUNNER_TEMP/m3-multivalue-main-sources.txt" \
            m3/collections/src/test/java/com/m3/collections/M3IndexedMultiValueOccurrenceSelfTest.java
          java -cp "$CLS" com.m3.collections.M3IndexedMultiValueOccurrenceSelfTest
          cc -std=c17 -fPIC -shared -O1 -g -Wall -Wextra -Werror \
            -fsanitize=undefined -fno-sanitize-recover=all \
            -I"$JAVA_HOME/include" -I"$JAVA_HOME/include/linux" \
            m3/collections/src/main/native/collections/m3_index_mask_compactor.c \
            -o "$NATIVE/libm3_collections_mask.so"
          java -Xcheck:jni -Djava.library.path="$NATIVE" -cp "$CLS" \
            com.m3.collections.M3IndexedMultiValueOccurrenceSelfTest
