package com.synexia.rewrite;

import static org.junit.jupiter.api.Assertions.assertEquals;
import static org.junit.jupiter.api.Assertions.assertFalse;
import static org.junit.jupiter.api.Assertions.assertNotNull;
import static org.junit.jupiter.api.Assertions.assertThrows;
import static org.junit.jupiter.api.Assertions.assertTrue;

import java.io.IOException;
import java.nio.charset.StandardCharsets;
import java.nio.file.Path;
import java.util.List;
import java.util.Map;
import java.util.TreeMap;
import org.junit.jupiter.api.Test;
import org.openrewrite.InMemoryExecutionContext;
import org.openrewrite.Parser;
import org.openrewrite.SourceFile;
import org.openrewrite.internal.InMemoryLargeSourceSet;
import org.openrewrite.java.JavaParser;

/** Five exact original M3 source postimages recovered without manual output edits. */
final class M3FilteredListHistoryRestore20261017RecipeTest {
    private static final String NAME = "m3jdk21-filtered-list-history-restore-20261017";
    private static final String RESOURCE = "/com/synexia/rewrite/hash-pinned-java/" + NAME + "/";
    private static final String SOURCE =
            "m3/collections/src/main/java/com/m3/collections/M3FilteredLists.java";
    private static final List<String> TARGETS = List.of(
            SOURCE,
            "m3/collections/src/test/java/com/m3/collections/M3FilteredListsDeepCompositionProof.java",
            "m3/collections/src/test/java/com/m3/collections/M3FilteredListsDeepCompositionTest.java",
            "m3/collections/src/test/java/com/m3/collections/M3FilteredListsMaskBridgeProof.java",
            "m3/collections/src/test/java/com/m3/collections/M3FilteredListsReadonlyDepthProof.java",
            "m3/collections/src/test/java/com/m3/collections/M3FilteredListsReadonlyDepthTest.java");

    private static String text(String relative) {
        try (var input = M3FilteredListHistoryRestore20261017RecipeTest.class
                .getResourceAsStream(RESOURCE + relative)) {
            assertNotNull(input, relative);
            return new String(input.readAllBytes(), StandardCharsets.UTF_8);
        } catch (IOException failure) { throw new IllegalStateException(failure); }
    }
    private static InMemoryExecutionContext ctx() {
        return new InMemoryExecutionContext(error -> { throw new AssertionError(error); });
    }
    private static List<SourceFile> parse(String source) {
        return JavaParser.fromJavaVersion().build().parseInputs(
                List.of(Parser.Input.fromString(Path.of(SOURCE), source)),
                null, ctx()).toList();
    }

    @Test void exactHistoricalOwnerAndFourOraclesReachFixedPoint() {
        var recipe = new M3HashPinnedJavaSnapshotRecipe(NAME);
        assertEquals(TARGETS, recipe.targetPaths());
        var pass = recipe.run(new InMemoryLargeSourceSet(
                parse(text("pre/M3FilteredLists.java.txt"))), ctx(), 1);
        assertEquals(6, pass.getChangeset().getAllResults().size());
        Map<String, SourceFile> actual = new TreeMap<>();
        for (var change : pass.getChangeset().getAllResults()) {
            SourceFile output = change.getAfter();
            assertNotNull(output);
            actual.put(output.getSourcePath().toString().replace('\\', '/'), output);
        }
        for (String path : TARGETS) {
            String filename = Path.of(path).getFileName().toString();
            assertNotNull(actual.get(path), path);
            assertEquals(text(filename + ".txt"), actual.get(path).printAll(), path);
        }
        String implementation = actual.get(SOURCE).printAll();
        assertTrue(implementation.contains("private static final class FilterChain<E>"));
        assertTrue(implementation.contains("private static final class FilterTerm<E>"));
        assertTrue(implementation.contains("readOnly.delegate instanceof View<?> nested"));
        assertTrue(implementation.contains("if (iterator == null) iterator = View.this.iterator()"));
        assertTrue(actual.get(TARGETS.get(3)).printAll().contains("Backend.NATIVE_REQUIRED"));
        var second = recipe.run(new InMemoryLargeSourceSet(
                List.copyOf(actual.values())), ctx(), 1);
        assertTrue(second.getChangeset().getAllResults().isEmpty());
    }

    @Test void sourceDriftMustRefuseWithoutPartialResults() {
        var recipe = new M3HashPinnedJavaSnapshotRecipe(NAME);
        String baseline = text("pre/M3FilteredLists.java.txt");
        String drift = baseline.replace(
                "this.predicate = item -> existing.test(item) && filter.test(item);",
                "this.predicate = item -> filter.test(item);");
        assertFalse(drift.equals(baseline));
        assertThrows(RuntimeException.class, () -> recipe.run(
                new InMemoryLargeSourceSet(parse(drift)), ctx(), 1));
    }

    @Test void missingRequiredOwnerMustRefuse() {
        var recipe = new M3HashPinnedJavaSnapshotRecipe(NAME);
        var outsider = JavaParser.fromJavaVersion().build().parseInputs(
                List.of(Parser.Input.fromString(Path.of("m3/collections/src/test/java/com/m3/collections/Other.java"),
                        "package com.m3.collections; public class Other {}")), null, ctx()).toList();
        assertThrows(RuntimeException.class, () -> recipe.run(
                new InMemoryLargeSourceSet(outsider), ctx(), 1));
    }
}
