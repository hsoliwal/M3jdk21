import org.apache.commons.lang3.StringUtils;

/** Prints commons-lang3's answers for the EDGE family on edge inputs, to read the rules off the oracle. */
public final class EdgeProbe {
    private EdgeProbe() {}

    public static void main(String[] args) {
        String[] texts = {"", "a", "\r", "\n", "\r\n", "\n\r", "a\r\n\r\n", "a\n\r", "ab\r", "\r\r", " a  b ",
            "a\u00A0b", "\u00A0a\u00A0", "\t a\n\nb\u000B", "a\u2000b", "a\u3000\u3000b", "\u001Fa\u001C",
            "\u00E9\u00C5\u00E7\u0141\u0142\u0300x", "e\u0301", "\u00F8\u00D8\u0131", "x\uFEFFy"};
        for (String text : texts) {
            System.out.printf("%-22s chomp=%s chop=%s del=%s norm=%s acc=%s strip(null)=%s strip(\"a \")=%s start=%s end=%s%n",
                show(text), show(StringUtils.chomp(text)), show(StringUtils.chop(text)),
                show(StringUtils.deleteWhitespace(text)), show(StringUtils.normalizeSpace(text)),
                show(StringUtils.stripAccents(text)), show(StringUtils.strip(text, null)), show(StringUtils.strip(text, "a ")),
                show(StringUtils.stripStart(text, "a\r")), show(StringUtils.stripEnd(text, "\n\r")));
        }
        System.out.println("strip(\"xa\", \"\")=" + show(StringUtils.strip("xa", "")) + " stripToNull(\" \")="
            + StringUtils.stripToNull(" ") + " trimToNull(\"\\u0001\")=" + StringUtils.trimToNull("\u0001")
            + " defaultString(null)=" + show(StringUtils.defaultString(null)));
    }

    private static String show(String s) {
        if (s == null) return "null";
        StringBuilder out = new StringBuilder("\"");
        for (char c : s.toCharArray()) out.append(c >= 32 && c < 127 ? String.valueOf(c) : String.format("\\u%04X", (int) c));
        return out.append('"').toString();
    }
}
