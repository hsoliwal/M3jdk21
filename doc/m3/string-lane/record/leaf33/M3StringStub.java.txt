package java.lang;

/**
 * Test stub of M3 storage for the stock-JVM proof. With -Dm3.stub.force=true, String.m3() returns
 * one of these for every string, so the M3 paths of the new methods run: their bulk windows read
 * through String.getChars, and factsIfPrepared() returns facts computed with the exact rules of
 * M3StringFacts (ascii, latin1, regexAllClasses, regexAnyClasses) when -Dm3.stub.facts=true.
 */
final class M3String {
    static final boolean FORCE = Boolean.getBoolean("m3.stub.force");
    static final boolean FACTS = Boolean.getBoolean("m3.stub.facts");
    private final String text;

    M3String(String text) {
        this.text = text;
    }

    int length() {
        return text.length();
    }

    M3StringFacts factsIfPrepared() {
        return FACTS ? M3StringFacts.of(text) : null;
    }
}

/** The fields and class rules of M3StringFacts that String reads, computed like M3StringFacts.scan. */
final class M3StringFacts {
    static final int REGEX_DIGIT = 1;
    static final int REGEX_WORD = 1 << 1;
    static final int REGEX_SPACE = 1 << 2;
    static final int REGEX_LOWER = 1 << 3;
    static final int REGEX_UPPER = 1 << 4;
    static final int REGEX_ALPHA = 1 << 5;
    static final int REGEX_ALNUM = 1 << 6;
    static final int REGEX_HEX = 1 << 7;
    static final int REGEX_ASCII = 1 << 8;
    static final int REGEX_LOWER_HEX = 1 << 9;
    static final int REGEX_UPPER_HEX = 1 << 10;
    static final int REGEX_DOT = 1 << 11;
    static final int REGEX_HORIZONTAL = 1 << 12;
    static final int REGEX_VERTICAL = 1 << 13;
    static final int REGEX_CLASSES = (1 << 14) - 1;

    final boolean ascii;
    final boolean latin1;
    final int regexAllClasses;
    final int regexAnyClasses;
    final int stripStart;
    final int stripEnd;
    final char firstUtf16Unit;
    final char lastUtf16Unit;
    final int asciiLowerHash;
    private final String text;

    private M3StringFacts(String text, boolean ascii, boolean latin1, int all, int any) {
        this.text = text;
        this.ascii = ascii;
        this.latin1 = latin1;
        this.regexAllClasses = all;
        this.regexAnyClasses = any;
        int start = 0;
        while (start < text.length() && Character.isWhitespace(text.codePointAt(start))) {
            start += Character.charCount(text.codePointAt(start));
        }
        int end = text.length();
        while (end > start && Character.isWhitespace(text.codePointBefore(end))) {
            end -= Character.charCount(text.codePointBefore(end));
        }
        this.stripStart = start;
        this.stripEnd = end;
        this.firstUtf16Unit = text.isEmpty() ? 0 : text.charAt(0);
        this.lastUtf16Unit = text.isEmpty() ? 0 : text.charAt(text.length() - 1);
        int h = 0;
        for (int i = 0; i < text.length(); i++) {
            char c = text.charAt(i);
            h = 31 * h + (c >= 'A' && c <= 'Z' ? c - 'A' + 'a' : c);
        }
        this.asciiLowerHash = h;
    }

    /** Exact, so a valid answer of the real signal test (which may say true for an absent unit). */
    boolean mayContainCodeUnit(char unit) {
        return text.indexOf(unit) >= 0;
    }

    static M3StringFacts of(String text) {
        boolean ascii = true;
        boolean latin1 = true;
        int all = REGEX_CLASSES;
        int any = 0;
        for (int index = 0; index < text.length(); index++) {
            char unit = text.charAt(index);
            ascii &= unit < 128;
            latin1 &= unit < 256;
            int classes = regexClasses(unit);
            all &= classes;
            any |= classes;
        }
        return new M3StringFacts(text, ascii, latin1, all, any);
    }

    /** Copy of M3StringFacts.regexClasses (M3jdk21 master), with numeric char constants. */
    private static int regexClasses(char unit) {
        if (unit < 128) return asciiRegexClasses(unit);
        boolean vertical = unit == (char) 0x0085 || unit == (char) 0x2028 || unit == (char) 0x2029;
        int classes = vertical ? REGEX_VERTICAL : REGEX_DOT;
        if (unit == (char) 0x00a0 || unit == (char) 0x1680 || unit == (char) 0x180e
                || (unit >= (char) 0x2000 && unit <= (char) 0x200a)
                || unit == (char) 0x202f || unit == (char) 0x205f || unit == (char) 0x3000) {
            classes |= REGEX_HORIZONTAL;
        }
        return classes;
    }

    /** Copy of the table rule of M3StringFacts.asciiRegexClasses (M3jdk21 master). */
    private static int asciiRegexClasses(int unit) {
        int classes = REGEX_ASCII;
        if (unit >= '0' && unit <= '9') {
            classes |= REGEX_DIGIT | REGEX_WORD | REGEX_ALNUM | REGEX_HEX
                    | REGEX_LOWER_HEX | REGEX_UPPER_HEX;
        } else if (unit >= 'a' && unit <= 'z') {
            classes |= REGEX_LOWER | REGEX_ALPHA | REGEX_ALNUM | REGEX_WORD;
            if (unit <= 'f') classes |= REGEX_HEX | REGEX_LOWER_HEX;
        } else if (unit >= 'A' && unit <= 'Z') {
            classes |= REGEX_UPPER | REGEX_ALPHA | REGEX_ALNUM | REGEX_WORD;
            if (unit <= 'F') classes |= REGEX_HEX | REGEX_UPPER_HEX;
        } else if (unit == '_') {
            classes |= REGEX_WORD;
        }
        if (unit == ' ' || (unit >= '\t' && unit <= '\r')) classes |= REGEX_SPACE;
        if (unit == ' ' || unit == '\t') classes |= REGEX_HORIZONTAL;
        if (unit >= '\n' && unit <= '\r') classes |= REGEX_VERTICAL;
        if (unit != '\n' && unit != '\r') classes |= REGEX_DOT;
        return classes;
    }
}
